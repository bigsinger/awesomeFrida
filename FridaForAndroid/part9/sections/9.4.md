# 9.4 枚举模块与导出符号

> 核心要点：系统化的模块枚举和符号枚举是逆向分析的基础。本节深入 traceNative.js 中的枚举框架，展示如何批量收集和处理模块信息。

## 1. traceNative.js 枚举框架分析

```javascript
// 引用：Chap11/traceNative.js 第25-59行
function traceNativeExport() {
    // ① 枚举所有模块
    var modules = Process.enumerateModules();
    for (var i = 0; i < modules.length; i++) {
        var module = modules[i];

        // ② 过滤只关注 libssl.so
        if (module.name.indexOf("libssl.so") < 0) {
            continue;
        }

        // ③ 枚举该模块的所有导出
        var exports = module.enumerateExports();
        console.log('module.addr', module.base);
        
        for (var j = 0; j < exports.length; j++) {
            // ④ 只处理函数类型
            if (exports[j].type == "function") {
                // ⑤ 批量 Hook 所有函数
                attach(exports[j].name, exports[j].address);
            }
        }
    }
}
```

## 2. 按模块来源分类枚举

```javascript
// 针对不同来源的模块执行不同策略
function categorizeModules() {
    var modules = Process.enumerateModules();
    
    var categories = {
        "app": [],
        "system": [],
        "vendor": [],
        "other": []
    };
    
    modules.forEach(function(mod) {
        if (mod.path.indexOf("/data/app/") === 0) {
            categories.app.push(mod);
        } else if (mod.path.indexOf("/system/") === 0) {
            categories.system.push(mod);
        } else if (mod.path.indexOf("/vendor/") === 0) {
            categories.vendor.push(mod);
        } else {
            categories.other.push(mod);
        }
    });
    
    // 输出应用模块的详细信息
    console.log("\n===== App Modules =====");
    categories.app.forEach(function(mod) {
        var exports = mod.enumerateExports();
        var funcCount = exports.filter(function(e) { return e.type === "function"; }).length;
        console.log(mod.name + " @ " + mod.base + " (" + mod.size + " bytes)");
        console.log("  Path: " + mod.path);
        console.log("  Exports: " + exports.length + " (functions: " + funcCount + ")");
    });
    
    return categories;
}
```

## 3. 批量 Hook 所有应用的导出函数

```javascript
// 批量 Hook 应用模块中的所有导出函数
function batchHookAppModules() {
    var modules = Process.enumerateModules();
    var hooked = 0;
    
    modules.forEach(function(mod) {
        // 只处理应用自己的 SO
        if (mod.path.indexOf("/data/app/") < 0) return;
        
        console.log("[*] Processing: " + mod.name);
        var exports = mod.enumerateExports();
        
        exports.forEach(function(exp) {
            if (exp.type !== "function") return;
            
            // 跳过一些过于频繁的无关函数
            if (exp.name.indexOf("JNI_OnLoad") === 0) return;
            
            try {
                Interceptor.attach(exp.address, {
                    onEnter: function(args) {
                        // 输出模块+函数名+参数预览
                        console.log("[" + mod.name + "] " + exp.name + "()");
                    }
                });
                hooked++;
            } catch (e) {
                // 某些函数可能无法 Hook（如 PLT 桩函数）
            }
        });
    });
    
    console.log("[*] Total hooked: " + hooked + " functions");
}

batchHookAppModules();
```

## 4. 枚举符号的高级技巧

### 4.1 按偏移排序

```javascript
// 按偏移排序输出导出表，便于和 IDA 比对
function listExportsSorted(moduleName) {
    var mod = Process.findModuleByName(moduleName);
    if (!mod) return;
    
    var exports = mod.enumerateExports();
    
    // 计算偏移并排序
    var withOffsets = exports.map(function(exp) {
        return {
            name: exp.name,
            type: exp.type,
            address: exp.address,
            offset: exp.address.sub(mod.base)
        };
    });
    
    withOffsets.sort(function(a, b) {
        return a.offset.compare(b.offset);
    });
    
    console.log("=== " + moduleName + " Exports (sorted by offset) ===");
    withOffsets.forEach(function(e) {
        console.log("  0x" + e.offset.toString(16).padStart(8, '0') + 
                   "  " + e.type.padEnd(8) + " " + e.name);
    });
}
```

### 4.2 搜索特定前缀的导出

```javascript
// 搜索所有包含 "encrypt"、"decrypt"、"cipher" 的导出函数
function findCryptographicFunctions() {
    var keywords = ["encrypt", "decrypt", "cipher", "aes", "rsa", "md5", "sha", "hash"];
    var modules = Process.enumerateModules();
    var found = [];
    
    modules.forEach(function(mod) {
        if (mod.path.indexOf("/data/app/") < 0) return;
        
        var exports = mod.enumerateExports();
        exports.forEach(function(exp) {
            if (exp.type !== "function") return;
            
            var shouldCapture = keywords.some(function(kw) {
                return exp.name.toLowerCase().indexOf(kw) >= 0;
            });
            
            if (shouldCapture) {
                var offset = exp.address.sub(mod.base);
                found.push({
                    module: mod.name,
                    name: exp.name,
                    address: exp.address,
                    offset: "0x" + offset.toString(16)
                });
            }
        });
    });
    
    console.log("=== Cryptographic Functions ===");
    found.forEach(function(f) {
        console.log("[" + f.module + "+" + f.offset + "] " + f.name);
    });
}
```

## 5. 构建模块导出缓存

```javascript
// 建立导出函数缓存，避免重复枚举
var exportCache = {};

function getModuleExports(moduleName) {
    if (exportCache[moduleName]) {
        return exportCache[moduleName];
    }
    
    var mod = Process.findModuleByName(moduleName);
    if (!mod) return [];
    
    var exports = mod.enumerateExports();
    var byName = {};
    var byType = { functions: [], variables: [] };
    
    exports.forEach(function(exp) {
        byName[exp.name] = exp;
        if (exp.type === "function") {
            byType.functions.push(exp);
        } else {
            byType.variables.push(exp);
        }
    });
    
    exportCache[moduleName] = {
        module: mod,
        all: exports,
        byName: byName,
        byType: byType
    };
    
    return exportCache[moduleName];
}

// 使用缓存快速查询
function fastFindExport(moduleName, exportName) {
    var cache = getModuleExports(moduleName);
    return cache.byName[exportName] || null;
}
```

## 6. enumerateSymbols 深层枚举

`enumerateSymbols` 比 `enumerateExports` 返回更多符号（包括 LOCAL 符号），适用于深度分析：

```javascript
// 深度枚举：使用 enumerateSymbols 发现隐藏符号
function deepEnumerate(moduleName) {
    var mod = Process.findModuleByName(moduleName);
    if (!mod) return;
    
    var symbols = mod.enumerateSymbols();
    var exports = mod.enumerateExports();
    
    var exportNames = new Set(exports.map(function(e) { return e.name; }));
    
    // 找出 exports 中没有的符号
    var hiddenSymbols = symbols.filter(function(s) {
        return !exportNames.has(s.name) && s.type === "function";
    });
    
    console.log("=== " + moduleName + " Hidden Symbols (" + hiddenSymbols.length + ") ===");
    hiddenSymbols.slice(0, 30).forEach(function(s) {
        console.log("  " + s.name + " @ " + s.address + 
                   " (offset: 0x" + s.address.sub(mod.base).toString(16) + ")");
    });
}
```

## 7. 综合枚举输出工具

```javascript
// 完整的枚举模块信息输出工具
function dumpAllModules(showExports) {
    var modules = Process.enumerateModules();
    var output = [];
    
    modules.forEach(function(mod) {
        var entry = {
            name: mod.name,
            base: mod.base.toString(),
            size: ptr(mod.size).toString(),
            path: mod.path,
            exportCount: 0,
            functionCount: 0,
            exports: []
        };
        
        if (showExports) {
            var exports = mod.enumerateExports();
            entry.exportCount = exports.length;
            entry.functionCount = exports.filter(function(e) { 
                return e.type === "function"; 
            }).length;
            
            exports.forEach(function(exp) {
                var offset = exp.address.sub(mod.base);
                entry.exports.push({
                    offset: "0x" + offset.toString(16),
                    type: exp.type,
                    name: exp.name
                });
            });
        }
        
        output.push(entry);
    });
    
    // 排序：应用模块优先
    output.sort(function(a, b) {
        var aIsApp = a.path.indexOf("/data/app/") >= 0 ? 0 : 1;
        var bIsApp = b.path.indexOf("/data/app/") >= 0 ? 0 : 1;
        return aIsApp - bIsApp;
    });
    
    console.log(JSON.stringify(output, null, 2));
}

// 使用：dumpAllModules(true);
```

## 总结

- **3 层枚举**：`Process.enumerateModules()` → `module.enumerateExports()` → 逐个函数处理
- **模块过滤**：按路径（app/system/vendor）或名称模式过滤目标
- **偏移计算**：`exp.address.sub(mod.base)` 获取函数在 SO 中的偏移
- **enumerateExports vs enumerateSymbols**：前者快速（仅 GLOBAL），后者全面（含 LOCAL）
- **缓存机制**：建立导出缓存避免重复枚举
- **关键词搜索**：在枚举结果中搜索加密/网络/文件操作等关键词定位目标
