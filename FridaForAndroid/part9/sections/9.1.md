# 9.1 dlopen动态拦截与延迟Hook

> 核心要点：当 Frida 附加到进程时，目标 SO 可能尚未加载。延迟 Hook 技术通过在 dlopen 返回时立即安装 Hook，解决了"注入时 SO 未加载"的问题。

## 1. 问题的提出

### 1.1 时序问题

```
Frida 注入脚本（进程启动后）
    ↓
此时目标 SO 可能还未被 Java 代码调用 System.loadLibrary
    ↓
直接 Hook 会失败（函数不存在）
    ↓
需要等待 SO 加载后再 Hook
```

### 1.2 解决思路

```
Hook dlopen / android_dlopen_ext
    ↓
监控加载的 SO 名称
    ↓
当目标 SO 被加载时，在 onLeave 中安装 Hook
    ↓
此时 SO 已加载，但 JNI_OnLoad 还未执行
```

## 2. 基础延迟 Hook：dlopen.js

```javascript
// 引用：awesomeFrida/dlopen.js（完整代码35行）
const TARGET_LIB_NAME = "libmsaoaidsec.so";

function hook_dlopen() {
    // 同时 Hook dlopen 和 android_dlopen_ext
    ["android_dlopen_ext", "dlopen"].forEach(funcName => {
        let addr = Module.findExportByName(null, funcName);
        if (addr) {
            Interceptor.attach(addr, {
                onEnter(args) {
                    let libName = ptr(args[0]).readCString();
                    if (libName) {
                        console.log(`[+] ${funcName} onEnter: ${libName}`);
                    }
                },
                onLeave: function (retval) {
                    console.log(`[+] ${funcName} onLeave`);
                }
            });
        }
    });
}

hook_dlopen();
```

**核心技巧：**

1. **双 Hook**：同时 Hook `dlopen` 和 `android_dlopen_ext` 确保不遗漏
2. **`Module.findExportByName(null, funcName)`**：`null` 表示在所有模块中搜索
3. **`ptr(args[0]).readCString()`**：从第一个参数中读取 SO 路径

## 3. 延迟 Hook + JNI_OnLoad 拦截

```javascript
// 引用：awesomeFrida/JNI_OnLoad.js（综合第15-69行）
const TARGET_LIB_NAME = "libmsaoaidsec.so";

function hook_dlopen() {
    ["android_dlopen_ext", "dlopen"].forEach(funcName => {
        let addr = Module.findExportByName(null, funcName);
        if (addr) {
            Interceptor.attach(addr, {
                onEnter(args) {
                    let libName = ptr(args[0]).readCString();
                    if (libName && libName.indexOf(TARGET_LIB_NAME) >= 0) {
                        this.is_can_hook = true;
                        console.log(`[+] ${funcName} onEnter: ${libName}`);
                    }
                },
                onLeave: function (retval) {
                    if (this.is_can_hook) {
                        console.log(`[+] ${funcName} onLeave, start hook JNI_OnLoad`);
                        hook_JNI_OnLoad();  // ← 关键：在 onLeave 中安装 Hook
                    }
                }
            });
        }
    });
}
```

**关键时序点：**
- `onEnter`：SO 还未被加载，仅能读取路径
- `onLeave`：SO 已加载（`dlopen` 返回），`JNI_OnLoad` 还未执行
- 在 `onLeave` 中安装的 Hook 可以捕获 `JNI_OnLoad` 调用

## 4. SO Dump + 延迟 Hook 组合

```javascript
// 引用：awesomeFrida/dump-so.js（完整代码51行）
// 在目标 SO 加载时立即 dump 到文件
const TARGET_LIB_NAME = "libmsaoaidsec.so";
var TargetLibModule = null;

function dump_so(so_name) {
    var module = Process.getModuleByName(so_name);
    console.log("[name]:", module.name);
    console.log("[base]:", module.base);
    console.log("[size]:", ptr(module.size));
    console.log("[path]:", module.path);

    var path = "/data/local/tmp/" + module.name + 
               "_" + module.base + "_" + ptr(module.size) + ".so";
    var handle = new File(path, "wb");
    if (handle && handle != null) {
        Memory.protect(ptr(module.base), module.size, 'rwx');
        var buffer = ptr(module.base).readByteArray(module.size);
        handle.write(buffer);
        handle.flush();
        handle.close();
        console.log("[dump]:", path);
    }
}

function hook_dlopen() {
    ["android_dlopen_ext", "dlopen"].forEach(funcName => {
        let addr = Module.findExportByName(null, funcName);
        if (addr) {
            Interceptor.attach(addr, {
                onEnter(args) {
                    let libName = ptr(args[0]).readCString();
                    if (libName && libName.indexOf(TARGET_LIB_NAME) >= 0) {
                        this.canHook = true;
                    }
                },
                onLeave: function (retval) {
                    if (this.canHook) {
                        dump_so(TARGET_LIB_NAME);  // ← dump 完整 SO
                    }
                }
            });
        }
    });
}

hook_dlopen();
```

**dump_so 技术要点：**

1. `Process.getModuleByName(so_name)` — 获取模块信息
2. `Memory.protect(ptr(module.base), module.size, 'rwx')` — 确保内存可读
3. `ptr(module.base).readByteArray(module.size)` — 读取完整 SO 到字节数组
4. `new File(path, "wb")` — Frida 的文件写入 API

## 5. 通用延迟 Hook 框架

```javascript
// 通用的延迟 Hook 框架
function wait_and_hook(targetLibName, initHooks) {
    // 1. 先检查目标是否已加载
    var mod = Process.findModuleByName(targetLibName);
    if (mod) {
        console.log("[+] " + targetLibName + " already loaded, hooking now");
        initHooks(mod);
        return;
    }
    
    // 2. 未加载，启动延迟 Hook
    console.log("[*] " + targetLibName + " not loaded, waiting for dlopen...");
    
    var dlopenHooks = [];
    
    ["dlopen", "android_dlopen_ext"].forEach(function(funcName) {
        var addr = Module.findExportByName(null, funcName);
        if (!addr) return;
        
        var listener = Interceptor.attach(addr, {
            onEnter: function(args) {
                var libName = ptr(args[0]).readCString();
                if (libName && libName.indexOf(targetLibName) >= 0) {
                    this.found = true;
                    console.log("[+] Detected dlopen for " + libName);
                }
            },
            onLeave: function(retval) {
                if (this.found && retval.toInt32() !== 0) {
                    console.log("[+] " + targetLibName + " loaded successfully");
                    
                    // 延迟一小段时间确保完全初始化
                    setTimeout(function() {
                        var mod = Process.findModuleByName(targetLibName);
                        if (mod) {
                            initHooks(mod);
                            // 移除 dlopen Hook，减少干扰
                            dlopenHooks.forEach(function(h) { h.detach(); });
                        }
                    }, 100);
                }
            }
        });
        
        dlopenHooks.push(listener);
    });
}

// 使用示例
wait_and_hook("libtarget.so", function(mod) {
    console.log("[*] Installing hooks on " + mod.name + " @ " + mod.base);
    
    // 在这里安装所有目标函数 Hook
    Interceptor.attach(mod.base.add(0x1234), {
        onEnter: function(args) {
            console.log("[target func] called");
        }
    });
});
```

## 6. Java 层 + Native 层双层拦截

```javascript
// Java 层拦截 System.loadLibrary + Native 层 dlopen 拦截
function dual_layer_hook(targetLibName, onLoaded) {
    // Native 层
    var nativeHooks = [];
    ["dlopen", "android_dlopen_ext"].forEach(function(funcName) {
        var addr = Module.findExportByName(null, funcName);
        if (addr) {
            var h = Interceptor.attach(addr, {
                onLeave: function(retval) {
                    if (this.isTarget && retval.toInt32() !== 0) {
                        setTimeout(function() {
                            onLoaded(targetLibName);
                        }, 50);
                    }
                }
            });
            nativeHooks.push(h);
        }
    });
    
    // Java 层
    Java.perform(function() {
        var System = Java.use("java.lang.System");
        System.loadLibrary.implementation = function(libName) {
            var result = this.loadLibrary(libName);
            if (libName.indexOf(targetLibName) >= 0) {
                console.log("[Java] System.loadLibrary(" + libName + ") completed");
                setTimeout(function() {
                    onLoaded(targetLibName);
                }, 50);
            }
            return result;
        };
    });
}
```

## 7. 延迟 Hook 的常见问题

### 7.1 竞态条件

```javascript
// dlopen onLeave 时 SO 可能还没有完全初始化
// 如果直接 Hook 某些需要初始化后才能访问的函数可能失败

// 解决方案1：添加短暂延迟
setTimeout(function() {
    initHooks(mod);
}, 200);  // 200ms 后执行

// 解决方案2：利用 call_constructors 信号
// 见 9.6 节 init_array 追踪
```

### 7.2 重复 Hook

```javascript
// 防止同一个函数被多次 Hook
var hookedSet = new Set();

function safeHook(address, callbacks) {
    var key = address.toString();
    if (hookedSet.has(key)) {
        console.log("[!] Already hooked " + key);
        return null;
    }
    
    hookedSet.add(key);
    return Interceptor.attach(address, {
        onEnter: function(args) {
            if (callbacks.onEnter) callbacks.onEnter.call(this, args);
        },
        onLeave: function(retval) {
            if (callbacks.onLeave) callbacks.onLeave.call(this, retval);
        }
    });
}
```

## 总结

- **延迟 Hook 解决的核心问题**：Frida 注入时目标 SO 尚未加载
- **双 Hook 策略**：同时 Hook `dlopen` + `android_dlopen_ext` 确保不遗漏
- **关键时序**：`dlopen` onLeave → SO 已加载，`JNI_OnLoad` 尚未执行
- **结合 dump_so**：在加载瞬间 dump SO 到文件，用于后续 IDA 静态分析
- **Java+Native 双层拦截**：`System.loadLibrary` + `dlopen` 覆盖面更广
- **竞态处理**：用 `setTimeout` 或 `call_constructors` Hook 确保初始化完成
