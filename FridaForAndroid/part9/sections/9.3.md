# 9.3 dlopen/dlsym符号解析监控

> 核心要点：dlsym 是动态查找符号的 API，SO 通过它解析外部依赖符号。监控 dlsym 调用可以了解目标库的符号依赖关系和使用了哪些外部函数。

## 1. dlsym 函数签名

```c
// dlsym 标准接口
void *dlsym(void *handle, const char *symbol);

// 参数：
//   handle: dlopen 返回的句柄（或 RTLD_DEFAULT、RTLD_NEXT）
//   symbol: 要查找的符号名称
// 返回值：
//   符号的地址，失败返回 NULL
```

**特殊 handle 值：**

| 值 | 含义 |
|----|------|
| `RTLD_DEFAULT` (0) | 按默认搜索顺序在所有已加载库中查找 |
| `RTLD_NEXT` (-1) | 从当前调用者的下一优先级库开始查找 |

## 2. 完整代码分析：dlsym.js

```javascript
// 引用：awesomeFrida/dlsym.js（完整代码100行）
var libMap = {};  // 存储库的基地址 -> 库名称映射

// 1. 记录已加载库的基地址
function load_so_load() {
    var dlopen = Module.findExportByName(null, "dlopen");
    var android_dlopen_ext = Module.findExportByName(null, "android_dlopen_ext");

    // Hook dlopen
    Interceptor.attach(dlopen, {
        onEnter: function(args) {
            var path_ptr = args[0];
            var path = ptr(path_ptr).readCString();
            this.path = path;
            console.log("[+] dlopen onEnter: ", path);
        },
        onLeave: function(retval) {
            if (retval.toInt32() !== 0) {  // 加载成功
                var baseAddr = ptr(retval);
                libMap[baseAddr] = this.path;  // 建立 handle → 路径映射
                console.log("[+] dlopen onLeave: " + this.path + " at " + baseAddr);
            }
        }
    });

    // Hook android_dlopen_ext
    Interceptor.attach(android_dlopen_ext, {
        onEnter: function(args) {
            var path_ptr = args[0];
            var path = ptr(path_ptr).readCString();
            this.path = path;
            console.log("[+] android_dlopen_ext onEnter: ", path);
        },
        onLeave: function(retval) {
            if (retval.toInt32() !== 0) {
                var baseAddr = ptr(retval);
                libMap[baseAddr] = this.path;
                console.log("[+] android_dlopen_ext onLeave: " + this.path + " at " + baseAddr);
            }
        }
    });
}

// 2. Hook dlsym 解析符号并输出库名
function hook_dlsym() {
    var dlsym = Module.findExportByName(null, "dlsym");
    if (dlsym !== null) {
        Interceptor.attach(dlsym, {
            onEnter: function(args) {
                this.handle = args[0];                     // 记录库的 handle
                this.symbol = ptr(args[1]).readCString();  // 读取符号名

                // 从 libMap 中查找对应的库名
                this.libName = "Unknown";
                if (libMap[this.handle]) {
                    this.libName = libMap[this.handle];
                } else if (this.handle.toInt32() === 0) {
                    this.libName = "RTLD_DEFAULT";
                }
            },
            onLeave: function(retval) {
                var status = retval.isNull() ? "NOT FOUND" : retval.toString();
                console.log("\t[+] dlsym: " + this.symbol + 
                          " (from " + this.libName + ") -> " + status);
            }
        });
    }
}

console.log("[+] hook dlopen android_dlopen_ext dlsym--------");
load_so_load();
hook_dlsym();
```

### 2.1 执行流程

```
进程启动，SO 被加载
    ↓
dlopen onEnter → 记录路径
    ↓
dlopen onLeave → 记录 handle → 路径 映射
    ↓
目标 SO 执行代码
    ↓
调用 dlsym(handle, "SSL_write")
    ↓
dlsym onEnter → 读取符号名 "SSL_write"
    ↓
dlsym onLeave → 输出解析结果
```

```

[+] dlopen onLeave: libssl.so at 0x7b8f0000

[+] dlopen onLeave: libcrypto.so at 0x7b9a0000
    
    [后来，当目标 SO 通过 dlsym 查找符号时：]
    [+] dlsym: SSL_write (from libssl.so) -> 0x7b8f5678
```

## 3. dlsym 监控与符号注入

通过拦截 `dlsym` 的返回值，我们可以注入自己的"伪符号"：

```javascript
// 符号注入：拦截 dlsym 返回伪地址
function hook_dlsym_inject() {
    var dlsym = Module.findExportByName(null, "dlsym");
    if (!dlsym) return;
    
    // 构造一个假的函数
    var fakeFunc = Memory.alloc(4096);
    Memory.protect(fakeFunc, 4096, 'rwx');
    Memory.patchCode(fakeFunc, 4, function(code) {
        var cw = new Arm64Writer(code, { pc: fakeFunc });
        cw.putRet();  // 直接返回
        cw.flush();
    });
    
    Interceptor.attach(dlsym, {
        onEnter: function(args) {
            this.symbol = ptr(args[1]).readCString();
        },
        onLeave: function(retval) {
            // 当目标 SO 查找特定符号时，替换返回值
            if (this.symbol === "pthread_create") {
                console.log("[!] Replacing dlsym result for " + this.symbol);
                console.log("[!] Original: " + retval + " -> Fake: " + fakeFunc);
                retval.replace(fakeFunc);
            }
        }
    });
}
```

## 4. System.loadLibrary Java 层监控

结合 Java 层的 `System.loadLibrary` 可以构建更完整的加载监控：

```javascript
// 引用：awesomeFrida/System.loadLibrary.js（完整代码43行）
function hook_System_loadLibrary() {
    Java.perform(function() {
        var System = Java.use('java.lang.System');
        var Runtime = Java.use('java.lang.Runtime');
        var VMStack = Java.use('dalvik.system.VMStack');

        System.loadLibrary.implementation = function(libName) {
            console.log('\n[System.loadLibrary] "' + libName + '"');
            
            // 打印调用栈
            var trace = Java.use("android.util.Log")
                .getStackTraceString(Java.use("java.lang.Exception").$new());
            console.log(trace);
            
            // 调用原来的方法
            return this.loadLibrary(libName);
        };
    });
}
```

## 5. 综合监控框架：全链路追踪

```javascript
// 全链路监控：Java 层 → Native 层 dlopen → dlsym
function full_link_tracer() {
    var libMap = {};
    
    // 1. Java 层监控
    Java.perform(function() {
        var System = Java.use('java.lang.System');
        System.loadLibrary.implementation = function(libName) {
            console.log("\n========== System.loadLibrary(\"" + libName + "\") ==========");
            var result = this.loadLibrary(libName);
            
            // 延迟一下等待 native 层加载完成
            setTimeout(function() {
                var mod = Process.findModuleByName("lib" + libName + ".so");
                if (mod) {
                    console.log("[Native] " + mod.name + " loaded @ " + mod.base);
                    console.log("[Native] Size: " + ptr(mod.size));
                    
                    var exports = mod.enumerateExports();
                    console.log("[Native] Exports: " + exports.length);
                    exports.slice(0, 10).forEach(function(exp) {
                        if (exp.type === "function") {
                            console.log("  " + exp.name);
                        }
                    });
                }
            }, 500);
            
            return result;
        };
    });
    
    // 2. dlopen 监控
    ["dlopen", "android_dlopen_ext"].forEach(function(funcName) {
        var addr = Module.findExportByName(null, funcName);
        if (addr) {
            Interceptor.attach(addr, {
                onEnter: function(args) {
                    this.path = ptr(args[0]).readCString();
                },
                onLeave: function(retval) {
                    if (retval.toInt32() !== 0 && this.path) {
                        libMap[retval.toString()] = this.path;
                        console.log("[dlopen] " + this.path + " handle=" + retval);
                    }
                }
            });
        }
    });
    
    // 3. dlsym 监控
    var dlsym = Module.findExportByName(null, "dlsym");
    if (dlsym) {
        Interceptor.attach(dlsym, {
            onEnter: function(args) {
                this.handle = args[0];
                this.symbol = ptr(args[1]).readCString();
            },
            onLeave: function(retval) {
                var libName = libMap[this.handle.toString()] || 
                             (this.handle.toInt32() === 0 ? "RTLD_DEFAULT" : "Unknown");
                
                if (retval.isNull()) {
                    console.log("[dlsym] " + libName + " -> " + this.symbol + " : NOT FOUND");
                } else {
                    console.log("[dlsym] " + libName + " -> " + this.symbol + " @ " + retval);
                }
            }
        });
    }
}

setTimeout(full_link_tracer, 1000);
```

## 6. 使用场景分析

### 6.1 发现加固库的依赖

```javascript
// 捕获加固 SO 在 dlsym 中查询的所有符号
// 这有助于了解加固库使用了哪些系统功能

var dlsymRequests = {};

Interceptor.attach(dlsym, {
    onEnter: function(args) {
        var symbol = ptr(args[1]).readCString();
        if (!dlsymRequests[symbol]) {
            dlsymRequests[symbol] = { count: 0, callers: [] };
        }
        dlsymRequests[symbol].count++;
        
        // 记录调用者
        var caller = this.context.lr;
        var mod = Process.findModuleByAddress(caller);
        if (mod) {
            var callerName = mod.name + "+0x" + caller.sub(mod.base).toString(16);
            if (dlsymRequests[symbol].callers.indexOf(callerName) < 0) {
                dlsymRequests[symbol].callers.push(callerName);
            }
        }
    }
});

// 定期输出统计
setInterval(function() {
    console.log("\n=== dlsym Request Statistics ===");
    Object.keys(dlsymRequests).sort(function(a, b) {
        return dlsymRequests[b].count - dlsymRequests[a].count;
    }).slice(0, 20).forEach(function(sym) {
        console.log("  " + sym + ": " + dlsymRequests[sym].count + " times");
    });
}, 5000);
```

## 7. 注意事项

### 7.1 dlsym 与 dlopen 的关联

```javascript
// dlsym 的 handle 必须与 dlopen 返回的一致
// libMap 存储 handle → path 的映射是关键
// 注意：RTLD_DEFAULT (0) 和 RTLD_NEXT (-1) 是特殊值
```

### 7.2 性能影响

```javascript
// dlsym 可能被频繁调用（尤其是使用 RTLD_DEFAULT 的场景）
// 如果性能受影响，可以加上频率限制

var lastLog = {};
var THROTTLE_MS = 1000;  // 同一符号每秒最多输出一次

Interceptor.attach(dlsym, {
    onEnter: function(args) {
        var symbol = ptr(args[1]).readCString();
        var now = Date.now();
        
        if (!lastLog[symbol] || now - lastLog[symbol] > THROTTLE_MS) {
            lastLog[symbol] = now;
            console.log("[dlsym] " + symbol);
        }
    }
});
```

## 总结

- **dlsym Hook** 可以监控 SO 加载后的所有符号解析行为
- **handle → 路径映射**：通过 Hook `dlopen` onLeave 建立映射表是关键步骤
- **RTLD_DEFAULT/RTLD_NEXT**：特殊 handle 值，需要特殊处理
- **符号注入**：通过修改 `dlsym` 返回值，可以替换目标 SO 使用的函数
- **Java+Native 全链路**：`System.loadLibrary` + `dlopen` + `dlsym` 覆盖完整的加载-解析链路
- **应用场景**：了解加固库依赖、检测反调试符号查询、符号注入绕过
