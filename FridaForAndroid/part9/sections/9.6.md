# 9.6 init_array SO初始化函数追踪

> 核心要点：`init_array` 是 ELF SO 加载时自动执行的构造函数数组，加固方案常在此设反调试和完整性校验。本节通过两种方案——基础版 hook `call_constructors` 和高级版 CModule 解析 `soinfo`——实现 init_array 的全面追踪。

## 1. init_array 段与逆向意义

在 ELF 文件结构中，`.init_array` 段存储了一组函数指针数组。当 linker 加载 SO 时，会按顺序依次调用这些函数：

```
SO 加载流程：
  dlopen / android_dlopen_ext
    → linker 解析 ELF 结构
    → call_constructors()         ← 执行 .init_array 中的函数
      → init_array[0]()           ← 第一个构造函数
      → init_array[1]()           ← 第二个构造函数
      → ...
    → JNI_OnLoad()               ← 在 init_array 之后调用
```

**逆向意义：** 加固厂商（360、腾讯、爱加密等）广泛利用 `init_array` 执行反调试检测、环境校验、线程监控等敏感操作，原因如下：

1. **执行时机早** — 在 `JNI_OnLoad` 之前执行，普通 Hook 脚本还未注入
2. **隐蔽性强** — 函数地址隐藏在 soinfo 结构体中，常规符号表无法看到
3. **代码分离** — 将反调试逻辑从 JNI_OnLoad 剥离到 init_array，增加分析难度

## 2. 方案一：基础版 — Hook call_constructors

最简单的方案是直接 Hook linker 内部的 `call_constructors` 函数。当任意 SO 的构造函数被调用时，即可捕获。

```javascript
// 引用：awesomeFrida/call_constructors.js（完整60行代码）
const TARGET_LIB_NAME = "libmsaoaidsec.so";

function find_call_constructors() {
  is64Bit = Process.pointerSize === 8;
  var linkerModule = Process.getModuleByName(is64Bit ? "linker64" : "linker");
  var symbols = linkerModule.enumerateSymbols();
  for (var i = 0; i < symbols.length; i++) {
    if (symbols[i].name.indexOf('call_constructors') > 0) {
      console.warn(`call_constructors symbol name: ${symbols[i].name} address: ${symbols[i].address}`);
      return symbols[i].address;
    }
  }
}

function hook_call_constructors() {
  var ptr_call_constructors = find_call_constructors();
  var listener = Interceptor.attach(ptr_call_constructors, {
    onEnter: function (args) {
      console.warn(`call_constructors onEnter`);
      listener.detach();   // 只 Hook 一次，减少干扰
    },
  })
}

function hook_dlopen() {
  ["android_dlopen_ext", "dlopen"].forEach(funcName => {
    let addr = Module.findExportByName(null, funcName);
    if (addr) {
      Interceptor.attach(addr, {
        onEnter(args) {
          let libName = ptr(args[0]).readCString();
          if (libName && libName.indexOf(TARGET_LIB_NAME) >= 0) {
            hook_call_constructors();   // 目标SO加载时安装Hook
          }
        },
      });
    }
  });
}

var is64Bit = Process.pointerSize === 8;
hook_dlopen()
```

**方案一的核心思路：**

1. `hook_dlopen()` 监控 `dlopen` / `android_dlopen_ext`，检测目标 SO 加载
2. 检测到目标时调用 `hook_call_constructors()`，在 `call_constructors` 执行前安装拦截
3. 拦截后立即 `detach()`，避免干扰后续其它 SO 的构造函数

**局限：** 基础版只能检测到 `call_constructors` 被调用，无法获取具体的 `init_array` 函数地址和数量。

## 3. 方案二：高级版 — CModule 解析 soinfo

为了获取 `init_array` 的完整信息，需要直接解析 linker 内部的 `soinfo` 结构体。Frida 的 **CModule** 功能允许在脚本中编译和执行原生 C 代码，可在目标进程中直接操作内存结构。

```javascript
// 引用：awesomeFrida/init_array.js（完整275行代码，第1-205行）
// CModule 中定义 soinfo 结构体和辅助函数

let cm_code = `
#if defined(__LP64__)
#define USE_RELA 1
#endif

#if defined(__LP64__)
#define ElfW(type) Elf64_ ## type
#else
#define ElfW(type) Elf32_ ## type
#endif

// ...（省略重复的 ELF 类型定义）...

// soinfo 结构体（关键部分）
typedef struct {
    const ElfW(Phdr)* phdr;
    size_t phnum;
    ElfW(Addr) base;
    size_t size;
    ElfW(Dyn)* dynamic;
    void* next;
    uint32_t flags_;
    const char* strtab_;
    ElfW(Sym)* symtab_;
    size_t nbucket_;
    size_t nchain_;
    uint32_t* bucket_;
    uint32_t* chain_;

    // 重定位表
    ElfW(Rela)* plt_rela_;
    size_t plt_rela_count_;
    ElfW(Rela)* rela_;
    size_t rela_count_;

    // ===== 关键字段：init_array 信息 =====
    linker_ctor_function_t* preinit_array_;
    size_t preinit_array_count_;
    linker_ctor_function_t* init_array_;      // ← init_array 函数指针数组
    size_t init_array_count_;                  // ← init_array 函数数量
    linker_dtor_function_t* fini_array_;
    size_t fini_array_count_;
    linker_ctor_function_t init_func_;         // ← .init 函数（单个构造函数）
    linker_dtor_function_t fini_func_;
} soinfo;

// 回调函数：从 C 代码将数据传回 JavaScript
void tell_init_info(soinfo* ptr, void (*cb)(int, void*, void*)) {
    cb(ptr->init_array_count_, ptr->init_array_, ptr->init_func_);
}
`;

// CModule 初始化和 Hook 部分
function setup_cmodule() {
  if (Process.pointerSize == 4) {
    cm_code = cm_include + "#define __work_around_b_24465209__ 1" + cm_code;
  } else {
    cm_code = cm_include + "#define __LP64__ 1" + cm_code;
  }
  cm = new CModule(cm_code, {});
  tell_init_info = new NativeFunction(cm.tell_init_info, "void", ["pointer", "pointer"]);
}

function hook_call_constructors() {
  // 在 linker/linker64 中找到 call_constructors 和 get_soname
  let linker = Process.findModuleByName(
    Process.pointerSize == 4 ? "linker" : "linker64"
  );
  let symbols = linker.enumerateSymbols();

  for (let symbol of symbols) {
    if (symbol.name == "__dl__ZN6soinfo17call_constructorsEv") {
      call_constructors_addr = symbol.address;
    } else if (symbol.name == "__dl__ZNK6soinfo10get_sonameEv") {
      get_soname = new NativeFunction(symbol.address, "pointer", ["pointer"]);
    }
  }

  if (call_constructors_addr && get_soname) {
    Interceptor.attach(call_constructors_addr, {
      onEnter: function (args) {
        let soinfo = args[0];
        let soname = get_soname(soinfo).readCString();

        // 通过 CModule 回调获取 init_array 信息
        tell_init_info(soinfo, new NativeCallback((count, init_array_ptr, init_func) => {
          console.log();
          console.log(`[call_constructors] ${soname} count:${count}`);
          console.log(`[call_constructors] init_array_ptr:${init_array_ptr}`);
          console.log(`[call_constructors] init_func:${init_func}`);

          // 遍历所有 init_array 函数
          for (let index = 0; index < count; index++) {
            let init_array_func = init_array_ptr
              .add(Process.pointerSize * index).readPointer();
            let mm = new ModuleMap();
            let info = mm.find(init_array_func);
            let func_info = info == null ? "null" :
              `[${info.name} + ${init_array_func.sub(info.base)}]`;
            console.log(`[call_constructors] init_array:${index} ${init_array_func} -> ${func_info}`);
          }
        }, "void", ["int", "pointer", "pointer"]));
      }
    });
  }
}

function main() {
  setup_cmodule();
  hook_call_constructors();
}

setImmediate(main);
```

**方案二的关键技术：**

| 技术 | 说明 |
|------|------|
| **CModule** | 在 Frida 脚本中编译并执行 C 代码，直接操作目标进程内存 |
| **soinfo 结构体** | linker 内部用于管理 SO 的核心结构，包含 `init_array_` 和 `init_array_count_` 字段 |
| **tell_init_info 回调** | C 代码调用 JavaScript 回调，将结构体数据传回 |
| **__dl__ZN6soinfo...** | Android linker 导出符号的 C++ mangled name，用于定位函数地址 |
| **ModuleMap** | 将函数地址映射到 `[模块名 + 偏移]` 格式，便于分析 |

## 4. 运行效果

使用该脚本注入目标应用后，输出示例如下：

```
[call_constructors] libmsaoaidsec.so count:3
[call_constructors] init_array_ptr:0x7b8c2e4000
[call_constructors] init_func:0x7b8c2e4120 -> [libmsaoaidsec.so + 0x120]
[call_constructors] init_array:0 0x7b8c2e4500 -> [libmsaoaidsec.so + 0x500]
[call_constructors] init_array:1 0x7b8c2e4a00 -> [libmsaoaidsec.so + 0xa00]
[call_constructors] init_array:2 0x7b8c2e4e00 -> [libmsaoaidsec.so + 0xe00]

[call_constructors] libc.so count:1
[call_constructors] init_array:0 0x7a00012300 -> [libc.so + 0x12300]
```

从输出可以清晰地看到：
- **libmsaoaidsec.so** 有 3 个 init_array 函数，偏移分别为 `+0x500`、`+0xa00`、`+0xe00`
- 还有 1 个 `.init` 函数在 `+0x120`
- 系统库 `libc.so` 也有自己的 init_array 函数

## 5. 典型应用场景

### 场景一：发现加固的反调试入口

```bash
# 注入目标应用，输出所有 SO 的 init_array
frida -U -f com.target.app -l init_array.js -o init_trace.log

# 分析日志，重点关注 non-system SO
grep -v "libc\|libm\|libdl\|libutils" init_trace.log
```

加固库的 init_array 中的函数通常包含：
- **环境检测** — 检查 TracerPid、调试端口、frida-server 进程
- **完整性校验** — 计算自身代码段的 hash 值
- **线程监控** — `pthread_create` 创建守护线程
- **反 Hook 检测** — 检测 inline hook、PLT hook 等痕迹

### 场景二：结合 9.5 节的反调试绕过

发现 init_array 中的反调试函数后，可以在 `call_constructors` 的 `onEnter` 中安装 `pthread_create` 拦截：

```javascript
Interceptor.attach(call_constructors_addr, {
  onEnter: function (args) {
    let soinfo = args[0];
    let soname = get_soname(soinfo).readCString();
    if (soname.indexOf("libmsaoaidsec") >= 0) {
      hook_pthread_create();  // ← 在 init_array 执行前安装线程拦截
    }
  }
});
```

### 场景三：批量符号收集

将捕获的 init_array 函数地址与其所属模块关联，建立完整的 SO 初始化调用图谱：

```
libmsaoaidsec.so
├── init_array[0] → anti_debug_check (offset +0x500)
├── init_array[1] → integrity_verify  (offset +0xa00)
├── init_array[2] → start_monitor     (offset +0xe00)
└── init_func     → early_init        (offset +0x120)

libc.so
└── init_array[0] → libc_init         (offset +0x12300)
```

## 总结

- **init_array** 是 SO 加载时自动执行的构造函数数组，加固方案广泛使用
- **基础方案**：Hook `call_constructors`，可检测构造函数被调用的时机，但无法获取具体函数信息
- **高级方案**：通过 **CModule** + **soinfo 结构体解析**，完整获取 init_array 函数地址、数量、所属模块
- **典型应用**：发现加固反调试入口、结合反调试绕过、建立 SO 初始化调用图谱
- **CModule** 是 Frida 最强大的能力之一，允许在 JS 脚本中嵌入原生 C 代码，直接操作目标进程的任意内存结构
- 参考源码：`awesomeFrida/init_array.js`（275行完整实现）和 `awesomeFrida/call_constructors.js`（60行基础方案）
