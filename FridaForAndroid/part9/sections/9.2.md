# 9.2 Hook RegisterNatives追踪JNI注册

> 核心要点：RegisterNatives 是 JNI 动态注册的核心函数。Hook 它可以捕获进程中所有 SO 的动态 JNI 注册，获取 Java 类名、native 方法名、方法签名以及对应的 C 函数地址。

## 1. RegisterNatives 函数签名

```cpp
// JNI RegisterNatives 的函数签名
jint RegisterNatives(JNIEnv *env, 
                     jclass clazz,                    // Java 类
                     const JNINativeMethod *methods,  // 方法数组
                     jint nMethods);                  // 方法数量

// JNINativeMethod 结构体
typedef struct {
    const char* name;      // Java native 方法名
    const char* signature; // 方法签名（如 "()V"）
    void*       fnPtr;     // C 函数指针
} JNINativeMethod;
```

### 1.1 在 libart.so 中的符号名称

```cpp
// C++ mangled 名称（Android 8+）
_ZN3art3JNI15RegisterNativesEP7_JNIEnvP7_jclassPK15JNINativeMethodi

// 分解：
// _ZN3art3JNI15RegisterNativesE  → art::JNI::RegisterNatives
// P7_JNIEnv                       → JNIEnv*
// P7_jclass                       → jclass*
// PK15JNINativeMethod             → const JNINativeMethod*
// i                               → int
```

### 1.2 内存布局

```
RegisterNatives 的参数布局（ARM64 寄存器调用约定）：
    x0 = JNIEnv*        → args[0]
    x1 = jclass         → args[1]
    x2 = JNINativeMethod* → args[2]  ← 指向方法数组
    x3 = nMethods       → args[3]  ← 方法数量

JNINativeMethod 数组内存布局（每项 3 个指针大小）：
    [0] = name*         → 方法名指针
    [1] = signature*    → 签名指针
    [2] = fnPtr         → C 函数指针
    [3] = name*         → 第二个方法的名称
    [4] = signature*    → 第二个方法的签名
    [5] = fnPtr         → 第二个方法的 C 函数指针
    ...
```

## 2. 完整代码分析：hook_RegisterNatives.js

```javascript
// 引用：awesomeFrida/hook_RegisterNatives.js（完整69行代码）
function hook_RegisterNatives() {
    // 1. 在 libart.so 的符号表中查找 RegisterNatives
    var symbols = Module.enumerateSymbolsSync("libart.so");
    var addrRegisterNatives = null;
    
    for (var i = 0; i < symbols.length; i++) {
        var symbol = symbols[i];
        
        // 匹配 C++ mangled 名称：
        // _ZN3art3JNI15RegisterNativesEP7_JNIEnvP7_jclassPK15JNINativeMethodi
        if (symbol.name.indexOf("art") >= 0 &&
            symbol.name.indexOf("JNI") >= 0 &&
            symbol.name.indexOf("RegisterNatives") >= 0 &&
            symbol.name.indexOf("CheckJNI") < 0) {  // 排除 CheckJNI 版本
            addrRegisterNatives = symbol.address;
            console.log("RegisterNatives is at " + symbol.address + " " + symbol.name);
        }
    }

    // 2. 如果找到，安装 Interceptor
    if (addrRegisterNatives != null) {
        Interceptor.attach(addrRegisterNatives, {
            onEnter: function (args) {
                // 输出注册方法数量
                console.log("[RegisterNatives] method_count: " + args[3]);
                
                var env = args[0];
                var java_class = args[1];
                var class_name = Java.vm.tryGetEnv().getClassName(java_class);
                
                var methods_ptr = ptr(args[2]);
                var method_count = parseInt(args[3]);
                
                // 3. 遍历所有注册的方法
                for (var i = 0; i < method_count; i++) {
                    // 计算每个 JNINativeMethod 结构体指针
                    var name_ptr = Memory.readPointer(
                        methods_ptr.add(i * Process.pointerSize * 3));
                    var sig_ptr = Memory.readPointer(
                        methods_ptr.add(i * Process.pointerSize * 3 + Process.pointerSize));
                    var fnPtr_ptr = Memory.readPointer(
                        methods_ptr.add(i * Process.pointerSize * 3 + Process.pointerSize * 2));

                    var methodName = Memory.readCString(name_ptr);
                    var sig = Memory.readCString(sig_ptr);
                    var find_module = Process.findModuleByAddress(fnPtr_ptr);
                    
                    // 4. 输出详细信息
                    if (find_module) {
                        console.log("[RegisterNatives] java_class: " + class_name + 
                                  " name: " + methodName + 
                                  " sig: " + sig + 
                                  " fnPtr: " + fnPtr_ptr + 
                                  " module_name: " + find_module.name + 
                                  " module_base: " + find_module.base + 
                                  " offset: " + ptr(fnPtr_ptr).sub(find_module.base));
                    } else {
                        console.log("[RegisterNatives] java_class: " + class_name + 
                                  " name: " + methodName + 
                                  " sig: " + sig + 
                                  " fnPtr: " + fnPtr_ptr + 
                                  " module: null");
                    }
                }
            }
        });
    }
}

setImmediate(hook_RegisterNatives);
```

为什么需要排除 `CheckJNI`？
- Android 有 Debug 模式下的 `CheckJNI`，它包装了 `RegisterNatives` 做参数校验
- 如果 Hook 了 `CheckJNI` 版本，可能会漏掉真实调用或产生重复输出

## 3. 增强版：自动 Hook 动态注册的函数

```javascript
// 在发现 RegisterNatives 调用后，自动 Hook 注册的函数
function hook_RegisterNatives_auto() {
    var symbols = Module.enumerateSymbolsSync("libart.so");
    var addrRegisterNatives = null;
    
    for (var i = 0; i < symbols.length; i++) {
        var symbol = symbols[i];
        if (symbol.name.indexOf("art") >= 0 &&
            symbol.name.indexOf("JNI") >= 0 &&
            symbol.name.indexOf("RegisterNatives") >= 0 &&
            symbol.name.indexOf("CheckJNI") < 0) {
            addrRegisterNatives = symbol.address;
            break;
        }
    }
    
    if (!addrRegisterNatives) return;
    
    Interceptor.attach(addrRegisterNatives, {
        onEnter: function(args) {
            var methods_ptr = ptr(args[2]);
            var method_count = parseInt(args[3]);
            var env = args[0];
            
            try {
                var java_class = args[1];
                var class_name = Java.vm.tryGetEnv().getClassName(java_class);
            } catch(e) {
                var class_name = "unknown";
            }
            
            for (var i = 0; i < method_count; i++) {
                var fnPtr_ptr = Memory.readPointer(
                    methods_ptr.add(i * Process.pointerSize * 3 + Process.pointerSize * 2));
                var name_ptr = Memory.readPointer(
                    methods_ptr.add(i * Process.pointerSize * 3));
                
                var methodName = Memory.readCString(name_ptr);
                
                console.log("[+] Auto-hooking: " + class_name + "." + methodName);
                
                // 自动安装 Hook
                Interceptor.attach(fnPtr_ptr, {
                    onEnter: function(args) {
                        console.log("[Native] " + class_name + "." + methodName + " called");
                        console.log("  JNIEnv: " + args[0]);
                        console.log("  this/jclass: " + args[1]);
                        
                        // 如果有更多参数，尝试读取
                        for (var j = 2; j < args.length; j++) {
                            var arg = args[j];
                            if (arg instanceof NativePointer && !arg.isNull()) {
                                try {
                                    var str = arg.readCString();
                                    if (str && str.length < 256) {
                                        console.log("  arg[" + j + "]: \"" + str + "\"");
                                        continue;
                                    }
                                } catch(e) {}
                                console.log("  arg[" + j + "]: " + arg);
                            } else {
                                console.log("  arg[" + j + "]: " + arg);
                            }
                        }
                    },
                    onLeave: function(retval) {
                        console.log("  return: " + retval);
                    }
                });
            }
        }
    });
}

setImmediate(hook_RegisterNatives_auto);
```

## 4. 实战：发现加固 SO 中的隐藏 JNI 函数

加固 SO 通常会使用动态注册，且函数名可能是混淆后的无意义字符串。通过 Hook RegisterNatives，我们可以发现所有的隐藏 native 方法：

```javascript
// 搜集所有动态注册的记录
var registeredMethods = [];

function collect_all_registrations() {
    var symbols = Module.enumerateSymbolsSync("libart.so");
    var addrRegisterNatives = null;
    
    for (var i = 0; i < symbols.length; i++) {
        var s = symbols[i];
        if (s.name.indexOf("art") >= 0 &&
            s.name.indexOf("JNI") >= 0 &&
            s.name.indexOf("RegisterNatives") >= 0 &&
            s.name.indexOf("CheckJNI") < 0) {
            addrRegisterNatives = s.address;
            break;
        }
    }
    
    if (!addrRegisterNatives) return;
    
    Interceptor.attach(addrRegisterNatives, {
        onEnter: function(args) {
            var methods_ptr = ptr(args[2]);
            var method_count = parseInt(args[3]);
            var env = args[0];
            
            var class_name = "unknown";
            try {
                class_name = Java.vm.tryGetEnv().getClassName(args[1]);
            } catch(e) {}
            
            for (var i = 0; i < method_count; i++) {
                var entry = {
                    class: class_name,
                    name: Memory.readCString(
                        Memory.readPointer(methods_ptr.add(i * Process.pointerSize * 3))),
                    sig: Memory.readCString(
                        Memory.readPointer(methods_ptr.add(i * Process.pointerSize * 3 + Process.pointerSize))),
                    fnPtr: Memory.readPointer(
                        methods_ptr.add(i * Process.pointerSize * 3 + Process.pointerSize * 2))
                };
                
                var mod = Process.findModuleByAddress(entry.fnPtr);
                if (mod) {
                    entry.module = mod.name;
                    entry.offset = ptr(entry.fnPtr).sub(mod.base).toString(16);
                }
                
                registeredMethods.push(entry);
            }
        }
    });
    
    // 10秒后输出汇总
    setTimeout(function() {
        console.log("\n========== Registered Methods Summary ==========");
        console.log("Total: " + registeredMethods.length);
        registeredMethods.forEach(function(m, idx) {
            console.log("[" + idx + "] " + m.class + "." + m.name + 
                       " sig=" + m.sig);
            if (m.module) {
                console.log("    " + m.module + "+0x" + m.offset);
            }
        });
    }, 10000);
}

setImmediate(collect_all_registrations);
```

## 5. 安全注意事项

### 5.1 多线程竞争

```javascript
// RegisterNatives 可能在多线程环境下并发调用
// 使用锁或原子操作保护共享数据

var pendingLocks = {};

Interceptor.attach(addrRegisterNatives, {
    onEnter: function(args) {
        var threadId = Process.getCurrentThreadId();
        // 记录当前线程正在处理
        pendingLocks[threadId] = true;
    },
    onLeave: function(retval) {
        var threadId = Process.getCurrentThreadId();
        delete pendingLocks[threadId];
    }
});
```

### 5.2 版本兼容性

```javascript
// 不同 Android 版本 RegisterNatives 的符号名可能不同
function findRegisterNatives() {
    var candidates = [
        "RegisterNatives",                 // 简单名称
        "_ZN3art3JNI15RegisterNativesE",   // Android 8+ (itanium ABI)
    ];
    
    // 在 libart.so 中搜索
    var mod = Process.findModuleByName("libart.so");
    if (!mod) return null;
    
    var symbols = mod.enumerateSymbols();
    for (var i = 0; i < symbols.length; i++) {
        var s = symbols[i];
        if (s.name.indexOf("RegisterNatives") >= 0 && 
            s.name.indexOf("CheckJNI") < 0) {
            return s.address;
        }
    }
    return null;
}
```

## 总结

- **RegisterNatives Hook** 是追踪动态 JNI 注册的黄金技术
- **符号定位**：在 `libart.so` 中搜索包含 `art`、`JNI`、`RegisterNatives` 的符号
- **结构体解析**：`JNINativeMethod` 每项占 3 个指针大小（name/signature/fnPtr）
- **关键数据**：类名、方法名、签名、C 函数绝对地址、所属模块、模块内偏移
- **自动 Hook**：获取 fnPtr 后可以立即安装 Interceptor，实现函数级别的自动 Hook
- **适用场景**：加固 SO 分析、发现隐藏 native 方法、追踪所有 JNI 注册行为
