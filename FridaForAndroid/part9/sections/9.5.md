# 9.5 pthread_create反调试线程绕过

> 核心要点：加固 SO 常通过 phtread_create 创建反调试监控线程。本节将介绍三种绕过方案：直接拦截、dlsym 替换、以及综合替换+JNI 修复。

## 1. 反调试线程的工作原理

```cpp
// 典型的反调试线程逻辑（伪代码）
void *anti_debug_thread(void *arg) {
    while (1) {
        // 检查 TracerPid
        int pid = open("/proc/self/status", O_RDONLY);
        // 读取并解析 TracerPid
        if (tracer_pid != 0) {
            // 检测到调试器，退出或崩溃
            exit(-1);
        }
        sleep(1);
    }
}

// 在线程函数中使用 pthread_create
pthread_t tid;
pthread_create(&tid, NULL, anti_debug_thread, NULL);
```

反调试线程通常在两个时机创建：
1. **`.init_array` 构造函数中** — 在 JNI_OnLoad 前执行
2. **`JNI_OnLoad` 中** — 初始化时立即启动

## 2. 方案一：直接拦截 pthread_create

```javascript
// 引用：awesomeFrida/pthread_create.js（完整代码103行）
const TARGET_LIB_NAME = "libmsaoaidsec.so";
var TargetLibModule = null;

function hook_pthread_create() {
    // 1. 在 libc.so 中找到 pthread_create
    let pthread_create_addr = Module.findExportByName("libc.so", "pthread_create");
    if (!pthread_create_addr) {
        console.error("Failed to find pthread_create!");
        return;
    }

    // 2. Hook pthread_create
    Interceptor.attach(pthread_create_addr, {
        onEnter(args) {
            // args[2] = 线程函数地址（第三个参数）
            let thread_func_ptr = args[2];
            console.log("[+] pthread_create called, thread function address: " + 
                        thread_func_ptr);

            // 确保目标库已加载
            if (!TargetLibModule) {
                return;
            }

            // 3. 判断线程函数是否在目标库的范围内
            if (thread_func_ptr.compare(TargetLibModule.base) > 0 &&
                thread_func_ptr.compare(
                    TargetLibModule.base.add(TargetLibModule.size)) < 0) {

                console.warn("[!] Intercepted thread function at: " + thread_func_ptr +
                    " (Offset: " + thread_func_ptr.sub(TargetLibModule.base) + ")");

                // 4. 替换线程函数
                Interceptor.replace(thread_func_ptr, new NativeCallback(() => {
                    console.log("[*] Fake thread function executed, doing nothing...");
                }, "void", []));
            }
        }
    });
}
```

**技术要点：**

1. `pthread_create` 签名为：`int pthread_create(pthread_t *thread, const pthread_attr_t *attr, void *(*start_routine)(void *), void *arg)`
   - `args[0]` = `pthread_t*`（输出参数，线程 ID）
   - `args[1]` = `const pthread_attr_t*`（线程属性）
   - `args[2]` = `void* (*)(void*)`（线程函数指针）← **关键！**
   - `args[3]` = `void*`（传递给线程函数的参数）

2. **地址范围检查**：`thread_func_ptr.compare(TargetLibModule.base) > 0` 确认线程函数属于目标 SO

3. **Interceptor.replace 替换**：用空函数替换原始线程函数，使线程创建后什么都不做

## 3. 方案二：dlsym 替换法

```javascript
// 引用：awesomeFrida/pthread_create-fake.js（完整代码96行）
const TARGET_LIB_NAME = "libmsaoaidsec.so";
var TargetLibModule = null;

// 1. 创建虚假的 pthread_create
function create_fake_pthread_create() {
    const fake_pthread_create = Memory.alloc(4096);
    Memory.protect(fake_pthread_create, 4096, "rwx");
    Memory.patchCode(fake_pthread_create, 4096, code => {
        const cw = new Arm64Writer(code, { pc: ptr(fake_pthread_create) });
        cw.putRet();  // 直接返回（ret 指令）
        cw.flush();
    });
    return fake_pthread_create;
}

// 2. Hook dlsym，当目标 SO 查询 pthread_create 时返回假的
function hook_dlsym() {
    var dlsym = Module.findExportByName(null, "dlsym");
    if (dlsym !== null) {
        Interceptor.attach(dlsym, {
            onEnter: function(args) {
                // 获取调用 dlsym 的返回地址
                let caller = this.context.lr;
                
                // 确认调用者来自目标 SO
                if (caller.compare(TargetLibModule.base) > 0 &&
                    caller.compare(TargetLibModule.base.add(TargetLibModule.size)) < 0) {
                    
                    var name = ptr(args[1]).readCString();  // 读取符号名
                    if (name == 'pthread_create') {
                        console.warn(`replace symbol name: pthread_create`);
                        this.canFake = true;
                    }
                }
            },
            onLeave: function(retval) {
                if (this.canFake) {
                    // 替换 dlsym 的返回值
                    retval.replace(fake_pthread_create);
                    console.log("[*] dlsym pthread_create -> fake function");
                }
            }
        });
    }
}
```

**原理：**

```javascript
// 目标 SO 从系统获取 pthread_create 的地址过程：
// 方案一：dlsym("libc.so", "pthread_create")
//    → 通过 dlsym 查找
//    → 我们的 Hook 拦截，返回假的函数地址

// 方案二：直接调用 pthread_create（编译时链接）
//    → 不需要 dlsym
//    → 需要使用方案一（直接 Hook pthread_create）

// dlsym 替换法的优势：
// 对于使用 dlsym 动态查询的目标，可以在源头上替换
// 不需要担心和原始 pthread_create 的竞争
```

**Arm64Writer 说明：**

```javascript
// Arm64Writer 是 Frida 提供的 ARM64 汇编生成器
// 用于在运行时动态生成 ARM64 机器码
// 
// Memory.patchCode 用于安全地修改可执行内存
// putRet() 生成 RET 指令（ARM64 的返回指令）
// 
// 生成的假函数等价于：
//   ret    ← 立即返回，什么都不做
```

## 4. 方案三：综合替换 + JNI 修复

```javascript
// 引用：awesomeFrida/pthread_create-replace-jni.js（完整代码156行）
// 最完整的方案：拦截线程 + 修复 JNI 注册

function hook_pthread_create() {
    let pthread_create_addr = Module.findExportByName("libc.so", "pthread_create");
    if (!pthread_create_addr) return;

    Interceptor.attach(pthread_create_addr, {
        onEnter(args) {
            let thread_func_ptr = args[2];
            if (!TargetLibModule) return;

            // 检查线程函数是否在目标库中
            if (thread_func_ptr.compare(TargetLibModule.base) > 0 &&
                thread_func_ptr.compare(
                    TargetLibModule.base.add(TargetLibModule.size)) < 0) {

                console.warn("[!] Intercepted thread from " + TARGET_LIB_NAME + 
                           " at offset " + thread_func_ptr.sub(TargetLibModule.base));

                // 替换线程函数
                Interceptor.replace(thread_func_ptr, new NativeCallback(() => {
                    console.log("[*] Fake thread, doing nothing");
                }, "void", []));
            }
        }
    });
}

function hook_JNI_OnLoad() {
    let jniOnLoad = Module.findExportByName(TARGET_LIB_NAME, "JNI_OnLoad");
    if (!jniOnLoad) {
        // 如果找不到，遍历所有符号
        for (let symbol of module.enumerateSymbols()) {
            if (symbol.name.indexOf("JNI_OnLoad") >= 0) {
                jniOnLoad = symbol.address;
                break;
            }
        }
    }
    
    if (!jniOnLoad) return;

    Interceptor.attach(jniOnLoad, {
        onEnter(args) {
            console.log("[Hooked] JNI_OnLoad 被调用");
            bypass();
        }
    });
}

// JNI 修复：替换加固的 Java 层检测方法
function bypass() {
    try {
        var targetClassName = "com.bun.miitmdid.e";
        var targetClass = Java.use(targetClassName);
        
        targetClass.a.overloads.forEach(function(overload) {
            overload.implementation = function() {
                console.log("[*] Hooked " + targetClassName + ".a()");
                return 0;  // 返回 0 绕过检测
            };
        });
    } catch (e) {
        console.log("[!] bypass error: " + e);
    }
}
```

## 5. 三种方案对比

| 方案 | 原理 | 优点 | 缺点 | 代码行数 |
|------|------|------|------|---------|
| **直接拦截** | Hook pthread_create 检查参数 | 简单直接，覆盖所有线程 | 可能漏掉使用 clone 的情况 | ~103行 |
| **dlsym 替换** | 拦截 dlsym 返回假地址 | 源头拦截，不影响系统线程 | 仅对 dlsym 查询有效 | ~96行 |
| **综合替换+JNI** | 线程拦截 + JNI 修复 | 全面，同时处理 Java 层检测 | 代码较复杂 | ~156行 |

## 6. 部署时序控制

```javascript
// 所有方案都需要在目标 SO 加载后才能使用
// 通过 hook_dlopen + hook_call_constructors 控制时序

function find_call_constructors() {
    var is64Bit = Process.pointerSize === 8;
    var linkerModule = Process.getModuleByName(is64Bit ? "linker64" : "linker");
    var symbols = linkerModule.enumerateSymbols();
    
    for (var i = 0; i < symbols.length; i++) {
        if (symbols[i].name.indexOf('call_constructors') > 0) {
            return symbols[i].address;
        }
    }
}

function hook_call_constructors() {
    var ptr_call_constructors = find_call_constructors();
    var listener = Interceptor.attach(ptr_call_constructors, {
        onEnter: function(args) {
            console.warn('call_constructors onEnter');
            if (!TargetLibModule) {
                TargetLibModule = Process.findModuleByName(TARGET_LIB_NAME);
            }
            hook_pthread_create();  // ← 在构造函数执行前安装线程 Hook
            listener.detach();      // ← 只 Hook 一次
        },
    });
}

function hook_dlopen() {
    ["android_dlopen_ext", "dlopen"].forEach(funcName => {
        let addr = Module.findExportByName(null, funcName);
        if (addr) {
            Interceptor.attach(addr, {
                onEnter(args) {
                    let libName = ptr(args[0]).readCString();
                    if (libName && libName.indexOf(TARGET_LIB_NAME) >= 0) {
                        hook_call_constructors();
                    }
                }
            });
        }
    });
}

var is64Bit = Process.pointerSize === 8;
hook_dlopen();
```

## 7. 实战：检测并绕过反调试线程

```javascript
// 使用 Stalker 追踪目标 SO 的线程创建行为
function trace_thread_creation() {
    var pthread_create = Module.findExportByName("libc.so", "pthread_create");
    
    Interceptor.attach(pthread_create, {
        onEnter: function(args) {
            var start_routine = args[2];
            var mod = Process.findModuleByAddress(start_routine);
            
            if (mod) {
                var offset = start_routine.sub(mod.base);
                console.log("[pthread_create] " + mod.name + "+0x" + 
                           offset.toString(16));
                
                // 尝试读取线程函数的前几条指令
                console.log("  Function prologue:");
                for (var i = 0; i < 4; i++) {
                    try {
                        var inst = start_routine.add(i * 4).readU32();
                        console.log("    0x" + inst.toString(16));
                    } catch(e) { break; }
                }
            }
            
            // 记录调用栈
            var backtrace = Thread.backtrace(this.context, Backtracer.ACCURATE);
            console.log("  Backtrace:");
            backtrace.slice(0, 5).forEach(function(addr) {
                var m = Process.findModuleByAddress(addr);
                if (m) {
                    console.log("    " + m.name + "+0x" + 
                               addr.sub(m.base).toString(16));
                }
            });
        }
    });
}
```

## 总结

- **pthread_create 反调试**是最常见的 Native 反调试技术之一
- **三种绕过方案**：直接拦截（Hook 参数）、dlsym 替换（源头替换）、综合修复（线程+JNI）
- **时序控制**：通过 `call_constructors` 在构造函数执行前安装 Hook
- **替换技术**：`Interceptor.replace()` + `NativeCallback` 创建空函数替代原始线程函数
- **Arm64Writer**：Frida 提供的运行时汇编生成器，用于创建自定义的机器码
- **综合方案**最优：同时处理线程拦截和 Java 层检测方法的修复
