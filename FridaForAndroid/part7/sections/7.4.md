# 7.4 SO导出表与符号枚举

> 核心要点：掌握 Frida 中枚举模块和导出符号的 API，理解 `enumerateExports` 和 `enumerateSymbols` 的差异，学会通过符号信息定位目标函数。

## 1. 模块枚举基础

### 1.1 Process.enumerateModules()

`Process.enumerateModules()` 是 Frida 中最基础的模块枚举 API，它返回当前进程中所有已加载模块的信息：

```javascript
// 枚举所有已加载模块
var modules = Process.enumerateModules();
console.log("[*] Total modules loaded: " + modules.length);

modules.forEach(function(mod) {
    console.log(mod.name + " | " + mod.base + " | " + 
                ptr(mod.size) + " | " + mod.path);
});
```

**Module 对象包含的字段：**

| 字段 | 类型 | 说明 |
|------|------|------|
| `name` | string | 模块名称（如 `libc.so`） |
| `base` | NativePointer | 模块加载基址 |
| `size` | number | 模块大小（字节） |
| `path` | string | 模块的完整文件路径 |
| `enumerateExports()` | function | 枚举导出函数 → 返回 ExportDetails[] |
| `enumerateSymbols()` | function | 枚举所有符号 → 返回 SymbolDetails[] |
| `findExportByName(name)` | function | 按名称查找导出符号 |

### 1.2 过滤特定模块

```javascript
// 筛选出所有 native 库（排除 linker、libc 等系统库）
function enumerateAppNativeModules() {
    var modules = Process.enumerateModules();
    var appModules = modules.filter(function(mod) {
        return mod.name.indexOf("lib") === 0 &&      // 以 lib 开头
               mod.name.indexOf(".so") > 0 &&         // 包含 .so
               mod.path.indexOf("/data/app/") >= 0;   // 来自 data/app 目录
    });
    
    console.log("[*] App native modules:");
    appModules.forEach(function(mod) {
        console.log("  " + mod.name + " @ " + mod.base);
    });
    return appModules;
}
```

## 2. 导出函数枚举：enumerateExports()

### 2.1 基础用法

```javascript
// 枚举某个模块的所有导出函数
var module = Process.findModuleByName("libssl.so");
var exports = module.enumerateExports();

console.log("[*] Exports from " + module.name + ":");
exports.forEach(function(exp) {
    console.log("  " + exp.type + " | " + exp.name + " | " + exp.address);
});
```

**ExportDetails 结构：**

| 字段 | 类型 | 说明 |
|------|------|------|
| `type` | string | 符号类型：`"function"` 或 `"variable"` |
| `name` | string | 符号名称 |
| `address` | NativePointer | 符号的绝对内存地址 |

### 2.2 枚举并 Hook 所有导出函数

```javascript
// 引用：Chap11/traceNative.js 第25-59行（核心枚举逻辑）
function traceNativeExport() {
    var modules = Process.enumerateModules();
    for (var i = 0; i < modules.length; i++) {
        var module = modules[i];
        
        // 只关注 libssl.so
        if (module.name.indexOf("libssl.so") < 0) {
            continue;
        }
        
        var exports = module.enumerateExports();
        console.log('module.addr', module.base);
        
        for (var j = 0; j < exports.length; j++) {
            // 只 Hook 函数类型的导出
            if (exports[j].type == "function") {
                attach(exports[j].name, exports[j].address);
            }
        }
    }
}

// 通用 attach 函数
function attach(name, address) {
    console.log("attaching ", name);
    Interceptor.attach(address, {
        onEnter: function(args) {
            console.log("Entering => ", name);
        },
        onLeave: function(retval) {
            // 可以在这里处理返回值
        }
    });
}
```

### 2.3 计算导出函数的偏移

```javascript
// 枚举模块并计算每个导出函数的偏移量
function enumerateExportsWithOffset(moduleName) {
    var mod = Process.findModuleByName(moduleName);
    if (!mod) {
        console.log("Module not found: " + moduleName);
        return;
    }
    
    console.log("[*] Module: " + mod.name);
    console.log("[*] Base: " + mod.base);
    
    var exports = mod.enumerateExports();
    exports.forEach(function(exp) {
        // 计算偏移：绝对地址 - 基址
        var offset = exp.address.sub(mod.base);
        console.log("  " + exp.type.padEnd(10) + 
                    "0x" + offset.toString(16).padStart(8, '0') + 
                    "  " + exp.name);
    });
}

// 使用示例
enumerateExportsWithOffset("libnative-lib.so");
// 输出示例：
//   function   0x000005f0  Java_com_roysue_r0so_MainActivity_stringFromJNI
//   function   0x00000660  Java_com_roysue_r0so_MainActivity_stringFromJNI2
//   function   0x000006c4  stringFromJNI3
//   function   0x00000718  JNI_OnLoad
```

## 3. 符号枚举：enumerateSymbols()

### 3.1 enumerateExports() vs enumerateSymbols()

| API | 返回范围 | 是否包含非导出符号 | 性能 |
|-----|---------|------------------|------|
| `enumerateExports()` | 仅 `.dynsym` 中的 GLOBAL 符号 | ❌ | 快 |
| `enumerateSymbols()` | `.dynsym` + `.symtab` 中所有符号 | ✅（包括 LOCAL 符号） | 较慢 |

```javascript
// 对比两个 API 的返回结果
function compareExportVsSymbol(moduleName) {
    var mod = Process.findModuleByName(moduleName);
    if (!mod) return;
    
    var exports = mod.enumerateExports();
    var symbols = mod.enumerateSymbols();
    
    console.log("[*] " + moduleName + ":");
    console.log("  exports: " + exports.length + " symbols");
    console.log("  symbols: " + symbols.length + " symbols");
    
    // 查看 symbols 中多出的内容
    var exportNames = new Set(exports.map(e => e.name));
    symbols.forEach(function(sym) {
        if (!exportNames.has(sym.name) && sym.type === "function") {
            console.log("  [symbol-only] " + sym.name + " @ " + sym.address);
        }
    });
}
```

### 3.2 使用 enumerateSymbols 查找重载的 C++ 符号

当 C++ 代码中存在函数重载时，导出表中会包含 mangled 名称。`enumerateSymbols()` 可以帮我们找到这些符号：

```javascript
// 搜索包含特定名称的所有符号（包括 mangled 名称）
function findSymbolsByPattern(moduleName, pattern) {
    var mod = Process.findModuleByName(moduleName);
    if (!mod) return [];
    
    var results = [];
    var symbols = mod.enumerateSymbols();
    
    symbols.forEach(function(sym) {
        if (sym.name.indexOf(pattern) >= 0) {
            var offset = sym.address.sub(mod.base);
            results.push({
                name: sym.name,
                address: sym.address,
                offset: offset,
                type: sym.type
            });
            console.log("[+] " + sym.type + ": " + sym.name);
            console.log("    address: " + sym.address);
            console.log("    offset: 0x" + offset.toString(16));
        }
    });
    
    return results;
}

// 查找 libc.so 中所有以 "str" 开头的函数
findSymbolsByPattern("libc.so", "str");
```

## 4. traceNative.js 完整分析

下面是对 `traceNative.js`（引用：Chap11/traceNative.js 第25-59行）的完整分析：

```javascript
// 引用：Chap11/traceNative.js 第25-59行
function traceNativeExport() {
    var modules = Process.enumerateModules();     // ①枚举所有模块
    for (var i = 0; i < modules.length; i++) {
        var module = modules[i];
        
        if (module.name.indexOf("libssl.so") < 0) {  // ②过滤只关注 libssl.so
            continue;
        }
        
        var exports = module.enumerateExports();   // ③枚举该模块的所有导出
        console.log('module.addr', module.base);
        
        for (var j = 0; j < exports.length; j++) {
            if (exports[j].type == "function") {     // ④只处理函数类型
                attach(exports[j].name, exports[j].address);  // ⑤Hook 每个函数
            }
        }
    }
}
```

**逐行分析：**

1. **① `Process.enumerateModules()`**：获取进程所有模块列表
2. **② 字符串过滤**：`indexOf("libssl.so") < 0` — 跳过不匹配的模块
3. **③ `module.enumerateExports()`**：获取模块的所有导出项
4. **④ 类型过滤**：`exports[j].type == "function"` — 只关注函数，排除全局变量
5. **⑤ `attach()`**：对每个函数安装 Interceptor，实现批量 Hook

## 5. Module.findExportByName 与 Module.getExportByName

这两个 API 用于直接查找指定名称的导出符号：

```javascript
// Module.findExportByName(moduleName, exportName)
//   - moduleName 为 null 表示在所有模块中搜索
//   - 找不到时返回 null（不会抛异常）

// Module.getExportByName(moduleName, exportName)
//   - 功能同上，但找不到时抛异常

// 在所有模块中查找 SSL_write
var addr1 = Module.findExportByName(null, "SSL_write");
// 只在 libssl.so 中查找
var addr2 = Module.findExportByName("libssl.so", "SSL_write");

// 安全调用模式
function safeFindExport(moduleName, exportName) {
    var addr = Module.findExportByName(moduleName, exportName);
    if (addr) {
        console.log("[+] Found " + exportName + " @ " + addr);
        return addr;
    } else {
        console.log("[-] " + exportName + " not found");
        return null;
    }
}
```

## 7. 枚举未导出符号：进阶技巧

有些函数不会出现在 `.dynsym` 中，但仍存在于 `.symtab` 中。使用 `enumerateSymbols()` 可以找到它们：

```javascript
// 搜索一个模块中所有包含特定模式的符号（包括非导出符号）
function deepSearchSymbols(moduleName, keyword) {
    var mod = Process.findModuleByName(moduleName);
    if (!mod) return;
    
    var symbols = mod.enumerateSymbols();
    var found = 0;
    
    console.log("[*] Deep searching " + moduleName + " for '" + keyword + "'");
    console.log("[*] Total symbols: " + symbols.length);
    
    symbols.forEach(function(sym) {
        if (sym.name.toLowerCase().indexOf(keyword.toLowerCase()) >= 0) {
            var offset = sym.address.sub(mod.base);
            console.log("  [+] " + sym.type + 
                       " 0x" + offset.toString(16) + 
                       " " + sym.name);
            found++;
        }
    });
    
    console.log("[*] Found " + found + " matching symbols");
}

// 搜索 libc.so 中所有包含 "crypt" 的符号
deepSearchSymbols("libc.so", "crypt");
```

## 8. 符号枚举在逆向中的典型应用

### 8.1 定位被 strip 的 SO 中的函数

```javascript
// 对于被 strip 的 SO，导出符号可能很少
// 但通过 enumerateSymbols() 可能发现更多信息
function analyzeStrippedSo(moduleName) {
    var mod = Process.findModuleByName(moduleName);
    if (!mod) return;
    
    var exports = mod.enumerateExports();
    var symbols = mod.enumerateSymbols();
    
    console.log("[*] " + moduleName);
    console.log("    exports: " + exports.length);
    console.log("    symbols: " + symbols.length);
    
    // 如果 exports 远少于 symbols，说明被部分 strip
    if (exports.length < symbols.length * 0.3) {
        console.log("[!] This SO appears to be stripped");
    }
}
```

### 8.2 通过偏移快速定位函数

```javascript
// 在已知偏移的情况下，不依赖符号名直接 Hook
function hookByOffset(moduleName, offset, hookCallbacks) {
    var mod = Process.findModuleByName(moduleName);
    if (!mod) {
        console.log("[-] Module not found");
        return;
    }
    
    var targetAddr = mod.base.add(offset);
    console.log("[+] Hooking " + moduleName + "+0x" + offset.toString(16));
    console.log("[+] Target address: " + targetAddr);
    
    Interceptor.attach(targetAddr, {
        onEnter: function(args) {
            if (hookCallbacks && hookCallbacks.onEnter) {
                hookCallbacks.onEnter(args);
            }
        },
        onLeave: function(retval) {
            if (hookCallbacks && hookCallbacks.onLeave) {
                hookCallbacks.onLeave(retval);
            }
        }
    });
}

// 使用：Hook libnative-lib.so+0xf444（stringFromJNI3 的地址）
hookByOffset("libnative-lib.so", 0xf444, {
    onEnter: function(args) {
        console.log("[hookByOffset] Enter, env=" + args[0] + ", obj=" + args[1]);
    },
    onLeave: function(retval) {
        console.log("[hookByOffset] Leave, retval=" + retval);
    }
});
```

## 9. 实战：dump 模块导出表到文件

```javascript
// 将模块的所有导出信息保存到文件，便于离线分析
function dumpExportsToFile() {
    Java.perform(function() {
        var modules = Process.enumerateModules();
        
        modules.forEach(function(mod) {
            // 只处理应用自己加载的 SO
            if (mod.path.indexOf("/data/app/") < 0 && 
                mod.path.indexOf("/data/data/") < 0) {
                return;
            }
            
            var exports = mod.enumerateExports();
            var filename = "/sdcard/Download/exports_" + mod.name + ".txt";
            var file = new File(filename, "w");
            
            file.write("Module: " + mod.name + "\n");
            file.write("Base: " + mod.base + "\n");
            file.write("Size: " + ptr(mod.size) + "\n");
            file.write("Path: " + mod.path + "\n");
            file.write("Export count: " + exports.length + "\n\n");
            
            exports.forEach(function(exp) {
                var offset = exp.address.sub(mod.base);
                file.write(
                    "0x" + offset.toString(16) + " " +
                    exp.type + " " +
                    exp.name + "\n"
                );
            });
            
            file.close();
            console.log("[dump] " + filename + " written (" + exports.length + " exports)");
        });
    });
}
```

## 总结

- **3 个核心 API**：`Process.enumerateModules()` / `module.enumerateExports()` / `module.enumerateSymbols()`
- **enumerateExports vs enumerateSymbols**：前者只返回 GLOBAL 导出符号（快），后者返回所有符号（更全）
- **偏移计算**：`exp.address.sub(mod.base)` 将绝对地址转换为相对于基址的偏移，便于 IDA Pro 等静态分析工具定位
- **过滤策略**：通过模块名、符号名模式匹配来精确筛选目标函数
- **批量 Hook**：结合 `enumerateExports()` + `Interceptor.attach()` 可以实现对模块中所有导出函数的批量 Hook
- 导出符号枚举是 **Part8 Native Hook** 的基础，所有 Hook 操作都依赖于准确的符号或地址定位
