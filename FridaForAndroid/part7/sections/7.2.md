# 7.2 SO加载流程：dlopen与linker

> 核心要点：理解 Android 动态库加载的完整链路——从 Java 层的 `System.loadLibrary` 到底层 linker 的 `dlopen`，以及 linker 内部的 `soinfo` 结构体，是实现 SO 动态拦截和延迟 Hook 的理论基础。

## 1. SO 加载的全链路

```
Java: System.loadLibrary("native-lib")
    ↓
Java: Runtime.loadLibrary0(ClassLoader, libName)
    ↓
Java: ClassLoader.findLibrary → 搜索路径找到完整路径
    ↓
JNI: native 方法 Runtime.nativeLoad(name, loader, librarySearchPath)
    ↓
C++: JavaVMExt::LoadNativeLibrary → 调用 android_dlopen_ext
    ↓
C:   __loader_android_dlopen_ext / dlopen
    ↓
linker: soinfo::load(address) → mmap 映射文件到内存
    ↓
linker: soinfo::prelink_image() → 解析 program headers
    ↓
linker: soinfo::link_image() → 重定位 + 符号解析
    ↓
linker: soinfo::call_constructors() → 执行 .init / .init_array
    ↓
返回: dlopen 返回 soinfo* 句柄
    ↓
JNI: 查找 JNI_OnLoad 并调用 → 动态注册 JNI 函数
```

## 2. dlopen 与 android_dlopen_ext

### 2.1 dlopen 标准接口

```c
#include <dlfcn.h>

// 标准 dlopen 接口
void *dlopen(const char *filename, int flags);

// 常用 flags:
// RTLD_LAZY   = 0x00001  → 延迟绑定（使用时才解析符号）
// RTLD_NOW    = 0x00002  → 立即绑定（加载时解析所有符号）
// RTLD_LOCAL  = 0x00000  → 符号不导出给其他库
// RTLD_GLOBAL = 0x00100  → 符号导出给其他库
```

### 2.2 android_dlopen_ext（Android 特有）

```c
// Android 扩展接口，支持更多加载选项
void *android_dlopen_ext(const char *filename, int flags, 
                         const android_dlextinfo *extinfo);

typedef struct {
    uint64_t flags;                    // 扩展选项
    void *library_fd;                  // 已打开的文件描述符
    void *library_address;             // 指定加载地址（用于加载已映射的库）
    void *reserved_addr;               // 保留地址
    size_t reserved_size;              // 保留大小
    int32_t relro_fd;                  // RELRO 文件描述符
} android_dlextinfo;
```

`android_dlopen_ext` 在 System.loadLibrary 的底层调用链路中被优先使用，它提供了更细粒度的加载控制（如指定文件描述符、指定基址等）。

### 2.3 Frida Hook dlopen 实现加载监控

以下是来自 awesomeFrida 仓库的 `dlopen.js` 脚本（第1-35行），功能是监控目标动态库是否被加载：

```javascript
// 引用：awesomeFrida/dlopen.js（完整代码35行）
// 目标动态库名称
const TARGET_LIB_NAME = "libmsaoaidsec.so";

function hook_dlopen() {
    ["android_dlopen_ext", "dlopen"].forEach(funcName => {
        let addr = Module.findExportByName(null, funcName);
        if (addr) {
            Interceptor.attach(addr, {
                onEnter(args) {
                    let libName = ptr(args[0]).readCString();
                    if (libName) {
                        console.log(`[+] ${funcName} onEnter: ${libName}`);
                    }
                },
                onLeave: function (retval) {
                    console.log(`[+] ${funcName} onLeave`);
                }
            });
        }
    });
}

hook_dlopen();
```

**代码要点：**

1. `Module.findExportByName(null, funcName)` — `null` 表示在所有已加载模块中查找导出符号
2. `args[0]` 是 `dlopen` 的第一个参数，即 SO 文件路径字符串指针
3. `ptr(args[0]).readCString()` 读取路径字符串
4. 同时 hook `dlopen` 和 `android_dlopen_ext` 确保无遗漏

### 2.4 Java 层 System.loadLibrary Hook

```javascript
// 引用：awesomeFrida/System.loadLibrary.js（完整代码43行）
// 作用：监控 System.loadLibrary 加载的库文件

const TargetLibName = 'cocos2djs';
const MyLibName = 'test';

function hook() {
    const System = Java.use('java.lang.System');
    const Runtime = Java.use('java.lang.Runtime');
    const VMStack = Java.use('dalvik.system.VMStack');

    System.loadLibrary.implementation = function (libName) {
        try {
            console.log('\nSystem.loadLibrary("' + libName + '")');
            
            // 调用原始的 loadLibrary 方法
            const loaded = Runtime.getRuntime().loadLibrary0(
                VMStack.getCallingClassLoader(), libName);
            
            if (libName == TargetLibName) {
                console.log(TargetLibName + " is loaded!");
                // 在目标库加载后，可以主动加载自定义库
                try {
                    Runtime.getRuntime().loadLibrary0(
                        VMStack.getCallingClassLoader(), MyLibName);
                } catch (e) {
                    console.log(e);
                }
            }
            return loaded;
        } catch (ex) {
            // 异常处理
        }
    };
}

hook();
```

**代码要点：**

1. Hook Java 层的 `System.loadLibrary`，在库加载前记录日志
2. 调用原始 `Runtime.loadLibrary0` 完成实际加载
3. 在目标库加载后可以自动加载自定义 SO，实现库注入
4. `VMStack.getCallingClassLoader()` 获取调用者的 ClassLoader，确保正确的类加载上下文

## 3. soinfo 结构体深度解析

`soinfo` 是 Android linker 内部用于管理已加载 SO 的核心数据结构。不同 Android 版本的结构体定义有所差异，但核心字段基本一致。

### 3.1 soinfo 核心字段

```cpp
// Android linker 内部 soinfo 结构体（简化版）
// 来源：bionic/linker/linker_soinfo.h
struct soinfo {
    // 名称字段（Android 5.0+ 使用字符指针而非定长数组）
#if defined(__work_around_b_24465209__)
    char old_name_[128];          // 兼容旧版本的定长名称
#endif
    const ElfW(Phdr)* phdr;       // program header 表指针
    size_t phnum;                  // program header 数量
    ElfW(Addr) base;              // SO 加载基址 ★ 最常用字段
    size_t size;                  // SO 在内存中的大小
    ElfW(Dyn)* dynamic;           // .dynamic 段指针
    
    // 链表结构
    void* next;                   // 下一个 soinfo（链表）
    uint32_t flags_;              // 状态标志
    
    // 符号表信息
    const char* strtab_;          // 字符串表
    ElfW(Sym)* symtab_;          // 符号表
    size_t nbucket_;              // 哈希表桶数
    size_t nchain_;               // 哈希表链数
    uint32_t* bucket_;           // 哈希桶
    uint32_t* chain_;            // 哈希链
    
    // 重定位信息
#if defined(USE_RELA)
    ElfW(Rela)* plt_rela_;       // PLT 重定位表
    size_t plt_rela_count_;
    ElfW(Rela)* rela_;           // 普通重定位表
    size_t rela_count_;
#else
    ElfW(Rel)* plt_rel_;         // PLT 重定位表（32位）
    size_t plt_rel_count_;
    ElfW(Rel)* rel_;             // 普通重定位表
    size_t rel_count_;
#endif
    
    // 初始化/析构函数数组
    linker_ctor_function_t* preinit_array_;    // 预初始化函数（preinit_array）
    size_t preinit_array_count_;
    linker_ctor_function_t* init_array_;       // 初始化函数数组 ★
    size_t init_array_count_;
    linker_dtor_function_t* fini_array_;       // 析构函数数组
    size_t fini_array_count_;
    linker_ctor_function_t init_func_;         // .init 函数
    linker_dtor_function_t fini_func_;         // .fini 函数
};
```

### 3.2 关键字段用途

| 字段 | 用途 | 在 Frida Hook 中的价值 |
|------|------|----------------------|
| `base` | SO 加载基址 | `Module.findBaseAddress()` 返回的就是这个值 |
| `size` | SO 内存大小 | 用于 dump 整个 SO |
| `init_array_` | 构造函数数组 | Hook `call_constructors` 拦截初始化逻辑 |
| `symtab_`/`strtab_` | 符号表/字符串表 | `enumerateSymbols()` 的基础 |
| `phdr`/`phnum` | 程序头 | 解析加载段信息 |

### 3.3 通过 Frida 读取 soinfo

// 在 Android linker 中，soinfo 通常可以从 dlopen 返回值获取

function inspect_soinfo(moduleName) {

    if (!mod) return;

    console.log("[*] Base: " + mod.base);
    console.log("[*] Size: " + ptr(mod.size));
    console.log("[*] Path: " + mod.path);
    
    // 读取 ELF header 获取 program headers
    var base = mod.base;
    var e_phoff = base.add(32).readU32();   // ELF64 下是 64位偏移
    // 在实际逆向中，可以通过解析 linker 的内存来获取 soinfo
    // 这里展示概念性示例
}

// 更实用的：通过 dlopen 返回值获取 soinfo
function hook_dlopen_get_soinfo() {

            this.path = ptr(args[0]).readCString();

                // retval 实际上就是 soinfo* 指针

            }
        }
    });

## 4. Linker 加载流程详解

### 4.1 加载阶段（load）

```cpp
// 伪代码：soinfo::load 流程
soinfo* soinfo::load(int fd, const char* name, int flags) {
    // 1. 创建 soinfo 对象
    si = new soinfo();
    
    // 2. mmap 文件到内存（只读映射 ELF header 和 program headers）
    elf_header = mmap(fd, 0, PAGE_SIZE, PROT_READ, MAP_PRIVATE);
    
    // 3. 读取 program headers
    phdr = elf_header + elf_header->e_phoff;
    phnum = elf_header->e_phnum;
    
    // 4. 计算所需总内存大小
    //    遍历所有 PT_LOAD 段，计算 vaddr 范围
    load_size = phdr[last_pt_load].vaddr + phdr[last_pt_load].memsz
                - phdr[first_pt_load].vaddr;
    
    // 5. 预留地址空间
    base = mmap(预留地址, load_size, PROT_NONE, MAP_PRIVATE | MAP_ANONYMOUS);
    si->base = base;
    si->size = load_size;
    
    // 6. 逐段映射 PT_LOAD 段
    for each PT_LOAD:
        mmap(base + phdr->vaddr, fd, phdr->offset, phdr->flags);
    
    // 7. 解析 .dynamic 段
    si->dynamic = base + phdr_dynamic->vaddr;
    
    // 8. 解析符号表和字符串表
    for each DT_* entry in dynamic:
        case DT_SYMTAB: si->symtab_ = base + d_un.d_ptr; break;
        case DT_STRTAB: si->strtab_ = base + d_un.d_ptr; break;
        case DT_INIT_ARRAY: si->init_array_ = base + d_un.d_ptr; break;
    
    return si;
}
```

### 4.2 链接阶段（link_image）

```cpp
// 伪代码：soinfo::link_image 流程
bool soinfo::link_image() {
    // 1. 解析依赖库（DT_NEEDED 条目）
    for each DT_NEEDED:
        dlopen(needed_lib);
    
    // 2. 处理重定位（.rel / .rela 段）
    for each relocation:
        lookup_symbol(relocation.symbol_name);
        apply_relocation();
    
    // 3. 执行构造函数
    call_constructors();
}
```

### 4.3 构造函数执行阶段（call_constructors）

```cpp
// 伪代码：soinfo::call_constructors
void soinfo::call_constructors() {
    // 1. 先确保依赖库的构造函数已执行
    for each DT_NEEDED lib:
        call_constructors_recursive(lib);
    
    // 2. 执行 .preinit_array（仅可执行文件有）
    for i in 0..preinit_array_count_:
        preinit_array_[i]();
    
    // 3. 执行 .init_array（逆向关注点！）
    for i in 0..init_array_count_:
        init_array_[i](argc, argv, envp);  // 构造函数调用
    
    // 4. 执行 .init 函数
    if (init_func_):
        init_func_(argc, argv, envp);
}
```

## 5. 延迟 Hook 的关键时机

理解 SO 加载流程后，我们可以确定以下几个关键的 Hook 时机：

| Hook 时机 | 特点 | 适用场景 |
|-----------|------|---------|
| `dlopen` onEnter | 库尚未加载 | 准备环境，设置标志 |
| `dlopen` onLeave | 库已加载，但 JNI_OnLoad 未执行 | dump SO、Hook .init_array |
| `call_constructors` onEnter | 构造函数即将执行 | 拦截 .init_array 反调试 |
| `JNI_OnLoad` onEnter | JNI 函数即将注册 | Hook RegisterNatives |
| `JNI_OnLoad` onLeave | JNI 函数已注册 | Hook 所有 JNI 函数 |

## 6. 完整监控脚本

```javascript
// 综合监控：追踪某个 SO 的完整加载过程
const TARGET_LIB = "target.so";
var hookInstalled = false;

function monitor_so_load() {
    // 1. Hook dlopen
    ["dlopen", "android_dlopen_ext"].forEach(funcName => {
        var addr = Module.findExportByName(null, funcName);
        if (addr) {
            Interceptor.attach(addr, {
                onEnter: function(args) {
                    var libName = ptr(args[0]).readCString();
                    if (libName && libName.indexOf(TARGET_LIB) >= 0) {
                        this.isTarget = true;
                        console.log("[dlopen onEnter] Loading: " + libName);
                    }
                },
                onLeave: function(retval) {
                    if (this.isTarget && retval.toInt32() !== 0) {
                        console.log("[dlopen onLeave] " + TARGET_LIB + 
                                  " loaded, handle = " + retval);
                        
                        // 在这个时机可以 dump SO
                        dump_so(TARGET_LIB);
                    }
                }
            });
        }
    });
    
    // 2. Hook call_constructors 监控 init_array
    hook_call_constructors();
    
    // 3. 如果目标已加载，直接 Hook
    var mod = Process.findModuleByName(TARGET_LIB);
    if (mod) {
        console.log("[*] " + TARGET_LIB + " already loaded at " + mod.base);
        hook_native_functions(mod);
    }
}

function hook_call_constructors() {
    var is64 = Process.pointerSize === 8;
    var linkerName = is64 ? "linker64" : "linker";
    var linker = Process.findModuleByName(linkerName);
    
    var symbols = linker.enumerateSymbols();
    var callConstructors = null;
    
    for (var i = 0; i < symbols.length; i++) {
        if (symbols[i].name.indexOf("call_constructors") > 0) {
            callConstructors = symbols[i].address;
            break;
        }
    }
    
    if (callConstructors) {
        Interceptor.attach(callConstructors, {
            onEnter: function(args) {
                var soinfo_ptr = args[0];
                var mod = Process.findModuleByAddress(soinfo_ptr);
                if (mod && mod.name.indexOf(TARGET_LIB) >= 0) {
                    console.log("[call_constructors] " + TARGET_LIB + 
                              " constructors executing!");
                }
            }
        });
    }
}

function dump_so(name) {
    try {
        var mod = Process.getModuleByName(name);
        var path = "/data/local/tmp/" + mod.name + ".dump.so";
        var f = new File(path, "wb");
        Memory.protect(mod.base, mod.size, 'rwx');
        f.write(mod.base.readByteArray(mod.size));
        f.flush();
        f.close();
        console.log("[dump] SO dumped to " + path);
    } catch (e) {
        console.log("[dump] Failed: " + e);
    }
}

function hook_native_functions(mod) {
    // 对目标 SO 中的导出函数进行 Hook
    var exports = mod.enumerateExports();
    exports.forEach(function(exp) {
        if (exp.type === "function") {
            Interceptor.attach(exp.address, {
                onEnter: function(args) {
                    console.log("[hook] " + exp.name + " called");
                }
            });
        }
    });
}

monitor_so_load();
```

## 7. 不同 Android 版本的差异

| Android 版本 | Linker 路径 | 关键变化 |
|-------------|------------|---------|
| Android 4.x | `/system/bin/linker` | soinfo 名称用固定数组 |
| Android 5.x | `/system/bin/linker` | 引入 `__work_around_b_24465209__` |
| Android 6.x | `/system/bin/linker` | soinfo 结构体优化 |
| Android 7.x | `/system/bin/linker` | 符号查找优化 |
| Android 8.x | `/system/bin/linker` / `linker64` | 分离 32/64 位 linker |
| Android 10+ | `/system/bin/linker` / `linker64` | 安全性增强，soinfo 结构体变化较大 |

## 总结

- SO加载有三个关键阶段：**load（映射）→ link（重定位）→ construct（初始化）**
- `dlopen` 返回的指针实际上就是 **soinfo***，包含了 SO 的所有元信息
- `call_constructors` 是执行 `.init_array` 的函数，**在此之前的 Hook 都是安全的**
- `android_dlopen_ext` 是 Android 扩展接口，Android 系统框架层优先使用
- 通过同时 Hook `dlopen` + `android_dlopen_ext` + `call_constructors` 可以完整追踪 SO 加载
- 不同 Android 版本的 linker 内部结构有差异，Frida 提供了抽象的 Module API 屏蔽了这些差异
