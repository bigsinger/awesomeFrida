# 2.5 Python 绑定：Frida 客户端编写

## 概述

Python 绑定是 Frida 最强大的能力之一。通过 Python 客户端，你可以：

- 自动化注入流程
- 接收和处理结构化数据
- 调用脚本导出的 RPC 方法
- 批量测试和自动化逆向

## 完整 Python 客户端

以下是一个完整的 Frida Python 客户端，包含了所有核心要素：

```python
# 引用自 Chap04/loader.py 第 1-24 行（完整结构，略作扩展）
import frida
import sys

def on_message(message, data):
    """消息回调：处理从 JS 端 send 来的消息"""
    if message['type'] == 'send':
        print("[*] {0}".format(message['payload']))
    else:
        print(message)


# ========== 连接方式选择 ==========

# 方式一：USB 设备（推荐）
# device = frida.get_usb_device()

# 方式二：远程网络设备
device = frida.get_device_manager().add_remote_device('192.168.50.129:1234')

# ========== 进程附加 ==========
process = device.attach('com.example.junior')

# ========== 加载 JS 脚本 ==========
with open('call.js') as f:
    jscode = f.read()
script = process.create_script(jscode)

# 注册消息回调
script.on('message', on_message)

# 加载脚本
script.load()

# ========== 保持运行并调用 RPC ==========
print("[*] Script loaded, calling RPC methods...")
for i in range(20, 30):
    for j in range(0, 10):
        script.exports.sub(str(i), str(j))
```

## 核心 API 详解

### 1. 设备连接

```python
# USB 设备
device = frida.get_usb_device()                    # 获取唯一 USB 设备
device = frida.get_usb_device(timeout=5)           # 带超时

# 远程设备
device = frida.get_device_manager().add_remote_device('ip:port')

# 枚举所有设备
devices = frida.enumerate_devices()
for d in devices:
    print(f"Type: {d.type}, Name: {d.name}, ID: {d.id}")
```

### 2. 进程附加

```python
# Attach 模式
process = device.attach('com.example.app')   # 按进程名
process = device.attach(1234)                # 按 PID

# Spawn 模式
pid = device.spawn(['com.example.app'])
process = device.attach(pid)
device.resume(pid)  # 恢复进程

# 枚举进程
processes = device.enumerate_processes()
for p in processes:
    print(f"PID: {p.pid}, Name: {p.name}")
```

### 3. 脚本生命周期

```python
# 创建脚本
script = process.create_script(jscode)

# 注册回调
script.on('message', on_message)    # send() 消息
script.on('detached', on_detach)    # 进程分离
script.on('destroyed', on_destroy)  # 脚本销毁

# 加载/卸载
script.load()       # 注入脚本
script.unload()     # 卸载脚本

# RPC 调用
result = script.exports.sub("123", "111")  # 同步调用
```

## 完整的高级客户端模板

```python
"""
advanced_loader.py - 高级 Frida Python 客户端
包含：错误处理、重连、RPC、日志分级
"""
import frida
import sys
import time
import json
import logging

logging.basicConfig(level=logging.INFO)
logger = logging.getLogger("FridaClient")

class FridaClient:
    def __init__(self, target_app, script_path, remote_addr=None):
        self.target_app = target_app
        self.script_path = script_path
        self.remote_addr = remote_addr
        self.device = None
        self.session = None
        self.script = None
        
    def connect(self):
        """连接设备"""
        try:
            if self.remote_addr:
                self.device = frida.get_device_manager().add_remote_device(self.remote_addr)
            else:
                self.device = frida.get_usb_device()
            logger.info(f"Connected to device: {self.device}")
        except Exception as e:
            logger.error(f"Device connection failed: {e}")
            raise
            
    def attach(self, use_spawn=False):
        """附加到目标进程"""
        try:
            if use_spawn:
                pid = self.device.spawn([self.target_app])
                self.session = self.device.attach(pid)
                self._load_script()
                self.device.resume(pid)
                logger.info(f"Spawned {self.target_app} (PID: {pid})")
            else:
                self.session = self.device.attach(self.target_app)
                self._load_script()
                logger.info(f"Attached to {self.target_app}")
        except Exception as e:
            logger.error(f"Attach failed: {e}")
            raise
            
    def _load_script(self):
        """加载 JS 脚本"""
        with open(self.script_path) as f:
            jscode = f.read()
        self.script = self.session.create_script(jscode)
        self.script.on('message', self._on_message)
        self.script.load()
        logger.info("Script loaded")
        
    def _on_message(self, message, data):
        """消息处理"""
        if message['type'] == 'send':
            payload = message['payload']
            if isinstance(payload, dict):
                level = payload.get('level', 'INFO')
                msg = payload.get('msg', payload)
                getattr(logger, level.lower(), logger.info)(msg)
            else:
                logger.info(f"[DATA] {payload}")
        elif message['type'] == 'error':
            logger.error(f"Script error: {message}")
            
    def call_rpc(self, method, *args):
        """调用 RPC 方法"""
        if self.script:
            return getattr(self.script.exports, method)(*args)
            
    def close(self):
        """清理资源"""
        if self.script:
            self.script.unload()
        if self.session:
            self.session.detach()
            
# 使用示例
if __name__ == "__main__":
    client = FridaClient(
        target_app="com.example.junior",
        script_path="call.js"
    )
    client.connect()
    client.attach(use_spawn=False)
    
    # 调用 RPC
    for i in range(20, 30):
        result = client.call_rpc("sub", str(i), str(111))
        print(f"RPC result: {result}")
        
    client.close()
```

## 错误处理最佳实践

```python
# 重连逻辑
def safe_attach(device, target, max_retries=3):
    for attempt in range(max_retries):
        try:
            process = device.attach(target)
            return process
        except frida.ProcessNotFoundError:
            logger.warning(f"Process '{target}' not found, retrying...")
            time.sleep(2)
        except Exception as e:
            logger.error(f"Attach error: {e}")
            time.sleep(1)
    raise Exception(f"Failed to attach after {max_retries} attempts")
```

## 小结

- Python 绑定是 Frida 最强大的编程接口
- 核心流程：`get_usb_device()` → `attach()` → `create_script()` → `load()`
- `rpc.exports` 在 JS 端导出方法，`script.exports.X()` 在 Python 端调用
- `script.on('message', callback)` 接收 JS 端的 `send()`
- 推荐将逻辑封装为 `FridaClient` 类，便于复用
