# 2.6 Frida 脚本模板与最佳实践

## 标准脚本模板

经过大量实战验证的 Frida 脚本模板，推荐所有 Frida 脚本遵循此结构：

```javascript
/**
 * 脚本名: template.js
 * 描述: Frida 标准脚本模板
 * 目标: com.roysue.demo02
 * 日期: 2024-01-01
 */

// ======== 配置区域 ========
const CONFIG = {
    DEBUG: true,
    TARGET_CLASS: 'com.roysue.demo02.MainActivity',
    TARGET_METHOD: 'fun',
    LOG_LEVEL: 'INFO'  // DEBUG | INFO | WARN | ERROR
};

// ======== 辅助函数 ========
function log_debug(msg) { if (CONFIG.DEBUG) send({level: 'DEBUG', msg}); }
function log_info(msg)  { send({level: 'INFO', msg}); }
function log_warn(msg)  { send({level: 'WARN', msg}); }
function log_error(msg) { send({level: 'ERROR', msg}); }

function printStack() {
    Java.perform(function () {
        console.log(
            Java.use("android.util.Log")
                .getStackTraceString(Java.use("java.lang.Exception").$new())
        );
    });
}

// ======== Hook 逻辑 ========
function hookTarget() {
    Java.perform(function() {
        log_info("Starting hook...");
        
        var TargetClass = Java.use(CONFIG.TARGET_CLASS);
        log_info("Class found: " + CONFIG.TARGET_CLASS);
        
        TargetClass[CONFIG.TARGET_METHOD].overload('int', 'int')
            .implementation = function(x, y) {
                log_info(`Method called: fun(${x}, ${y})`);
                
                // 可选：打印调用栈
                if (CONFIG.DEBUG) {
                    printStack();
                }
                
                // 可选：修改参数
                var modified_x = x;
                var modified_y = y;
                
                // 调用原始方法
                var result = this[CONFIG.TARGET_METHOD](modified_x, modified_y);
                
                log_info(`Result: ${result}`);
                
                // 可选：修改返回值
                return result;
            };
        
        log_info("Hook installed successfully");
    });
}

// ======== 入口 ========
function main() {
    log_info("Script loaded");
    
    // 延迟执行，确保环境就绪
    setTimeout(function() {
        hookTarget();
    }, 100);
}

setImmediate(main);

// ======== RPC 导出 ========
// rpc.exports = {
//     hook: function() {
//         hookTarget();
//         return "hook installed";
//     },
//     getConfig: function() {
//         return JSON.stringify(CONFIG);
//     }
// };
```

> **结构参考**：整合了 `Chap03/1.js`（入口结构）、`Chap04/call.js`（RPC 导出）、以及 awesomeFrida 仓库中各脚本的通用模式。

## 最佳实践清单

### 1. 命名规范

| 元素 | 规范 | 示例 |
|------|------|------|
| 脚本文件 | 有意义的英文名 | `hook_login.js`, `bypass_ssl.js` |
| 函数名 | camelCase | `hookTarget()`, `printStack()` |
| 常量 | UPPER_SNAKE_CASE | `CONFIG`, `TARGET_CLASS` |
| 变量 | camelCase | `result`, `modifiedArg` |

### 2. 注释规范

```javascript
/**
 * 文件头注释：说明脚本用途
 * @target com.roysue.demo02
 * @version 1.0
 */

// 区块标题注释
// ======== 配置区域 ========

// 行内注释：说明复杂逻辑
```

### 3. 配置与逻辑分离

```javascript
// ✅ 推荐：配置集中管理
const CONFIG = {
    TARGET_CLASS: 'com.roysue.demo02.MainActivity',
    // ...
};

// ❌ 避免：配置散落在代码中
// Java.use('com.roysue.demo02.MainActivity')  // 硬编码
```

### 4. 错误处理

```javascript
function safeHook(className) {
    try {
        Java.perform(function() {
            var clazz = Java.use(className);
            // hook logic...
        });
    } catch (e) {
        console.log("[ERROR] Hook failed: " + e);
        send({level: "ERROR", msg: e.toString()});
    }
}
```

### 5. 延迟与重试

```javascript
function waitAndHook(className, retries = 5) {
    var attempt = 0;
    function tryHook() {
        attempt++;
        try {
            Java.perform(function() {
                Java.use(className);
                // 类存在，继续 hook
                doHook();
            });
        } catch (e) {
            if (attempt < retries) {
                console.log(`Retry ${attempt}/${retries} for ${className}`);
                setTimeout(tryHook, 500);
            } else {
                console.log(`Failed after ${retries} retries: ${className}`);
            }
        }
    }
    tryHook();
}
```

## 常见反模式

### ❌ 避免的做法

```javascript
// 1. 过度嵌套
Java.perform(function() {
    Java.perform(function() {
        Java.perform(function() {
            // 不要这样
        });
    });
});

// 2. 硬编码路径
Java.use('com.example.app.v1.MainActivity');  // 版本更新后就失效

// 3. 漏写 setImmediate
function main() {
    console.log("永远不会执行");
}
// 忘记 setImmediate(main)

// 4. 忽略错误
Java.perform(function() {
    var clazz = Java.use('SomeClass');
    clazz.method.implementation = function() {
        this.method();  // 如果抛异常会导致后续全部失效
    };
});
```

### ✅ 推荐的做法

```javascript
// 1. try-catch 保护
function safeCall(fn) {
    try { return fn(); } 
    catch (e) { console.log(e); }
}

// 2. 配置驱动
var config = { /*...*/ };

// 3. 总是 setImmediate
setImmediate(function safeMain() {
    try { main(); }
    catch(e) { console.log("Main error: " + e); }
});
```

## 最终推荐模板

对于 90% 的逆向场景，以下极简模板就足够了：

```javascript
// minimal_template.js - 极简 Frida 模板
function main() {
    Java.perform(function() {
        var Target = Java.use('com.roysue.demo02.MainActivity');
        console.log("[+] Class found");
        
        Target.fun.implementation = function(x, y) {
            console.log(`fun(${x}, ${y})`);
            var ret = this.fun(x, y);
            console.log(`=> ${ret}`);
            return ret;
        };
    });
}
setImmediate(main);
```

## 小结

| 实践 | 说明 |
|------|------|
| 使用 `setImmediate(main)` | 标准入口 |
| 配置集中管理 | 易于维护和修改 |
| try-catch 包裹关键逻辑 | 防止因异常导致整个脚本失效 |
| 注释清晰 | 便于团队协作和复盘 |
| 分离 JS 与 Python | Python 负责流程，JS 负责 Hook |
| 延迟/重试机制 | 应对类加载时机不确定的情况 |

> **黄金法则**：写 Frida 脚本就像写生产代码一样——可读、可维护、可复用。
