# 8.1 Interceptor.attach基础

> 核心要点：Interceptor 是 Frida 的 Native Hook 核心框架。本节从 attach 的基础用法开始，理解它的工作原理和适用场景。

## 1. Interceptor 框架概述

Frida 的 Interceptor 提供两种 Hook 模式：

| 模式 | API | 特点 | 可恢复 | 适用场景 |
|------|-----|------|--------|---------|
| **Attach** | `Interceptor.attach()` | 在原函数前后插入回调 | ✅ 可 detach | 监控、日志、分析 |
| **Replace** | `Interceptor.replace()` | 完全替换函数实现 | ❌ 不可恢复 | 修改返回值、绕过检测 |

## 2. Interceptor.attach 基本语法

```javascript
Interceptor.attach(targetAddress, {
    onEnter: function(args) {
        // 函数进入时执行
        // args 包含函数的所有参数
        console.log("Function called with args:", args);
    },
    onLeave: function(retval) {
        // 函数返回时执行
        // retval 包含返回值
        console.log("Function returned:", retval);
    }
});
```

**参数说明：**

| 参数 | 类型 | 说明 |
|------|------|------|
| `targetAddress` | NativePointer | 目标函数的绝对内存地址 |
| `onEnter` | Function | （可选）函数进入时调用的回调 |
| `onLeave` | Function | （可选）函数返回时调用的回调 |

### 2.1 最简单的 Hook 示例

```javascript
// Hook libc.so 中的 puts 函数
var puts_addr = Module.findExportByName("libc.so", "puts");
console.log("[*] puts address: " + puts_addr);

Interceptor.attach(puts_addr, {
    onEnter: function(args) {
        console.log("puts called with: " + args[0].readCString());
    },
    onLeave: function(retval) {
        console.log("puts returned: " + retval);
    }
});
```

### 2.2 引用代码：hookNative.js 第1-12行

```javascript
// 引用：Chap11/hookNative.js 第1-12行
function hook_native(){
    // 通过导出函数名定位
    var addr = Module.getExportByName(
        "libnative-lib.so", 
        "Java_com_roysue_r0so_MainActivity_stringFromJNI"
    );
    
    Interceptor.attach(addr, {   
        onEnter: function(args){
            console.log("jnienv pointer =>", args[0]);
            console.log("jobj pointer =>", args[1]);
        },
        onLeave: function(retval){
            console.log("retval is =>", 
                Java.vm.getEnv().getStringUtfChars(retval, null).readCString());
            console.log("=================");
        }
    });
}
```

**代码分析：**

1. `Module.getExportByName("libnative-lib.so", "Java_com_roysue_r0so_MainActivity_stringFromJNI")`
   - 在 `libnative-lib.so` 中查找名为 `Java_com_roysue_r0so_MainActivity_stringFromJNI` 的导出符号
   - 返回该符号的**绝对内存地址**
   - 与 `Module.findExportByName` 的区别：找不到时抛异常

2. `onEnter` 中的参数：
   - `args[0]` = `JNIEnv*` — JNI 环境指针
   - `args[1]` = `jobject` — `this` 对象引用（这里是 `MainActivity` 实例）

3. `onLeave` 中的返回值处理：
   - `retval` 是 `jstring` 类型（Java 字符串的引用）
   - `Java.vm.getEnv().getStringUtfChars(retval, null).readCString()` 将 `jstring` 转为 JavaScript 字符串

## 3. targetAddress 的三种获取方式

### 3.1 通过导出符号名

```javascript
// 方式1：Module.findExportByName（返回 null 表示未找到）
var addr = Module.findExportByName("libssl.so", "SSL_write");

// 方式2：Module.getExportByName（未找到时抛异常）
var addr = Module.getExportByName("libssl.so", "SSL_write");

// 方式3：遍历 module.enumerateExports()
var mod = Process.findModuleByName("libssl.so");
var exports = mod.enumerateExports();
exports.forEach(function(exp) {
    if (exp.name === "SSL_write") {
        Interceptor.attach(exp.address, { ... });
    }
});
```

### 3.2 通过基址 + 偏移

```javascript
// 已知函数在 SO 中的偏移
var mod = Process.findModuleByName("libnative-lib.so");
var offset = 0xf444;  // 从 IDA Pro 或其他分析工具获得
var addr = mod.base.add(offset);

Interceptor.attach(addr, { ... });
```

### 3.3 通过内存扫描 / 模式匹配

```javascript
// 通过字节码特征搜索函数（已知 ARM64 的 prologue）
var mod = Process.findModuleByName("libnative-lib.so");
// ARM64 函数 prologue 常见的: stp x29, x30, [sp, #-0x10]!
var pattern = "FD 7B BF A9";  // 对应的机器码
var matches = Memory.scan(mod.base, mod.size, pattern, {
    onMatch: function(address, size) {
        console.log("[+] Found match at: " + address);
    },
    onComplete: function() {
        console.log("[+] Scan complete");
    }
});
```

## 4. Interceptor.attach vs Interceptor.replace

### 4.1 Interceptor.replace 示例

```javascript
// 用 replace 完全替换函数行为
var puts_addr = Module.findExportByName("libc.so", "puts");

Interceptor.replace(puts_addr, new NativeCallback(function(str) {
    // 自定义实现：在原始字符串前加前缀
    var prefix = Memory.allocUtf8String("[Hooked] ");
    var newStr = prefix + str;
    // 调用原始函数
    var original_puts = new NativeFunction(puts_addr, "int", ["pointer"]);
    return original_puts(newStr);
}, "int", ["pointer"]));
```

### 4.2 attach vs replace 对比

```javascript
// attach 模式：可以随时 detach
var listener = Interceptor.attach(addr, {
    onEnter: function(args) {
        console.log("enter");
    }
});
// 随时移除 Hook
listener.detach();

// replace 模式：无法恢复原始函数
Interceptor.replace(addr, new NativeCallback(function() {
    console.log("replaced");
    return 0;
}, "void", []));
// ❌ 无法恢复为原始函数
```

## 5. 基础 Hook 模板

```javascript
'use strict';

// 通用 Hook 模板
function hook_function(moduleName, funcNameOrOffset, callbacks) {
    var targetAddr = null;
    
    // 判断是符号名还是偏移
    if (typeof funcNameOrOffset === 'string') {
        // 通过导出符号名查找
        targetAddr = Module.findExportByName(moduleName, funcNameOrOffset);
        if (!targetAddr) {
            console.log("[-] Export not found: " + funcNameOrOffset);
            return;
        }
    } else if (typeof funcNameOrOffset === 'number') {
        // 通过基址+偏移定位
        var mod = Process.findModuleByName(moduleName);
        if (!mod) {
            console.log("[-] Module not found: " + moduleName);
            return;
        }
        targetAddr = mod.base.add(funcNameOrOffset);
    } else {
        console.log("[-] Invalid function specification");
        return;
    }
    
    console.log("[+] Hooking " + moduleName + " at " + targetAddr);
    
    Interceptor.attach(targetAddr, {
        onEnter: function(args) {
            console.log("[+] Entering " + (funcNameOrOffset.name || funcNameOrOffset));
            if (callbacks && callbacks.onEnter) {
                callbacks.onEnter(args);
            }
        },
        onLeave: function(retval) {
            if (callbacks && callbacks.onLeave) {
                callbacks.onLeave(retval);
            }
        }
    });
}

// 使用示例
hook_function("libc.so", "puts", {
    onEnter: function(args) {
        console.log("puts(\"" + args[0].readCString() + "\")");
    }
});

hook_function("libnative-lib.so", 0xF444, {
    onEnter: function(args) {
        console.log("Function at offset 0xF444 called");
    }
});
```

## 6. 基础调试技巧

### 6.1 打印调用栈

```javascript
// 在 onEnter 中打印堆栈
Interceptor.attach(addr, {
    onEnter: function(args) {
        console.log("Called from:");
        console.log(Thread.backtrace(this.context, Backtracer.ACCURATE)
            .map(DebugSymbol.fromAddress).join('\n'));
    }
});
```

### 6.2 只 Hook 一次

```javascript
// Hook 一次后自动 detach
function hookOnce(addr, onEnter) {
    var listener = Interceptor.attach(addr, {
        onEnter: function(args) {
            onEnter(args);
            listener.detach();
            console.log("[*] Hook detached after first call");
        }
    });
}
```

## 总结

- **Interceptor.attach** 是最常用的 Native Hook 方式，它不会修改原函数代码
- **targetAddress** 可以通过三种方式获取：导出符号名、基址+偏移、内存模式扫描
- **onEnter** 中访问参数 `args`，**onLeave** 中访问返回值 `retval`
- **attach 可恢复**（`listener.detach()`），**replace 不可恢复**
- 第一个参数 `args[0]` 是函数参数列表中的第一个参数（对于 JNI 函数是 `JNIEnv*`）
