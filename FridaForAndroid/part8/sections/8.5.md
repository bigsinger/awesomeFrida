# 8.5 Hook JNI函数实战

> 核心要点：JNI 函数的参数类型特殊（JNIEnv*, jobject/jclass），需要特殊的处理方法。本节从 JNI 函数签名分析开始，到参数读取、返回值修改，覆盖完整的 JNI Hook 实战。

## 1. JNI 函数参数结构

### 1.1 参数布局

所有 JNI 函数的前两个参数是固定的：

```
实例方法:  JNIEnv* env, jobject this, [arg1, arg2, ...]
静态方法:  JNIEnv* env, jclass clazz, [arg1, arg2, ...]
```

```javascript
// JNI 函数的通用 Hook 模板
Interceptor.attach(targetAddr, {
    onEnter: function(args) {
        // args[0] = JNIEnv*  ← JNI 环境指针
        // args[1] = jobject  ← this 对象 (实例方法)
        //           或 jclass  ← 当前类 (静态方法)
        // args[2] = 第一个实际参数
        // args[3] = 第二个实际参数
        // ...
        
        console.log("[JNI] JNIEnv*: " + args[0]);
        console.log("[JNI] this/clazz: " + args[1]);
    }
});
```

### 1.2 获取 JNIEnv

`JNIEnv*` 是一个指向函数指针表的指针。在 Frida 中，可以通过 `Java.vm.getEnv()` 或 `Java.vm.tryGetEnv()` 获取 JNIEnv：

```javascript
// 从 args[0] 获取 JNIEnv*
var env = args[0];
// 或者使用 Frida 内置 API
var env = Java.vm.getEnv();
var env = Java.vm.tryGetEnv();  // 不会抛异常
```

## 2. JNI 字符串处理

### 2.1 读取 jstring 参数

```javascript
// 引用：Chap11/hookNative.js 综合实践
// 读取 JNI 函数中的 jstring 参数
Interceptor.attach(jniFuncAddr, {
    onEnter: function(args) {
        // 假设第三个参数是 jstring
        var jstr = args[2];
        
        // 方法1：使用 Java.vm.getEnv()
        var env = Java.vm.getEnv();
        var str1 = env.getStringUtfChars(jstr, null).readCString();
        console.log("[jstring] " + str1);
        
        // 方法2：使用 NativeFunction 调用 GetStringUTFChars
        // 更底层的方式，不依赖 Java.vm API
    },
    onLeave: function(retval) {
        // 如果返回值是 jstring，同样可以读取
        if (!retval.isNull()) {
            var env = Java.vm.getEnv();
            var str = env.getStringUtfChars(retval, null).readCString();
            console.log("[return jstring] " + str);
        }
    }
});
```

### 2.2 修改 jstring 返回值

```javascript
// 修改 jstring 类型返回值
Interceptor.attach(jniFuncAddr, {
    onLeave: function(retval) {
        // 读取原始返回值
        var env = Java.vm.getEnv();
        var original = env.getStringUtfChars(retval, null).readCString();
        console.log("[original] " + original);
        
        // 构造新的 jstring
        var newStr = "[MODIFIED] " + original;
        var newJstr = env.newStringUtf8(newStr);
        
        // 替换返回值
        retval.replace(newJstr);
        console.log("[modified] " + newStr);
    }
});
```

## 3. 处理 JNI 数组参数

### 3.1 jbyteArray 读取

```javascript
// 处理 jbyteArray 类型的参数
Interceptor.attach(jniFuncAddr, {
    onEnter: function(args) {
        // 假设 args[2] 是 jbyteArray
        var byteArray = args[2];
        
        var env = Java.vm.getEnv();
        
        // 获取数组长度
        var len = env.getArrayLength(byteArray);
        console.log("[jbyteArray] length: " + len);
        
        // 获取字节数组元素
        var elements = env.getByteArrayElements(byteArray, null);
        if (elements) {
            console.log("[jbyteArray] first bytes: " + 
                       hexdump(elements, {length: Math.min(len, 64)}));
        }
    }
});
```

### 3.2 jintArray 读取

```javascript
// 处理 int 数组参数
Interceptor.attach(jniFuncAddr, {
    onEnter: function(args) {
        var intArray = args[2];
        var env = Java.vm.getEnv();
        
        var len = env.getArrayLength(intArray);
        var elements = env.getIntArrayElements(intArray, null);
        
        console.log("[jintArray] length=" + len);
        for (var i = 0; i < Math.min(len, 10); i++) {
            console.log("  [" + i + "] = " + elements[i]);
        }
        
        // 释放数组元素
        // env.releaseIntArrayElements(intArray, elements, 0);
    }
});
```

## 4. JNIEnv 函数表调用

`JNIEnv*` 实际上是一个指向函数指针数组的指针。我们可以通过它直接调用 JNI 函数：

```javascript
// 直接通过 JNIEnv 函数表调用 JNI 函数
function call_jni_function(env, funcName, ...) {
    // JNIEnv 是一个指向函数指针数组的指针
    // 数组的每个元素指向一个 JNI 函数
    
    // 例如：GetStringUTFChars 在 JNI 函数表中的索引
    // 不同 JDK 版本索引可能不同，建议使用 Java.vm API
    
    // 推荐的调用方式：
    var env_instance = Java.vm.getEnv();
    // 使用 env_instance 的方法
}

// 更常用的方式是通过 NativeFunction 绑定 JNI API
var GetStringUTFChars = new NativeFunction(
    Module.findExportByName("libart.so", "_ZN3art3JNI21GetStringUTFCharsEP7_JNIEnvP8_jstringPh"),
    "pointer", ["pointer", "pointer", "pointer"]
);
```

## 5. 完整 JNI Hook 实战：traceNative.js 分析

```javascript
// 引用：Chap11/traceNative.js 综合实践
// 将枚举导出函数和 Hook JNI 函数结合

function hookAllJNIFunctions() {
    var modules = Process.enumerateModules();
    
    modules.forEach(function(mod) {
        // 只处理应用自己的 SO
        if (mod.path.indexOf("/data/app/") < 0) {
            return;
        }
        
        var exports = mod.enumerateExports();
        var jniCount = 0;
        
        exports.forEach(function(exp) {
            // JNI 函数通常以 Java_ 开头
            if (exp.type === "function" && 
                (exp.name.indexOf("Java_") === 0 ||
                 exp.name.indexOf("JNI_") === 0)) {
                
                jniCount++;
                console.log("[JNI] Hooking " + exp.name);
                
                Interceptor.attach(exp.address, {
                    onEnter: function(args) {
                        console.log("\n[JNI Call] " + exp.name);
                        console.log("  JNIEnv*: " + args[0]);
                        console.log("  this: " + args[1]);
                        
                        // 如果有更多参数，尝试打印
                        if (args.length > 2) {
                            for (var i = 2; i < args.length; i++) {
                                var arg = args[i];
                                if (arg instanceof NativePointer && !arg.isNull()) {
                                    try {
                                        var str = arg.readCString();
                                        if (str && str.length < 256) {
                                            console.log("  arg[" + i + "]: \"" + str + "\"");
                                            continue;
                                        }
                                    } catch(e) {}
                                    console.log("  arg[" + i + "]: " + arg);
                                } else {
                                    console.log("  arg[" + i + "]: " + arg);
                                }
                            }
                        }
                    },
                    onLeave: function(retval) {
                        console.log("  Return: " + retval);
                    }
                });
            }
        });
        
        if (jniCount > 0) {
            console.log("[*] " + mod.name + ": " + jniCount + " JNI functions hooked");
        }
    });
}

hookAllJNIFunctions();
```

## 7. 综合案例：Hook JNI 加密函数

```javascript
// 假设有一个 SO 中的 JNI 函数进行 AES 加密：
// Java_com_example_app_NativeLib_encrypt([BI)[B

function hook_jni_encrypt() {
    // 尝试不同的函数名变体
    var possibleNames = [
        "Java_com_example_app_NativeLib_encrypt",
        "Java_com_example_app_NativeLib_decrypt",
        "Java_com_example_app_NativeLib_aesEncrypt"
    ];
    
    possibleNames.forEach(function(funcName) {
        var addr = Module.findExportByName(null, funcName);
        if (addr) {
            console.log("[+] Found " + funcName + " at " + addr);
            
            Interceptor.attach(addr, {
                onEnter: function(args) {
                    console.log("\n[" + funcName + "] called");
                    
                    // JNIEnv*, jobject 后，第一个实际参数
                    if (args.length > 2) {
                        // args[2] 可能是 jbyteArray（输入数据）
                        // args[3] 可能是 jint（数据长度）
                        // 或其他参数
                        console.log("  arg[0] (JNIEnv): " + args[0]);
                        console.log("  arg[1] (this): " + args[1]);
                        if (args[2]) {
                            console.log("  arg[2]: " + args[2]);
                        }
                    }
                },
                onLeave: function(retval) {
                    // 返回值可能是 jbyteArray（加密/解密结果）
                    if (!retval.isNull()) {
                        console.log("[" + funcName + "] returned: " + retval);
                        
                        try {
                            var env = Java.vm.getEnv();
                            var arr = retval;
                            var len = env.getArrayLength(arr);
                            var elems = env.getByteArrayElements(arr, null);
                            console.log("  result length: " + len);
                            console.log("  result hex: " + 
                                hexdump(elems, {length: Math.min(len, 64)}));
                        } catch (e) {
                            console.log("  Cannot read result: " + e);
                        }
                    }
                }
            });
        }
    });
}
```

## 8. JNI Hook 中的常见问题

### 8.1 JNIEnv 跨线程

```javascript
// JNIEnv 是线程局部的，不能在另一个线程中使用
// 正确做法：在 onEnter 中立即读取需要的值，保存到 this
Interceptor.attach(jniFuncAddr, {
    onEnter: function(args) {
        // ✅ 正确：在 onEnter 中读取并保存
        var env = Java.vm.getEnv();
        var jstr = args[2];
        this.savedString = env.getStringUtfChars(jstr, null).readCString();
        
        // ❌ 错误：不能在 onLeave 中再用 args[2]
        // onLeave 时 JNIEnv 可能已经失效
    },
    onLeave: function(retval) {
        // ✅ 使用 this 中保存的值
        console.log("String was: " + this.savedString);
    }
});
```

### 8.2 jstring 到 JavaScript 字符串

```javascript
// 多种 jstring 读取方式
function readJString(env, jstr) {
    // 方式1：通过 Java.vm API
    try {
        return Java.vm.getEnv().getStringUtfChars(jstr, null).readCString();
    } catch(e) {}
    
    // 方式2：通过 NativeFunction 绑定
    try {
        var GetStringUTFChars = new NativeFunction(
            Module.findExportByName("libart.so", "_ZN3art3JNI21GetStringUTFCharsEP7_JNIEnvP8_jstringPh"),
            "pointer", ["pointer", "pointer", "pointer"]
        );
        var result = GetStringUTFChars(env, jstr, ptr(0));
        var str = result.readCString();
        return str;
    } catch(e) {}
    
    return null;
}
```

## 总结

- JNI 函数的前两个参数固定为 **`JNIEnv*`** 和 **`jobject`/`jclass`**
- **jstring** 通过 `Java.vm.getEnv().getStringUtfChars(jstr, null).readCString()` 读取
- **jbyteArray** 通过 `getArrayLength()` + `getByteArrayElements()` 读取内容
- **返回值修改**：构造新的 jstring/jbyteArray，用 `retval.replace()` 替换
- **线程安全**：JNIEnv 是线程局部的，跨回调保存数据用 `this` 对象
- 结合 `enumerateExports()` 可以自动发现并 Hook 所有 JNI 函数
