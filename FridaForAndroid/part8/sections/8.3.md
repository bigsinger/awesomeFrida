# 8.3 Hook导出函数：getExportByName

> 核心要点：通过导出函数名直接定位并 Hook 函数是最直接的方式。本节以 Hook libssl.so 的 SSL_write 为例，展示导出函数 Hook 的完整流程。

## 1. 定位导出函数

### 1.1 Module.findExportByName  vs  Module.getExportByName

```javascript
// 两个 API 的区别：

// findExportByName：找不到返回 null，不抛异常
var addr1 = Module.findExportByName("libssl.so", "SSL_write");
if (addr1) {
    console.log("SSL_write found at: " + addr1);
} else {
    console.log("SSL_write not found");
}

// getExportByName：找不到抛异常
try {
    var addr2 = Module.getExportByName("libssl.so", "SSL_write");
    console.log("SSL_write found at: " + addr2);
} catch (e) {
    console.log("Error: " + e.message);
}
```

**参数说明：**

| 参数 | 类型 | 说明 |
|------|------|------|
| `moduleName` | string 或 null | 模块名称，`null` 表示在所有已加载模块中搜索 |
| `exportName` | string | 要查找的导出符号名称 |

### 1.2 在所有模块中查找

```javascript
// moduleName 为 null 时在所有模块中搜索
// 这在不确定目标函数在哪个模块时非常有用
var addr = Module.findExportByName(null, "SSL_write");

if (addr) {
    // 找到后确认所属模块
    var mod = Process.findModuleByAddress(addr);
    console.log("SSL_write found in " + mod.name + " at " + addr);
}
```

## 2. Hook SSL_write 实战

### 2.1 完整代码

```javascript
// 引用：Chap11/hookSSL_write.js（完整代码21行）
function hook_ssl_write(){
    // 1. 定位导出函数
    var addr = Module.getExportByName("libssl.so", "SSL_write");
    
    // 2. 安装 Interceptor
    Interceptor.attach(addr, {   
        onEnter: function(args){
            // args[1] = 输出缓冲区指针
            // args[2] = 输出长度
            console.log("\n", hexdump(args[1], {length: args[2].toInt32()}));
        },
        onLeave: function(retval){
            console.log("================== onLeave =================");
        }
    });
}

function main(){
    console.log("Entering main");
    hook_ssl_write();
}
setImmediate(main);
```

**代码分析：**

1. **`Module.getExportByName("libssl.so", "SSL_write")`** 定位导出函数
2. `SSL_write` 的标准签名：`int SSL_write(SSL *ssl, const void *buf, int num)`
   - `args[0]` = `SSL*` 指针
   - `args[1]` = `const void*` 要写入的数据缓冲区
   - `args[2]` = `int` 要写入的字节数
3. **`hexdump(args[1], {length: args[2].toInt32()})`** — Frida 内置的十六进制转储函数，直接输出 SSL 写入的明文数据

### 2.2 hexdump 详解

```javascript
// hexdump 的完整参数
hexdump(targetAddress, {
    offset: 0,           // 从目标地址偏移多少字节开始
    length: 256,         // 要转储的字节数
    header: true,        // 是否显示列头
    ansi: true          // 是否使用 ANSI 颜色
});

// 运行输出示例：
/*

7b9f1234  48 65 6c 6c 6f 20 57 6f 72 6c 64 21 00 00 00 00  Hello World!....
*/
```

### 2.3 从 HTTPS 流量中提取请求内容

```javascript
// 增强版 SSL_write Hook，提取明文 HTTP 请求
function hook_ssl_write_enhanced() {
    var ssl_write = Module.findExportByName("libssl.so", "SSL_write");
    if (!ssl_write) {
        ssl_write = Module.findExportByName("libssl.so", "SSL_write");
    }
    
    if (!ssl_write) {
        console.log("[-] SSL_write not found, trying alternative...");
        // 某些自定义 TLS 实现可能使用不同名称
        var mod = Process.findModuleByName("libssl.so");
        if (mod) {
            var exports = mod.enumerateExports();
            exports.forEach(function(exp) {
                if (exp.name.indexOf("SSL_write") >= 0 || 
                    exp.name.indexOf("ssl_write") >= 0) {
                    console.log("[+] Found: " + exp.name);
                    install_hook(exp.address);
                }
            });
        }
        return;
    }
    
    install_hook(ssl_write);
}

function install_hook(addr) {
    Interceptor.attach(addr, {
        onEnter: function(args) {
            var buf = args[1];
            var len = args[2].toInt32();
            
            if (len > 0 && len < 65536) {
                try {
                    var data = buf.readByteArray(len);
                    console.log("\n[SSL_write] " + len + " bytes:");
                    console.log(hexdump(buf, {length: Math.min(len, 512)}));
                    
                    // 尝试提取 HTTP 请求行
                    var str = String.fromCharCode.apply(null, 
                        new Uint8Array(data));
                    if (str.indexOf("HTTP") >= 0 || 
                        str.indexOf("GET ") === 0 || 
                        str.indexOf("POST ") === 0) {
                        var lines = str.split('\n');
                        console.log("[HTTP Request] " + lines[0]);
                    }
                } catch (e) {
                    console.log("[SSL_write] Error reading data: " + e);
                }
            }
        },
        onLeave: function(retval) {
            // retval 是实际写入的字节数
            console.log("[SSL_write] wrote " + retval + " bytes");
        }
    });
}

hook_ssl_write_enhanced();
```

## 3. 通用导出函数 Hook 模板

```javascript
// 通用导出函数 Hook 工具
function hook_export(moduleName, exportName, handlers) {
    var addr = Module.findExportByName(moduleName, exportName);
    if (!addr) {
        console.log("[-] " + exportName + " not found in " + moduleName);
        return null;
    }
    
    console.log("[+] " + exportName + " @ " + addr + " (" + moduleName + ")");
    
    var listener = Interceptor.attach(addr, {
        onEnter: function(args) {
            if (handlers && handlers.onEnter) {
                handlers.onEnter.call(this, args);
            }
        },
        onLeave: function(retval) {
            if (handlers && handlers.onLeave) {
                handlers.onLeave.call(this, retval);
            }
        }
    });
    
    return listener; // 返回 listener 用于 detach
}

// 使用示例：Hook SSL_read
var ssl_read_listener = hook_export("libssl.so", "SSL_read", {
    onEnter: function(args) {
        this.buf = args[1];
        this.len = args[2].toInt32();
        console.log("[SSL_read] request to read " + this.len + " bytes");
    },
    onLeave: function(retval) {
        var bytesRead = retval.toInt32();
        if (bytesRead > 0) {
            console.log("[SSL_read] read " + bytesRead + " bytes:");
            console.log(hexdump(this.buf, {length: bytesRead}));
        }
    }
});

// 5 秒后移除 Hook
setTimeout(function() {
    if (ssl_read_listener) {
        ssl_read_listener.detach();
        console.log("[*] SSL_read hook removed");
    }
}, 5000);
```

## 4. 跨模块的导出函数 Hook

某些函数可能通过 `dlsym` 被动态解析后调用。此时我们需要在调用前完成 Hook：

```javascript
// 在不确定目标模块时，延迟 Hook
function hook_dynamic_export(exportName, handlers) {
    // 先检查是否已加载
    var addr = Module.findExportByName(null, exportName);
    if (addr) {
        console.log("[+] " + exportName + " already loaded, hooking now");
        return installHook(addr, handlers);
    }
    
    // 未加载，监控 dlopen
    console.log("[*] " + exportName + " not loaded yet, monitoring dlopen");
    var dlopenAddr = Module.findExportByName(null, "dlopen");
    
    Interceptor.attach(dlopenAddr, {
        onEnter: function(args) {
            this.path = ptr(args[0]).readCString();
        },
        onLeave: function(retval) {
            if (retval.toInt32() !== 0 && this.path) {
                // dlopen 完成后，再次尝试查找
                var addr = Module.findExportByName(null, exportName);
                if (addr) {
                    console.log("[+] " + exportName + " found after loading " + 
                               this.path);
                    installHook(addr, handlers);
                }
            }
        }
    });
}

function installHook(addr, handlers) {
    return Interceptor.attach(addr, {
        onEnter: function(args) {
            if (handlers.onEnter) handlers.onEnter.call(this, args);
        },
        onLeave: function(retval) {
            if (handlers.onLeave) handlers.onLeave.call(this, retval);
        }
    });
}
```

## 5. 批量 Hook 多个导出函数

```javascript
// 批量 Hook libcrypto.so 中的哈希函数
var targetExports = [
    "EVP_DigestInit_ex",
    "EVP_DigestUpdate",
    "EVP_DigestFinal_ex",
    "EVP_EncryptInit_ex",
    "EVP_EncryptUpdate",
    "EVP_EncryptFinal_ex"
];

targetExports.forEach(function(expName) {
    var addr = Module.findExportByName("libcrypto.so", expName);
    if (addr) {
        console.log("[+] Hooking " + expName);
        Interceptor.attach(addr, {
            onEnter: function(args) {
                console.log("[*] " + expName + " called");
            }
        });
    } else {
        console.log("[-] " + expName + " not found");
    }
});
```

## 6. 实战：Hook 网络数据加密函数

```javascript
// 追踪加密函数的输入输出，提取加密前后的数据
function trace_encryption() {
    // Hook SSL_write（加密输出）
    var ssl_write = Module.findExportByName("libssl.so", "SSL_write");
    if (ssl_write) {
        Interceptor.attach(ssl_write, {
            onEnter: function(args) {
                console.log("\n========== SSL_write ==========");
                console.log("SSL*: " + args[0]);
                console.log("Plaintext (" + args[2].toInt32() + " bytes):");
                console.log(hexdump(args[1], {length: Math.min(args[2].toInt32(), 256)}));
            }
        });
    }
    
    // Hook SSL_read（解密输入）
    var ssl_read = Module.findExportByName("libssl.so", "SSL_read");
    if (ssl_read) {
        Interceptor.attach(ssl_read, {
            onEnter: function(args) {
                this.buf = args[1];
                this.num = args[2].toInt32();
            },
            onLeave: function(retval) {
                var bytesRead = retval.toInt32();
                if (bytesRead > 0) {
                    console.log("\n========== SSL_read ==========");
                    console.log("Plaintext (" + bytesRead + " bytes):");
                    console.log(hexdump(this.buf, {length: Math.min(bytesRead, 256)}));
                }
            }
        });
    }
}

trace_encryption();
```

## 7. 常见问题与排查

### 7.1 导出函数找不到

```javascript
// 排查导出函数找不到的原因
function debug_find_export(moduleName, exportName) {
    var mod = Process.findModuleByName(moduleName);
    if (!mod) {
        console.log("[-] Module " + moduleName + " is not loaded");
        return;
    }
    
    console.log("[*] " + moduleName + " is loaded at " + mod.base);
    
    // 列出所有导出函数，帮助确认正确的函数名
    var exports = mod.enumerateExports();
    console.log("[*] Total exports: " + exports.length);
    
    // 搜索相似名称
    var similar = exports.filter(function(exp) {
        return exp.name.toLowerCase().indexOf(
            exportName.toLowerCase().substring(0, 5)) >= 0;
    });
    
    if (similar.length > 0) {
        console.log("[!] Similar exports found:");
        similar.forEach(function(exp) {
            console.log("    " + exp.name);
        });
    }
}

// 使用
debug_find_export("libssl.so", "SSL_write");
```

### 7.2 导出表被隐藏

某些加固 SO 会从 `.dynsym` 中删除自身函数的符号，使 `Module.findExportByName` 找不到。这时需要使用**偏移 Hook**（详见 8.4 节）。

```javascript
// 判断导出表是否存在
function checkExportsStripped(moduleName) {
    var mod = Process.findModuleByName(moduleName);
    if (!mod) return;
    
    var exports = mod.enumerateExports();
    var symbols = mod.enumerateSymbols();
    
    console.log("[*] " + moduleName);
    console.log("    Exports count: " + exports.length);
    console.log("    Symbols count: " + symbols.length);
    
    // 如果两个计数差距很大，说明被 strip 了
    if (symbols.length > exports.length * 3) {
        console.log("[!] Exports appear to be stripped");
        console.log("[!] Consider using offset-based hook instead");
    }
}
```

## 总结

- **`Module.findExportByName`** 通过导出符号名定位函数，是最直接的 Hook 方式
- **`moduleName=null`** 在所有已加载模块中搜索，用于不确定目标模块的场景
- **SSL_write Hook** 是抓取 HTTPS 明文流量的经典应用
- **hexdump** 是 Frida 内置的十六进制查看工具，对分析二进制数据非常有用
- **导出函数可能被隐藏**（strip），此时需要转向基址+偏移的 Hook 方式
- 导出函数 Hook 适用于：libc 标准函数、SSL 库函数、未加固的 SO 导出函数
