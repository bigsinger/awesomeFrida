# 8.2 onEnter与onLeave详解

> 核心要点：深入理解 onEnter 和 onLeave 回调的参数访问与修改能力，掌握 args、retval、context 和 this 对象的高级用法。

## 1. onEnter 回调详解

### 1.1 args 参数数组

`args` 是一个类数组对象，通过索引访问函数的各个参数：

```javascript
Interceptor.attach(targetAddr, {
    onEnter: function(args) {
        // 基本参数访问
        console.log("arg[0] = " + args[0]);   // 第一个参数
        console.log("arg[1] = " + args[1]);   // 第二个参数
        console.log("arg[2] = " + args[2]);   // 第三个参数
        
        // 参数数量
        console.log("args.length = " + args.length);
    }
});
```

### 1.2 this 上下文对象

`onEnter` 中的 `this` 是一个跨 `onEnter`/`onLeave` 的共享对象，可以在 `onEnter` 中存储数据，在 `onLeave` 中读取：

```javascript
Interceptor.attach(targetAddr, {
    onEnter: function(args) {
        // 存储数据到 this，可在 onLeave 中访问
        this.startTime = Date.now();
        this.callArgs = [];
        for (var i = 0; i < args.length; i++) {
            this.callArgs.push(args[i]);
        }
        this.tid = Process.getCurrentThreadId();
        console.log("[+] Thread " + this.tid + " called function");
    },
    onLeave: function(retval) {
        // 访问 onEnter 中存储的数据
        var elapsed = Date.now() - this.startTime;
        console.log("[+] Returned after " + elapsed + "ms");
        console.log("[+] Original args: " + this.callArgs.join(", "));
    }
});
```

### 1.3 参数类型判断与转换

```javascript
Interceptor.attach(targetAddr, {
    onEnter: function(args) {
        for (var i = 0; i < args.length; i++) {
            var arg = args[i];
            
            // 判断是否是 NativePointer
            if (arg instanceof NativePointer) {
                // 尝试读取字符串
                try {
                    var str = arg.readCString();
                    if (str && str.length > 0 && str.length < 256) {
                        console.log("arg[" + i + "] (string): " + str);
                        continue;
                    }
                } catch (e) {
                    // 不是有效的字符串指针
                }
                
                // 作为整数输出
                console.log("arg[" + i + "] (ptr): " + arg + 
                           " = 0x" + arg.toString(16));
            } else if (typeof arg === 'number') {
                console.log("arg[" + i + "] (int): " + arg);
            } else {
                console.log("arg[" + i + "] (" + typeof arg + "): " + arg);
            }
        }
    }
});
```

## 2. onLeave 回调详解

### 2.1 retval 返回值操作

```javascript
Interceptor.attach(targetAddr, {
    onEnter: function(args) {
        console.log("[+] enter");
    },
    onLeave: function(retval) {
        console.log("[+] leave");
        
        // 读取返回值
        console.log("retval = " + retval);
        console.log("retval (int) = " + retval.toInt32());
        
        // 修改返回值
        retval.replace(ptr(0));    // 将返回值改为 0
        // retval.replace(42);      // 将返回值改为 42
    }
});
```

### 2.2 修改返回值的完整示例

```javascript
// 修改 strcmp 的返回值，使比较永远返回相等
var strcmp_addr = Module.findExportByName("libc.so", "strcmp");

Interceptor.attach(strcmp_addr, {
    onEnter: function(args) {
        this.str1 = args[0].readCString();
        this.str2 = args[1].readCString();
    },
    onLeave: function(retval) {
        console.log("strcmp(\"" + this.str1 + "\", \"" + 
                    this.str2 + "\") = " + retval);
        
        // 如果比较的字符串中包含特定关键字，修改返回值
        if (this.str1.indexOf("password") >= 0 || 
            this.str2.indexOf("password") >= 0) {
            console.log("[!] Overriding strcmp result to 0 (equal)");
            retval.replace(0);  // 让比较结果为"相等"
        }
    }
});
```

## 3. 引用代码：hookNative.js 第13-36行（偏移Hook详解）

```javascript
// 引用：Chap11/hookNative.js 第13-36行
function hook_native3(){
    // 1. 获取 SO 基址
    var libnative_addr = Module.findBaseAddress('libnative-lib.so');
    console.log("libnative_addr is => ", libnative_addr);
    
    // 2. 通过偏移定位未导出函数
    var stringfromJNI3 = libnative_addr.add(0xf444);
    console.log("stringfromJNI3 address is =>", stringfromJNI3);

    // 3. 使用 Interceptor.attach
    Interceptor.attach(stringfromJNI3, {
        onEnter: function(args){
            // JNI 函数的前两个参数
            console.log("jnienv pointer =>", args[0]);    // JNIEnv*
            console.log("jobj pointer =>", args[1]);      // jobject (this)
        },
        onLeave: function(retval){
            // 读取 jstring 返回值
            console.log("retval is =>", 
                Java.vm.getEnv().getStringUtfChars(retval, null).readCString());
            console.log("=================");
        }
    });
}
```

**逐行分析：**

1. **`Module.findBaseAddress('libnative-lib.so')`** — 获取 SO 在内存中的基址。与 `Process.findModuleByName()` 不同，它直接返回 NativePointer 类型的基址

2. **`libnative_addr.add(0xf444)`** — 基址 + 偏移 = 目标函数的绝对地址。这里的 `0xf444` 是通过 IDA Pro 反编译分析得到的函数偏移

3. **`args[0]` = `JNIEnv*`** — 所有 JNI 函数的第一个参数都是 `JNIEnv*`，指向 JNI 函数表

4. **`args[1]` = `jobject`** — 对于实例方法，第二个参数是 Java 对象本身（`this`）；对于静态方法，第二个参数是 `jclass`

5. **`retval` → `jstring`** — 返回值是 Java 字符串引用，需要通过 JNI API 读取

## 4. context 对象：访问寄存器

`this.context` 在 onEnter 和 onLeave 中都可以访问，它提供了当前线程的 CPU 寄存器快照：

```javascript
Interceptor.attach(targetAddr, {
    onEnter: function(args) {
        // ARM64 寄存器
        if (Process.pointerSize === 8) {
            console.log("x0 = " + this.context.x0);   // 第一个参数寄存器
            console.log("x1 = " + this.context.x1);   // 第二个参数寄存器
            console.log("x2 = " + this.context.x2);   // 第三个参数寄存器
            console.log("lr = " + this.context.lr);   // 链接寄存器（返回地址）
            console.log("sp = " + this.context.sp);   // 栈指针
            console.log("fp = " + this.context.fp);   // 帧指针
        } else {
            // ARM32 寄存器
            console.log("r0 = " + this.context.r0);
            console.log("r1 = " + this.context.r1);
            console.log("lr = " + this.context.lr);
        }
    }
});
```

### 4.1 通过 context.lr 追踪调用者

```javascript
// 追踪谁调用了这个函数
Interceptor.attach(targetAddr, {
    onEnter: function(args) {
        var callerAddr = this.context.lr;  // 返回地址
        var callerMod = Process.findModuleByAddress(callerAddr);
        
        if (callerMod) {
            var callerOffset = callerAddr.sub(callerMod.base);
            console.log("[+] Called from " + callerMod.name + 
                       "+0x" + callerOffset.toString(16));
        } else {
            console.log("[+] Called from unknown module at " + callerAddr);
        }
    }
});
```

### 4.2 通过 context 修改寄存器

```javascript
// 注意：修改 context 需要谨慎，可能导致崩溃
Interceptor.attach(targetAddr, {
    onEnter: function(args) {
        // 修改函数返回值寄存器（ARM64 上 x0 保存返回值）
        // 这在函数进入时设置 x0，函数可能不会使用它
        // 但这种方式可以在函数调用前预置返回值
        console.log("[*] Original x0: " + this.context.x0);
        // this.context.x0 = ptr(0);  // ❌ 危险操作，谨慎使用
    },
    onLeave: function(retval) {
        // 在 onLeave 中修改 retval 是更安全的方式
    }
});
```

## 5. 参数修改

### 5.1 修改参数内容

```javascript
// 修改字符串参数的内容
Interceptor.attach(targetAddr, {
    onEnter: function(args) {
        // args[0] 是一个字符串指针
        // 我们可以修改它指向的内存内容
        if (args[0] instanceof NativePointer) {
            var original = args[0].readCString();
            console.log("Original arg: " + original);
            
            // 在堆上分配新字符串
            var newStr = Memory.allocUtf8String("modified");
            // 注意：这样只是替换了指针，不一定会生效
            // 更可靠的方式是直接修改原内存（需要确保可写）
        }
    }
});
```

### 5.2 修改指针参数

```javascript
// 对于指针类型的参数，可以替换指针指向
Interceptor.attach(targetAddr, {
    onEnter: function(args) {
        // 在堆上分配新内存
        var newBuffer = Memory.alloc(256);
        newBuffer.writeUtf8String("overwritten data");
        
        // 将参数指向新内存
        args[0] = newBuffer;
    }
});
```

## 6. 高级用法：多层嵌套 Hook

```javascript
// 在 onEnter 中动态安装新的 Hook
function hook_with_dynamic_subhook() {
    var addr = Module.findExportByName("libc.so", "malloc");
    
    Interceptor.attach(addr, {
        onEnter: function(args) {
            this.size = args[0].toInt32();
            console.log("[malloc] size=" + this.size);
        },
        onLeave: function(retval) {
            console.log("[malloc] returned " + retval + 
                       " for size " + this.size);
            
            // 对分配的内存区域设置访问断点
            if (this.size > 64 && this.size < 128) {
                console.log("[*] Setting memory access hook on " + retval);
                // 可以调用 MemoryAccessMonitor 等高级 API
            }
        }
    });
}
```

## 7. 实用调试模板

```javascript
// 完整的 onEnter/onLeave 调试模板
function debug_function(moduleName, funcSpec) {
    var addr;
    if (typeof funcSpec === 'string') {
        addr = Module.findExportByName(moduleName, funcSpec);
    } else {
        var mod = Process.findModuleByName(moduleName);
        if (!mod) { console.log("Module not found"); return; }
        addr = mod.base.add(funcSpec);
    }
    
    if (!addr) { console.log("Function not found"); return; }
    
    Interceptor.attach(addr, {
        onEnter: function(args) {
            console.log("\n======= " + (funcSpec.name || funcSpec) + " =======");
            console.log("[Thread] " + Process.getCurrentThreadId());
            console.log("[Time] " + new Date().toISOString());
            
            // 打印所有参数
            for (var i = 0; i < args.length; i++) {
                var a = args[i];
                if (a instanceof NativePointer) {
                    try {
                        var s = a.readCString();
                        if (s && s.length > 0 && s.length < 128) {
                            console.log("  arg[" + i + "] = \"" + s + "\"");
                            continue;
                        }
                    } catch(e) {}
                    console.log("  arg[" + i + "] = " + a + " (hex: 0x" + 
                               a.toString(16) + ", int: " + a.toInt32() + ")");
                } else {
                    console.log("  arg[" + i + "] = " + a);
                }
            }
            
            // 显示调用栈
            console.log("[Backtrace]:");
            Thread.backtrace(this.context, Backtracer.ACCURATE)
                .map(DebugSymbol.fromAddress)
                .forEach(function(frame) {
                    console.log("  " + frame);
                });
        },
        onLeave: function(retval) {
            console.log("[Return] " + retval + 
                       " (int: " + (retval.toInt32 ? retval.toInt32() : retval) + ")");
            console.log("======= END =======\n");
        }
    });
}

// 使用示例
debug_function("libc.so", "strcmp");
debug_function("libnative-lib.so", 0xF444);
```

## 8. 常见陷阱

### 8.1 args 数组越界

```javascript
// 不要假设 args 的长度
Interceptor.attach(addr, {
    onEnter: function(args) {
        // ❌ 危险：如果函数只有2个参数，args[5] 可能越界
        // console.log(args[5]);
        
        // ✅ 安全
        try {
            if (args.length > 5) {
                console.log(args[5]);
            }
        } catch(e) {
            console.log("Cannot access args[5]");
        }
    }
});
```

### 8.2 内存读取异常

```javascript
Interceptor.attach(addr, {
    onEnter: function(args) {
        // ❌ 如果 args[0] 不是有效的指针，readCString 会崩溃
        // console.log(args[0].readCString());
        
        // ✅ 安全读取
        try {
            if (args[0] instanceof NativePointer && 
                !args[0].isNull()) {
                var str = args[0].readCString();
                if (str) console.log(str);
            }
        } catch(e) {
            console.log("Cannot read string at " + args[0]);
        }
    }
});
```

## 总结

- **onEnter**：参数 `args` 包含函数的所有传入参数，`this` 可跨回调共享数据
- **onLeave**：通过 `retval` 读取返回值，`retval.replace(newVal)` 修改返回值
- **context**：通过 `this.context` 访问 CPU 寄存器，如 `x0`/`r0`（参数/返回值）、`lr`（返回地址）
- **偏移 Hook**：`Module.findBaseAddress()` + `.add(offset)` 定位未导出函数
- **参数读取**：字符串参数用 `.readCString()`，整数用 `.toInt32()`
- **异常处理**：始终用 try-catch 包裹指针操作，避免因无效内存访问导致进程崩溃
