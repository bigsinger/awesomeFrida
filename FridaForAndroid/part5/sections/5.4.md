# 5.4 Socket 与 SSL Socket Hook

> 当 App 使用底层 Socket（而非 OkHttp/HttpURLConnection）进行网络通信时，如 WebSocket、自定义协议、gRPC 等，需要在 Socket 层进行 Hook。Android 的 TLS 层同样有对应的 Socket 实现。

## 1. 分层 Hook 架构

```
应用层 (HTTP/HTTPS/WebSocket)
   ↓
Socket / SSLSocket ← 在此层 Hook
   ↓
TCP / TLS (Native)
   ↓
Linux Kernel
```

### 2.2 Frida JS 脚本：Hook 地址解析

```javascript
// 引用：AndroidFridaBeginnersBook-main/Chap08/hookSocket.js 第 8-24 行
function hookAddress() {
    Java.perform(function() {
        // java.net.InetSocketAddress.InetSocketAddress(java.net.InetAddress, int)
        Java.use('java.net.InetSocketAddress').$init.overload(
            'java.net.InetAddress', 'int'
        ).implementation = function(addr, port) {
            var result = this.$init(addr, port);

            if (addr.isSiteLocalAddress()) {
                console.log('Local address =>', addr.toString(), ', port is', port);
            } else {
                console.log('Server address =>', addr.toString(), ', port is', port);
            }
            return result;
        };
    });
}
```

### 2.3 Hook Socket 读写数据

```javascript
// 引用：AndroidFridaBeginnersBook-main/Chap08/hookSocket.js 第 25-59 行
function jhexdump(array) {
    var ptr = Memory.alloc(array.length);
    for (var i = 0; i < array.length; ++i)
        Memory.writeS8(ptr.add(i), array[i]);
    console.log(hexdump(ptr, { offset: 0, length: array.length, header: false, ansi: false }));
}

function hookSocket() {
    Java.perform(function() {
        // Hook SocketOutputStream.socketWrite
        Java.use('java.net.SocketOutputStream').socketWrite.overload(
            '[B', 'int', 'int'
        ).implementation = function(bytearray1, int1, int2) {
            var result = this.socketWrite(bytearray1, int1, int2);
            console.log('socketWrite: bytearray1, int1, int2 =>', result, bytearray1, int1, int2);
            jhexdump(bytearray1);
            return result;
        };

        // Hook SocketInputStream.read
        Java.use('java.net.SocketInputStream').read.overload(
            '[B', 'int', 'int'
        ).implementation = function(bytearray1, int1, int2) {
            var result = this.read(bytearray1, int1, int2);
            console.log('read result =>', result, bytearray1, int1, int2);
            jhexdump(bytearray1);
            return result;
        };
    });
}
```

## 3. SSL Socket Hook（Android 8+）

Android 8.0+ 使用 Conscrypt（BoringSSL 的 Java 封装）：

```javascript
// 引用：AndroidFridaBeginnersBook-main/Chap08/hookSocket.js 第 60-92 行
function hookSSLSocketAndroid8() {
    Java.perform(function() {
        // ConscryptFileDescriptorSocket$SSLOutputStream.write
        Java.use('com.android.org.conscrypt.ConscryptFileDescriptorSocket$SSLOutputStream')
            .write.overload('[B', 'int', 'int').implementation = function(bytearry, int1, int2) {
            var result = this.write(bytearry, int1, int2);
            console.log('SSLOutputStream.write =>', result, bytearry, int1, int2);

            var ByteString = Java.use("com.android.okhttp.okio.ByteString");
            console.log('contents:', ByteString.of(bytearry).hex());
            return result;
        };

        // ConscryptFileDescriptorSocket$SSLInputStream.read
        Java.use('com.android.org.conscrypt.ConscryptFileDescriptorSocket$SSLInputStream')
            .read.overload('[B', 'int', 'int').implementation = function(bytearry, int1, int2) {
            var result = this.read(bytearry, int1, int2);
            console.log('SSLInputStream.read =>', result, bytearry, int1, int2);

            var ByteString = Java.use("com.android.okhttp.okio.ByteString");
            console.log('contents:', ByteString.of(bytearry).hex());
            return result;
        };
    });
}
```

## 4. Native 层 SSL_write / SSL_read Hook

对于使用 Native 层 SSL 库的 App，可以在 `libssl.so` 层 Hook：

```javascript
// 引用：AndroidFridaBeginnersBook-main/Chap11/hookSSL_write.js 完整 21 行
function hook_ssl_write() {
    var addr = Module.getExportByName("libssl.so", "SSL_write");
    Interceptor.attach(addr, {
        onEnter: function(args) {
            console.log("\n", hexdump(args[1], { length: args[2].toInt32() }));
        },
        onLeave: function(retval) {
            console.log("================== onLeave =================");
        }
    });
}

function main() {
    console.log("Entering main");
    hook_ssl_write();
}
setImmediate(main);
```

## 7. Binder 调用追踪（辅助）

对于通过 Binder 进行 Socket 操作的场景，可辅助追踪：

```javascript
// 引用：awesomeFrida/BinderProxy.js 第 1-26 行
function hookBinderProxy() {
    const Trace = Java.use('android.os.Trace');
    const Thread = Java.use('java.lang.Thread');
    const BinderProxy = Java.use('android.os.BinderProxy');

    BinderProxy.transact.implementation = function (...args) {
        const stacktrace = Thread.currentThread().getStackTrace();
        const callingStack = stacktrace[3];
        Trace.beginSection(callingStack.toString());
        var result = this.transact(...args);
        Trace.endSection();
        return result;
    };
}
```
