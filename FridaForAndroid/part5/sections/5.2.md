# 5.2 OkHttp newCall Hook

> `OkHttpClient.newCall(Request)` 是每次 OkHttp 请求的入口方法。Hook 此方法可以拦截所有 OkHttp 请求，获取 Request 信息。

## 1. 基本原理

```javascript
// 引用：AndroidFridaBeginnersBook-main/Chap08/hookNewCall.js 完整 14 行
function main() {
    Java.perform(function() {
        var OkHttpClient = Java.use("okhttp3.OkHttpClient")

        OkHttpClient.newCall.implementation = function (request) {
            var result = this.newCall(request)
            console.log(request.toString())
            return result
        };
    });
}
setImmediate(main)
```

`newCall` 方法接收一个 `okhttp3.Request` 参数，返回 `okhttp3.Call` 对象。

## 2. 获取完整 Request 信息

```javascript
function hookNewCall() {
    Java.perform(function() {
        var OkHttpClient = Java.use("okhttp3.OkHttpClient");

        OkHttpClient.newCall.implementation = function (request) {
            // 请求 URL
            console.log("=== newCall ===");
            console.log("URL:", request.url().toString());

            // 请求方法
            console.log("Method:", request.method());

            // 请求头
            console.log("Headers:", request.headers().toString());

            // 请求体（非 GET 请求）
            var body = request.body();
            if (body) {
                console.log("Body contentLength:", body.contentLength());
                var Buffer = Java.use("com.android.okhttp.okio.Buffer");
                var buffer = Buffer.$new();
                body.writeTo(buffer);
                console.log("Body content:", buffer.readUtf8());
            }

            var result = this.newCall(request);
            console.log("Call object:", result);

            return result;
        };
    });
}
```

> **引用**：`awesomeFrida/hook_okhttp3.js` 第 1-121 行展示了更全面的 Hook，`newCall` Hook 是其轻量版本。

## 3. 获取 Response（进阶）

`newCall` 本身不返回 Response，但可以通过 Hook `Call.execute()`（同步）或 `Call.enqueue()`（异步）来获取：

### 3.1 Hook Call.execute（同步请求）

```javascript
Java.perform(function() {
    var OkHttpClient = Java.use("okhttp3.OkHttpClient");
    var Call = Java.use("okhttp3.Call");

    // Hook newCall 记录 Request
    OkHttpClient.newCall.implementation = function (request) {
        console.log("Request URL:", request.url().toString());
        return this.newCall(request);
    };

    // Hook Call.execute 获取 Response
    Call.execute.implementation = function () {
        var response = this.execute();
        console.log("Response code:", response.code());
        console.log("Response headers:", response.headers().toString());

        var body = response.body();
        if (body && body.contentLength() > 0) {
            var source = body.source();
            console.log("Response body:", source.readUtf8());
        }
        return response;
    };
});
```

### 3.2 Hook Call.enqueue（异步请求）

```javascript
Java.perform(function() {
    var Call = Java.use("okhttp3.Call");
    var Callback = Java.use("okhttp3.Callback");

    Call.enqueue.implementation = function (callback) {
        console.log("异步请求开始");

        // 包装 Callback 以获取 Response
        var WrappedCallback = Java.registerClass({
            name: "com.roysue.WrappedCallback",
            implements: [Callback],
            methods: {
                onFailure: function (call, e) {
                    console.log("请求失败:", e.toString());
                    callback.onFailure(call, e);
                },
                onResponse: function (call, response) {
                    console.log("请求成功, code:", response.code());
                    console.log("Response body:", response.body().string());
                    callback.onResponse(call, response);
                }
            }
        });

        return this.enqueue(WrappedCallback.$new());
    };
});
```

## 4. 区分不同 URL 的请求

```javascript
OkHttpClient.newCall.implementation = function (request) {
    var url = request.url().toString();

    if (url.indexOf("api.example.com") !== -1) {
        console.log("[重要API]", url);
        console.log("Headers:", request.headers().toString());
    } else if (url.indexOf("log.example.com") !== -1) {
        // 忽略日志请求
        return this.newCall(request);
    }

    return this.newCall(request);
};
```

## 5. 参数篡改

```javascript
OkHttpClient.newCall.implementation = function (request) {
    // 判断是否是目标 URL
    if (request.url().toString().indexOf("api.target.com") !== -1) {
        var newRequest = request.newBuilder()
            .header("Authorization", "Bearer fake_token_123")
            .url("http://192.168.1.100:8888/proxy")
            .build();
        return this.newCall(newRequest);
    }
    return this.newCall(request);
};
```

## 6. newCall vs Interceptor Hook 对比

| 对比项 | newCall Hook | Interceptor Hook |
|:---:|:---:|:---:|
| **实现复杂度** | ⭐ 简单（几行代码） | ⭐⭐⭐ 中等（需注册类） |
| **请求体** | ✅ 可获取 | ✅ 可获取 |
| **响应体** | ❌ 需额外 Hook | ✅ 可直接获取 |
| **修改请求** | ⚠️ 通过 newBuilder | ✅ 拦截器中修改 |
| **修改响应** | ❌ 不支持 | ✅ 支持 |
| **性能开销** | 低 | 中 |
| **兼容性** | 好（标准 API） | 好（标准 API） |
| **适用场景** | **快速验证、轻量监控** | **全量抓包、数据篡改** |

> **推荐**：如果只需要查看 Request URL 和方法，用 `newCall` Hook 就足够了。如果需要完整的 Request/Response Body，用 Interceptor 注入。
