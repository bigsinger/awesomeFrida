# 5.3 HttpURLConnection Hook

> `HttpURLConnection` 是 Java 标准库中的 HTTP 客户端。虽然 OkHttp 已逐步取代它，但在 Android 8.0 以下版本和中低端 App 中仍广泛使用。此外，`HttpURLConnection` 在 Android 4.4+ 底层也基于 OkHttp 实现。

## 1. 核心 Hook 点

| Hook 点 | 方法 | 作用 |
|:---:|:---|:---|
| **URL** | `$init(String)` | 捕获所有创建的 URL |
| **URL.openConnection** | `openConnection()` | 获取连接对象 |
| **HttpURLConnectionImpl** | `setRequestProperty(String, String)` | 获取请求头 |
| **HttpURLConnectionImpl** | `getInputStream()` / `getOutputStream()` | 获取数据流 |
| **HttpsURLConnection** | `setSSLSocketFactory()` | SSL 相关设置 |
| **HttpsURLConnection** | `setHostnameVerifier()` | Hostname 验证 |

> **引用**：`AndroidFridaBeginnersBook-main/Chap08/HttpURLConnectionHook.js`（24行）

## 2. 基础 Hook 脚本

```javascript
// 引用：AndroidFridaBeginnersBook-main/Chap08/HttpURLConnectionHook.js 完整 24 行
function main() {
    Java.perform(function() {
        // 1. Hook URL 构造函数
        var URL = Java.use('java.net.URL');
        URL.$init.overload('java.lang.String').implementation = function(urlstr) {
            console.log('url =>', urlstr);
            var result = this.$init(urlstr);
            return result;
        };

        // 2. Hook openConnection
        URL.openConnection.overload().implementation = function() {
            var result = this.openConnection();
            console.log('openConnection() returnType =>', result.$className);
            return result;
        };

        // 3. Hook setRequestProperty
        var HttpURLConnectionImpl = Java.use(
            'com.android.okhttp.internal.huc.HttpURLConnectionImpl');
        HttpURLConnectionImpl.setRequestProperty.implementation = function(key, value) {
            var result = this.setRequestProperty(key, value);
            console.log('setRequestProperty =>', key, ':', value);
            return result;
        };
    });
}
setImmediate(main)
```

## 3. 获取完整的 HTTP 请求

```javascript
function hookHttpURLConnectionFull() {
    Java.perform(function() {
        var URL = Java.use('java.net.URL');
        var HttpURLConnection = Java.use('java.net.HttpURLConnection');
        var HttpsURLConnection = Java.use('javax.net.ssl.HttpsURLConnection');

        // 3.1 Hook 所有 URL 创建
        URL.$init.overload('java.lang.String').implementation = function(url) {
            console.log('[URL]', url);
            return this.$init(url);
        };

        // 3.2 Hook HTTP 方法设置
        HttpURLConnection.setRequestMethod.implementation = function(method) {
            console.log('[HTTP Method]', method);
            return this.setRequestMethod(method);
        };

        // 3.3 Hook 获取连接
        HttpURLConnection.connect.implementation = function() {
            console.log('[connect] 正在连接...');
            return this.connect();
        };

        // 3.4 Hook 获取输入流（读取 Response）
        HttpURLConnection.getInputStream.implementation = function() {
            var result = this.getInputStream();
            console.log('[getInputStream] 获取响应流');
            return result;
        };

        // 3.5 Hook 获取输出流（写入 Request Body）
        HttpURLConnection.getOutputStream.implementation = function() {
            var result = this.getOutputStream();
            console.log('[getOutputStream] 获取请求流');
            return result;
        };

        // 3.6 Hook 设置请求头
        HttpURLConnection.setRequestProperty.implementation = function(key, value) {
            // 过滤敏感信息
            if (key.indexOf("Authorization") !== -1) {
                console.log('[Header]', key, ':', value.substring(0, 10) + '...');
            } else {
                console.log('[Header]', key, ':', value);
            }
            return this.setRequestProperty(key, value);
        };

        // 3.7 Hook HttpsURLConnection 的 SSL 设置
        HttpsURLConnection.setSSLSocketFactory.implementation = function(factory) {
            console.log('[SSL] setSSLSocketFactory');
            return this.setSSLSocketFactory(factory);
        };

        // 3.8 Hook 响应码
        HttpURLConnection.getResponseCode.implementation = function() {
            var code = this.getResponseCode();
            console.log('[Response Code]', code);
            return code;
        };
    });
}
```

## 4. 读取 Request Body

`HttpURLConnection` 的请求体通过 `getOutputStream().write()` 写入：

```javascript
function hookOutputStream() {
    Java.perform(function() {
        var OutputStream = Java.use('java.io.OutputStream');

        OutputStream.write.overload('[B').implementation = function(buffer) {
            console.log('[OutputStream.write]', bytesToHex(buffer));
            return this.write(buffer);
        };

        OutputStream.write.overload('[B', 'int', 'int').implementation = function(b, off, len) {
            var subArray = Java.array('byte',
                Array.from({length: len}, (_, i) => b[off + i]));
            console.log('[OutputStream.write(offset)]:', bytesToHex(subArray));
            return this.write(b, off, len);
        });
    });
}

function bytesToHex(bytes) {
    var hex = '';
    for (var i = 0; i < bytes.length && i < 256; i++) {
        hex += ('0' + (bytes[i] & 0xFF).toString(16)).substr(-2);
    }
    return hex;
}
```

## 5. 读取 Response Body

```javascript
function hookInputStream() {
    Java.perform(function() {
        var InputStream = Java.use('java.io.InputStream');
        var BufferedInputStream = Java.use('java.io.BufferedInputStream');
        var BufferedReader = Java.use('java.io.BufferedReader');

        // Hook read 方法
        InputStream.read.overload('[B', 'int', 'int').implementation = function(b, off, len) {
            var bytesRead = this.read(b, off, len);
            if (bytesRead > 0) {
                var data = Java.array('byte',
                    Array.from({length: bytesRead}, (_, i) => b[off + i]));
                console.log('[InputStream.read]', bytesRead, 'bytes:', bytesToHex(data));
            }
            return bytesRead;
        });
    });
}
```

## 6. 实战：完整抓包脚本

```javascript
// hook_httpurlconnection_complete.js
function main() {
    Java.perform(function() {
        var URL = Java.use('java.net.URL');
        var HttpsURLConnection = Java.use('javax.net.ssl.HttpsURLConnection');
        var BufferedReader = Java.use('java.io.BufferedReader');
        var InputStreamReader = Java.use('java.io.InputStreamReader');
        var StringBuilder = Java.use('java.lang.StringBuilder');

        URL.openConnection.implementation = function() {
            var urlStr = this.toString();
            console.log('\n=== HTTP Request ===');
            console.log('URL:', urlStr);

            var conn = this.openConnection();

            if (conn.$className.indexOf('HttpsURLConnection') !== -1) {
                var sslConn = Java.cast(conn, HttpsURLConnection);

                // 替换 HostnameVerifier 为信任所有
                sslConn.setHostnameVerifier(
                    Java.use('com.roysue.TrustAllHostnameVerifier').$new()
                );

                // 获取原始的 SSLSocketFactory 信息
                console.log('SSL SocketFactory:', sslConn.getSSLSocketFactory());
            }

            return conn;
        };

        // Hook 输入流读取操作
        var HttpURLConnection = Java.use('java.net.HttpURLConnection');
        HttpURLConnection.getInputStream.implementation = function() {
            var inputStream = this.getInputStream();
            var currentUrl = this.getURL().toString();
            console.log('📥 Response from:', currentUrl);

            return inputStream;
        };
    });
}
```

## 7. HttpURLConnection 底层实现

在 Android 4.4+ 以上，`HttpURLConnection` 底层实际上也是 OkHttp 实现的：

```
HttpURLConnection
   └── com.android.okhttp.internal.huc.HttpURLConnectionImpl
          └── okhttp3.OkHttpClient (内部)
                 └── newCall → RealCall
```

因此，对于 Android 4.4+ 的设备，**Hook OkHttp 同样可以捕获 HttpURLConnection 的请求**。
