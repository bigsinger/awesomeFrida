# 5.5 SSL Pinning 绕过基础

> **SSL Pinning（证书绑定）** 是 App 防止中间人攻击的核心手段。它将服务端证书（或公钥）固定在客户端代码中，即使系统信任了代理证书，App 也会拒绝连接。本节讲解 Frida 绕过 SSL Pinning 的基础方法。

## 1. SSL Pinning 实现方式

| 类型 | 实现方式 | 典型类/方法 |
|:---:|:---|:---|
| **CertificatePinner** | OkHttp 官方方案，固定证书 Hash | `okhttp3.CertificatePinner.check()` |
| **TrustManager** | 自定义 X509TrustManager | `checkServerTrusted()` |
| **HostnameVerifier** | 自定义 Hostname 验证 | `verify(String, SSLSession)` |
| **KeyStore** | 加载本地证书库 | `KeyStore.load()` |

> **引用**：
> - `Chap10/hookCertificatePinner.js`（14行）— 简单的 CertificatePinner Hook
> - `Chap10/hookCert.js`（41行）— KeyStore/HostnameVerifier 综合绕过

## 2. Hook CertificatePinner

OkHttp 的 `CertificatePinner.check()` 是 SSL Pinning 最常见的实现方式：

```javascript
// 引用：AndroidFridaBeginnersBook-main/Chap10/hookCertificatePinner.js 完整 14 行
function killCertificatePinner() {
    Java.perform(function() {
        console.log("Beginning killCertificatePinner !...");

        // 假设混淆后的类是 z1.g，方法是 a(String, List)
        Java.use("z1.g").a.implementation = function(str, list) {
            console.log("called z1.g.a ~");
            return;  // 直接 return，跳过证书校验
        };
    });
}

function main() {
    killCertificatePinner();
}
setImmediate(main);
```

### 通用版本

对于未混淆的 OkHttp：

```javascript
function bypassCertificatePinner() {
    Java.perform(function() {
        // OkHttp 3.x 的 CertificatePinner
        try {
            var CertificatePinner = Java.use('okhttp3.CertificatePinner');
            CertificatePinner.check.overload(
                'java.lang.String', 'java.util.List'
            ).implementation = function(hostname, pins) {
                console.log('[Bypass] CertificatePinner.check:', hostname);
                return;  // 直接返回，不抛异常
            };
        } catch (e) {
            console.log('okhttp3.CertificatePinner 不存在:', e);
        }

        // squareup.okhttp（Android 系统内置 OkHttp）
        try {
            var squareupPinner = Java.use('com.squareup.okhttp.CertificatePinner');
            squareupPinner.check.overload(
                'java.lang.String', 'java.util.List'
            ).implementation = function(hostname, pins) {
                console.log('[Bypass] squareup.CertificatePinner.check:', hostname);
                return;
            };
        } catch (e) {
            console.log('com.squareup.okhttp.CertificatePinner 不存在');
        }
    });
}
```

## 3. Hook TrustManager

当 App 自定义 `X509TrustManager` 时，需要 Hook `checkServerTrusted`：

```javascript
// 引用：AndroidFridaBeginnersBook-main/Chap10/hookCert.js 第 1-38 行
function hook_KeyStore_load() {
    Java.perform(function() {
        // Hook KeyStore.load 以保存客户端证书（当服务器要求客户端证书时）
        var KeyStore = Java.use("java.security.KeyStore");
        var buffer = Java.array('byte', new Array(1024));

        KeyStore.load.overload(
            'java.security.KeyStore$LoadStoreParameter'
        ).implementation = function(arg0) {
            console.log("KeyStore.load1:", arg0);
            this.load(arg0);
        };

        KeyStore.load.overload(
            'java.io.InputStream', '[C'
        ).implementation = function(arg0, arg1) {
            console.log("KeyStore.load2: filename =", arg0, ', password =',
                arg1 ? Java.use('java.lang.String').$new(arg1) : null);

            if (arg0) {
                // 保存证书到 SD 卡
                var filename = "/sdcard/Download/" + String(arg0);
                var file = Java.use("java.io.File").$new(filename);
                var out = Java.use("java.io.FileOutputStream").$new(file);
                var r;
                while ((r = arg0.read(buffer)) > 0) {
                    out.write(buffer, 0, r);
                }
                console.log('save_path =', filename, ", cert save success!");
                out.close();
            }
            this.load(arg0, arg1);
        };

        console.log("hook_KeyStore_load...");
    });
}
```

## 4. Hook HostnameVerifier

```javascript
function bypassHostnameVerifier() {
    Java.perform(function() {
        // OkHttp 的 HostnameVerifier
        try {
            var OkHostnameVerifier = Java.use('okhttp3.internal.tls.OkHostnameVerifier');
            OkHostnameVerifier.verify.overload(
                'java.lang.String', 'javax.net.ssl.SSLSession'
            ).implementation = function(hostname, session) {
                console.log('[Bypass] HostnameVerifier.verify:', hostname);
                return true;  // 永远返回 true
            };
            OkHostnameVerifier.verify.overload(
                'java.lang.String', 'java.security.cert.X509Certificate'
            ).implementation = function(hostname, cert) {
                console.log('[Bypass] HostnameVerifier.verify(cert):', hostname);
                return true;
            };
        } catch (e) {
            console.log('OkHostnameVerifier 不存在:', e);
        }

        // 标准 HttpsURLConnection 的 HostnameVerifier
        var HttpsURLConnection = Java.use('javax.net.ssl.HttpsURLConnection');
        HttpsURLConnection.setDefaultHostnameVerifier.implementation = function(v) {
            console.log('[Bypass] setDefaultHostnameVerifier blocked');
            // 什么都不做，阻止 App 设置自己的 Verifier
        };
        HttpsURLConnection.setHostnameVerifier.implementation = function(v) {
            console.log('[Bypass] setHostnameVerifier blocked');
        };
    });
}
```

## 5. Hook SSLContext.init

```javascript
function bypassSSLContext() {
    Java.perform(function() {
        var SSLContext = Java.use('javax.net.ssl.SSLContext');

        SSLContext.init.overload(
            '[Ljavax.net.ssl.KeyManager;',
            '[Ljavax.net.ssl.TrustManager;',
            'java.security.SecureRandom'
        ).implementation = function(km, tm, random) {
            console.log('[Bypass] SSLContext.init intercepted');

            // 创建一个信任所有证书的 TrustManager
            var TrustManager = Java.registerClass({
                name: 'com.roysue.TrustAllManager',
                implements: [Java.use('javax.net.ssl.X509TrustManager')],
                methods: {
                    checkClientTrusted: function(chain, authType) {},
                    checkServerTrusted: function(chain, authType) {},
                    getAcceptedIssuers: function() { return []; }
                }
            });

            var trustAllManagers = Java.array(
                'javax.net.ssl.X509TrustManager',
                [TrustManager.$new()]
            );

            return this.init(km, trustAllManagers, random);
        };
    });
}
```

## 6. 综合绕过脚本

```javascript
// ssl_bypass_basic.js
function sslBypass() {
    Java.perform(function() {
        // 1. 绕过 CertificatePinner
        try {
            Java.use('okhttp3.CertificatePinner').check.overload(
                'java.lang.String', 'java.util.List'
            ).implementation = function(h, p) { return; };
        } catch(e) {}

        // 2. 绕过 HostnameVerifier
        try {
            Java.use('okhttp3.internal.tls.OkHostnameVerifier').verify.overload(
                'java.lang.String', 'javax.net.ssl.SSLSession'
            ).implementation = function(h, s) { return true; };
        } catch(e) {}

        // 3. 绕过 TrustManager
        try {
            Java.use('com.android.org.conscrypt.TrustManagerImpl').checkServerTrusted.overload(
                '[Ljava.security.cert.X509Certificate;', 'java.lang.String', 'java.lang.String'
            ).implementation = function(c, a, t) {
                return Java.use('java.util.ArrayList').$new();
            };
        } catch(e) {}

        // 4. 阻止设置自定义 SSLSocketFactory
        Java.use('javax.net.ssl.HttpsURLConnection').setSSLSocketFactory.implementation = function(f) {
            console.log('[Bypass] setSSLSocketFactory blocked');
        };

        console.log('SSL Bypass 已加载');
    });
}
setImmediate(sslBypass);
```
