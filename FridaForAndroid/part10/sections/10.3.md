# 10.3 Frida 脚本手动 Dump DEX

> 核心要点：当自动化脱壳工具失效时，可以编写 Frida 脚本绕过反脱壳机制，手动从内存中定位并 dump 加固壳解密后的 DEX 文件。

## 1. DEX 内存布局搜索方案

### 1.1 方案一：DEX 魔数搜索

```javascript
// 引用：10.1 节 DEX Header magic 格式
// 搜索所有 DEX 版本魔数
function searchDexByMagic() {
    // 定义所有 DEX 版本的魔数
    var magics = [
        "64 65 78 0A 30 33 35 00",  // dex\n035\0
        "64 65 78 0A 30 33 37 00",  // dex\n037\0
        "64 65 78 0A 30 33 38 00",  // dex\n038\0
        "64 65 78 0A 30 33 39 00",  // dex\n039\0
    ];
    
    var allResults = [];
    
    Process.enumerateModules().forEach(function(module) {
        // 跳过 mapped 文件系统区域，减少扫描量
        if (module.path.indexOf(".dex") >= 0) return;
        
        magics.forEach(function(magic) {
            try {
                var matches = Memory.scanSync(module.base, module.size, magic);
                matches.forEach(function(m) {
                    var size = m.address.add(0x20).readU32();
                    var classDefsSize = m.address.add(0x58).readU32();
                    if (size > 1024 && (classDefsSize > 0 || size > 100000)) {
                        allResults.push({
                            address: m.address,
                            size: size,
                            module: module.name,
                            classes: classDefsSize
                        });
                    }
                });
            } catch (e) {
                // 跳过无法读取的内存区域
            }
        });
    });
    
    // 去重并排序
    return allResults;
}
```

### 1.2 方案二：Hook DexFile::OpenMemory

当壳调用 ART 的 `DexFile::OpenMemory` 加载 DEX 时，可以直接拦截并读取 DEX 数据：

```javascript
// Hook ART 的 OpenMemory 函数，在 DEX 被加载时直接获取
function hookOpenMemory() {
    // 引用：10.1 节 4.4 关键 hooks 点
    // libart.so 中 DexFile::OpenMemory 的符号名（Android 8+）
    var symbols = Module.enumerateSymbolsSync("libart.so");
    var openMemorySymbol = null;
    
    // 搜索 OpenMemory 符号
    for (var i = 0; i < symbols.length; i++) {
        var sym = symbols[i];
        if (sym.name.indexOf("DexFile") >= 0 &&
            sym.name.indexOf("OpenMemory") >= 0 &&
            sym.name.indexOf("Check") < 0) {
            openMemorySymbol = sym;
            break;
        }
    }
    
    if (!openMemorySymbol) {
        console.log("[-] OpenMemory not found");
        return;
    }
    
    Interceptor.attach(openMemorySymbol.address, {
        onEnter: function(args) {
            // 参数0: const uint8_t* base  ← DEX 数据指针
            // 参数1: size_t size            ← DEX 数据大小
            this.dexBase = args[0];
            this.dexSize = args[1];
            
            console.log("[+] OpenMemory called, DEX at: " + args[0] + 
                       " size: " + args[1]);
        },
        onLeave: function(retval) {
            if (this.dexBase && this.dexSize > 0) {
                dumpDexMemory(this.dexBase, this.dexSize);
            }
        }
    });
}

// 实际的 DEX dump 函数
function dumpDexMemory(base, size) {
    try {
        Memory.protect(base, size, 'rwx');
        var buffer = base.readByteArray(size);
        var filename = "/data/local/tmp/dex_" + 
                      base.toString(16) + "_" + size + ".dex";
        var file = new File(filename, "wb");
        file.write(buffer);
        file.flush();
        file.close();
        console.log("[+] DEX dumped to: " + filename);
    } catch (e) {
        console.log("[-] Dump failed: " + e);
    }
}

setImmediate(hookOpenMemory);
```

## 2. 通用 DEX Dump 脚本

### 2.1 全自动 DEX Dump 脚本

```javascript
// 综合所有搜索策略的完整 DEX Dump 脚本
'use strict';

var dumpedAddresses = [];  // 已 dump 的地址去重

function dumpDex(base, size, source) {
    // 去重
    var key = base.toString(16) + "_" + size;
    if (dumpedAddresses.indexOf(key) >= 0) return;
    dumpedAddresses.push(key);
    
    try {
        // 尝试修改内存保护为可读写
        Memory.protect(base, size, 'rwx');
    } catch(e) {
        // 内存不可修改，跳过
        return;
    }
    
    // 验证 DEX Header
    var magic = base.readCString(8);
    if (magic.indexOf("dex\n") !== 0) {
        console.log("[-] Invalid magic at " + base + ": " + magic);
        return;
    }
    
    var fileSize = base.add(0x20).readU32();
    var classDefsSize = base.add(0x58).readU32();
    
    console.log("[+] Found DEX at " + base + 
               " size=" + fileSize + 
               " classes=" + classDefsSize +
               " source=" + source);
    
    // 读取完整 DEX 数据
    var buffer = base.readByteArray(fileSize);
    
    // 保存到文件
    var filename = "/data/local/tmp/dex_" + 
                  base.toString(16) + "_" + 
                  fileSize + ".dex";
    var file = new File(filename, "wb");
    file.write(buffer);
    file.flush();
    file.close();
    console.log("[+] Saved: " + filename);
}

// 搜索策略 1：魔数扫描
function scanByMagic() {
    var magics = [
        "64 65 78 0A 30 33 35 00",
        "64 65 78 0A 30 33 37 00",
        "64 65 78 0A 30 33 38 00",
        "64 65 78 0A 30 33 39 00",
    ];
    
    Process.enumerateModules().forEach(function(m) {
        magics.forEach(function(magic) {
            try {
                Memory.scanSync(m.base, m.size, magic).forEach(function(match) {
                    dumpDex(match.address, 0x100000, "magic-scan");
                });
            } catch(e) {}
        });
    });
}

// 搜索策略 2：扫描已加载的 DexFile 对象
function scanDexFileObjects() {
    Java.perform(function() {
        var DexFile = Java.use("dalvik.system.DexFile");
        var BaseDexClassLoader = Java.use("dalvik.system.BaseDexClassLoader");
        
        // 枚举所有 ClassLoader
        Java.enumerateClassLoaders({
            onMatch: function(loader) {
                try {
                    var pathList = loader.pathList.value;
                    var dexElements = pathList.dexElements.value;
                    
                    for (var i = 0; i < dexElements.length; i++) {
                        var element = dexElements[i];
                        var dexFile = element.dexFile.value;
                        
                        if (dexFile !== null) {
                            var mCookie = dexFile.mCookie.value;
                            console.log("[+] DexFile found: " + mCookie);
                        }
                    }
                } catch(e) {}
            },
            onComplete: function() {}
        });
    });
}

// 搜索策略 3：Hook RegisterNatives 辅助定位
// 引用：awesomeFrida/hook_RegisterNatives.js（69行）
function hookRegisterNatives() {
    var symbols = Module.enumerateSymbolsSync("libart.so");
    var addr = null;
    
    for (var i = 0; i < symbols.length; i++) {
        var sym = symbols[i];
        if (sym.name.indexOf("art") >= 0 &&
            sym.name.indexOf("JNI") >= 0 &&
            sym.name.indexOf("RegisterNatives") >= 0 &&
            sym.name.indexOf("CheckJNI") < 0) {
            addr = sym.address;
            break;
        }
    }
    
    if (addr) {
        Interceptor.attach(addr, {
            onEnter: function(args) {
                console.log("[RegisterNatives] method_count: " + args[3]);
                var env = args[0];
                var java_class = args[1];
                try {
                    var class_name = Java.vm.tryGetEnv().getClassName(java_class);
                    console.log("[RegisterNatives] class: " + class_name);
                } catch(e) {}
            }
        });
    }
}

// 主入口
function main() {
    console.log("[*] Starting DEX Dump...");
    
    // 启动所有搜索策略
    scanByMagic();
    scanDexFileObjects();
    hookRegisterNatives();
    
    // 5 秒后再次扫描（等待更多 DEX 加载）
    setTimeout(function() {
        console.log("[*] Second scan...");
        scanByMagic();
    }, 5000);
    
    console.log("[*] DEX Dump script loaded, waiting for DEX files...");
}

setImmediate(main);
```

## 3. 对抗壳的反脱壳机制

### 3.1 绕过内存保护

```javascript
// 部分加固壳会对 DEX 内存区域设置 PROT_NONE
// 使用 mprotect 修改保护属性

function bypassMemoryProtect(address, size) {
    // 引用：awesomeFrida/dump_so.js（51行）第13行
    // Memory.protect(ptr(libso.base), libso.size, 'rwx');
    
    // 方法一：直接修改
    try {
        Memory.protect(address, size, 'rwx');
        return true;
    } catch(e) {
        console.log("[-] Memory.protect failed: " + e);
    }
    
    // 方法二：通过 mprotect syscall
    var mprotectAddr = Module.findExportByName("libc.so", "mprotect");
    var mprotect = new NativeFunction(mprotectAddr, "int", 
        ["pointer", "pointer", "int"]);
    
    // 对齐到页边界
    var pageSize = Process.pageSize;
    var pageStart = ptr(address).and(~(pageSize - 1));
    var pageEnd = ptr(address).add(size);
    var alignedSize = pageEnd.sub(pageStart).toInt32();
    
    var ret = mprotect(pageStart, alignedSize, 7);  // PROT_READ|PROT_WRITE|PROT_EXEC
    return ret === 0;
}
```

### 3.3 绕过魔数篡改

```javascript
// 当壳篡改了魔数但 DEX 结构仍完整时
// 搜索 ClassDef 表来定位

function findDexByClassDef() {
    // ClassDef 起始特征：
    // - classIdx: 合理的 TypeId 索引
    // - accessFlags: 合理取值（1-0x20000）
    // - superclassIdx: 0xFFFFFFFF（无父类）或合理索引
    
    Process.enumerateRanges('rwx').forEach(function(range) {
        // 逐 4 字节扫描
        for (var offset = 0; offset < range.size - 0x70; offset += 4) {
            try {
                var candidate = range.base.add(offset);
                
                // 假设这是 ClassDefsOff，反推 Header 位置
                // 查找 dataOff
                var dataOff = candidate.add(0x68 - 0x5C).readU32();
                
                if (dataOff > 0x70 && dataOff < range.size) {
                    var classDefsSize = candidate.add(0x58 - 0x5C).readU32();
                    if (classDefsSize > 0 && classDefsSize < 10000) {
                        // 验证找到的结构
                        var headerAddr = candidate.sub(0x5C);
                        var magic = headerAddr.readCString(8);
                        
                        if (magic.indexOf("dex") === 0 || 
                            isLikelyDexHeader(headerAddr)) {
                            dumpDex(headerAddr, 0x100000, "classdef-scan");
                        }
                    }
                }
            } catch(e) { break; }
        }
    });
}

function isLikelyDexHeader(addr) {
    try {
        var firstByte = addr.readU8();
        // DEX 文件通常以 0x64 ('d') 开头
        if (firstByte !== 0x64) return false;
        
        var fileSize = addr.add(0x20).readU32();
        if (fileSize < 1024 || fileSize > 0x4000000) return false;  // 1KB-64MB
        
        var dataOff = addr.add(0x68).readU32();
        var dataSize = addr.add(0x6C).readU32();
        if (dataOff < 0x70 || dataSize < 100) return false;
        
        return true;
    } catch(e) {
        return false;
    }
}
```

## 4. 实战：完整脱壳流程

### 4.1 快速脱壳脚本

```javascript
// 快速 DEX Dump - 适用于大部分加固
// 使用方式: frida -U com.example.app -l fast_dex_dump.js

function fastDump() {
    // 等待 3 秒确保壳完成初始化
    setTimeout(function() {
        console.log("[*] Scanning for DEX files...");
        
        Process.enumerateRanges('rwx').forEach(function(range) {
            try {
                var magic = range.base.readCString(8);
                if (magic.indexOf("dex\n") === 0) {
                    var size = range.base.add(0x20).readU32();
                    if (size > 4096 && size < 0x4000000) {
                        console.log("[+] Found DEX: " + range.base + 
                                   " size=" + size + 
                                   " range=" + range.size);
                        
                        Memory.protect(range.base, size, 'rwx');
                        var buffer = range.base.readByteArray(size);
                        
                        var path = "/data/local/tmp/fast_dex_" + 
                                  range.base.toString(16) + ".dex";
                        var f = new File(path, "wb");
                        f.write(buffer);
                        f.flush();
                        f.close();
                        console.log("[+] Saved: " + path);
                    }
                }
            } catch(e) {
                // 跳过不可读区域
            }
        });
        
        console.log("[*] Scan complete");
    }, 3000);
}

setImmediate(fastDump);
```

## 5. 多 DEX 支持

许多应用包含多个 DEX 文件 (classes.dex, classes2.dex, classes3.dex...)，特别是在使用了 MultiDex 的情况下：

```javascript
// 改进的 DEX dump：自动处理多 DEX
function dumpAllDexFiles() {
    var magicPatterns = [
        "64 65 78 0A 30 33 35 00",  // dex035
        "64 65 78 0A 30 33 37 00",  // dex037
        "64 65 78 0A 30 33 38 00",  // dex038
        "64 65 78 0A 30 33 39 00",  // dex039
    ];
    
    var count = 0;
    
    Process.enumerateRanges('rw-').forEach(function(range) {
        magicPatterns.forEach(function(pattern) {
            try {
                var results = Memory.scanSync(range.base, range.size, pattern);
                results.forEach(function(match) {
                    var size = match.address.add(0x20).readU32();
                    if (size > 256 && size < 0x4000000) {
                        count++;
                        var name = "/data/local/tmp/dex_" + count + "_" +
                                  match.address.toString(16) + ".dex";
                        var buf = match.address.readByteArray(size);
                        var f = new File(name, "wb");
                        f.write(buf);
                        f.flush();
                        f.close();
                        console.log("[+] " + count + ": " + name);
                    }
                });
            } catch(e) {}
        });
    });
    
    console.log("[*] Total DEX files dumped: " + count);
}
```

## 总结

- **三种主要方案**：魔数搜索、Hook OpenMemory、扫描 DexFile 对象
- **去重机制**：记录已 dump 的地址，避免重复
- **内存保护绕过**：使用 `Memory.protect(address, size, 'rwx')` 修改保护属性
- **反篡改**：当魔数被修改时，通过 ClassDef 等结构特征定位 DEX
- **组合使用**：推荐同时启动多个搜索策略，在不同时间点重复扫描
- **多 DEX 支持**：循环扫描所有 `rw-` / `rwx` 内存区域
