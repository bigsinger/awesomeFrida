# 10.4 内存 Dump：SO 文件提取

> 核心要点：加固后的 SO（.so）文件在磁盘上是加密的，运行时由壳在内存中解密。通过 Frida 脚本可以从内存中完整读取解密后的 SO，保存为可分析的 ELF 文件。

## 1. SO 脱壳原理

### 1.1 加固 SO 的生命周期

```
磁盘中的加密 SO → 被 App 或壳通过 dlopen 加载
    ↓
linker 读取加密数据到内存
    ↓
壳的 decrypt 函数解密内存中的 SO 内容
    ↓
壳修正 ELF 结构（修复 section header / dynamic segment）
    ↓
★★★ 此时内存中的 SO 是完全解密的状态 ★★★
    ↓
执行 init_array 构造函数 → 壳的反调试等逻辑
    ↓
JNI_OnLoad → 注册 native 方法
```

### 1.2 脱壳时机

| 时机 | 描述 | 成功率 | 风险 |
|------|------|--------|------|
| dlopen onLeave 后 | SO 刚加载完毕，解密已完成 | 高 | 壳可能立即修改内存 |
| 延迟 1-5 秒后 | 等待所有解密完成 | 最高 | 部分壳会抹除内存 |
| init_array Hook 后 | 在构造函数执行前 dump | 中 | 需要精细控制时序 |
| JNI_OnLoad 之前 | 在注册 JNI 函数前 dump | 低 | 时间窗口极短 |

## 2. dump_so.js 完整代码分析

```javascript
// 引用：awesomeFrida/dump_so.js（完整51行代码）
//
// 功能：通过模块名定位已加载的 SO 模块，读取其完整内存数据
// 并保存到应用私有目录

function dump_so(so_name) {
  Java.perform(function () {
    var currentApplication = Java.use("android.app.ActivityThread")
        .currentApplication();
    var dir = currentApplication.getApplicationContext()
        .getFilesDir().getPath();
    
    // 1. 获取 SO 模块信息
    var libso = Process.getModuleByName(so_name);
    console.log("[name]:", libso.name);
    console.log("[base]:", libso.base);
    console.log("[size]:", ptr(libso.size));
    console.log("[path]:", libso.path);
    
    // 2. 构造输出路径
    var file_path = dir + "/" + libso.name + "_" + 
                   libso.base + "_" + ptr(libso.size) + ".so";
    
    // 3. 打开文件
    var file_handle = new File(file_path, "wb");
    if (file_handle && file_handle != null) {
      // 4. 修改内存保护属性（有些 SO 的内存不可读）
      Memory.protect(ptr(libso.base), libso.size, 'rwx');
      
      // 5. 读取 SO 完整内容
      var libso_buffer = ptr(libso.base).readByteArray(libso.size);
      
      // 6. 写入文件
      file_handle.write(libso_buffer);
      file_handle.flush();
      file_handle.close();
      console.log("[dump]:", file_path);
    }
  });
}

// 使用示例
setImmediate(dump_so("libshield.so"));
// 运行: frida -U com.xingin.xhs -l dump_so.js
```

### 2.1 逐行解析

| 行号 | 代码 | 说明 |
|------|------|------|
| 2-4 | `Java.perform` + `ActivityThread.currentApplication` | 获取当前应用上下文，获取私有目录路径 |
| 5 | `Process.getModuleByName(so_name)` | 根据名称获取已加载的 SO 模块对象 |
| 13 | `Memory.protect(ptr(libso.base), libso.size, 'rwx')` | **关键步骤**：将 SO 内存页设为可读写执行，确保能完整读取 |
| 14 | `ptr(libso.base).readByteArray(libso.size)` | 从 SO 基址开始读取完整内存数据 |
| 15-17 | `File.write/flush/close` | 将内存数据写入文件系统 |

## 3. 增强版：dlopen 拦截 + 自动 SO Dump

```javascript
// 引用：awesomeFrida/dump-so.js（完整51行代码）
//
// 增强版：在目标 SO 被 dlopen 加载时自动 dump

const TARGET_LIB_NAME = "libmsaoaidsec.so";
var TargetLibModule = null;

// dump_so 函数同上方
function dump_so(so_name) {
    var module = Process.getModuleByName(so_name);
    console.log("[name]:", module.name);
    console.log("[base]:", module.base);
    console.log("[size]:", ptr(module.size));
    console.log("[path]:", module.path);

    var path = "/data/local/tmp/" + module.name + 
               "_" + module.base + "_" + ptr(module.size) + ".so";
    var handle = new File(path, "wb");
    if (handle && handle != null) {
        Memory.protect(ptr(module.base), module.size, 'rwx');
        var buffer = ptr(module.base).readByteArray(module.size);
        handle.write(buffer);
        handle.flush();
        handle.close();
        console.log("[dump]:", path);
    }
}

// Hook dlopen，在目标 SO 加载时自动 dump
function hook_dlopen() {
    ["android_dlopen_ext", "dlopen"].forEach(funcName => {
        let addr = Module.findExportByName(null, funcName);
        if (addr) {
            Interceptor.attach(addr, {
                onEnter(args) {
                    let libName = ptr(args[0]).readCString();
                    if (libName && libName.indexOf(TARGET_LIB_NAME) >= 0) {
                        this.canHook = true;
                    }
                },
                onLeave: function (retval) {
                    if (this.canHook) {
                        // SO 已加载完毕，立即 dump
                        dump_so(TARGET_LIB_NAME);
                    }
                }
            });
        }
    });
}

hook_dlopen();
// 运行: frida -U com.example.app -l dump-so.js
```

## 4. SO 脱壳进阶技巧

### 4.1 处理多段加密的 SO

某些加固 SO 采用分段解密方式，需等待所有段解密完成：

```javascript
function dumpSoWithDelay(soName, delayMs) {
    setTimeout(function() {
        try {
            var module = Process.findModuleByName(soName);
            if (!module) {
                console.log("[-] Module " + soName + " not loaded yet");
                return;
            }
            
            console.log("[+] Dumping " + soName + 
                       " after " + delayMs + "ms delay");
            
            // 修改保护属性
            Memory.protect(module.base, module.size, 'rwx');
            
            // 读取并保存
            var buffer = module.base.readByteArray(module.size);
            var path = "/data/local/tmp/" + soName + 
                      "_delayed_" + delayMs + ".so";
            var file = new File(path, "wb");
            file.write(buffer);
            file.flush();
            file.close();
            console.log("[+] Saved: " + path);
            
        } catch(e) {
            console.log("[-] Error: " + e);
        }
    }, delayMs);
}

// 使用方法：在多个时间点 dump
function multiDump(soName) {
    dumpSoWithDelay(soName, 1000);   // 1 秒后
    dumpSoWithDelay(soName, 3000);   // 3 秒后
    dumpSoWithDelay(soName, 5000);   // 5 秒后
    dumpSoWithDelay(soName, 10000);  // 10 秒后
}
```

### 4.2 自动修复 ELF 结构

部分壳在解密后不完整修复 section header，需要使用 IDA Pro 或手动修复：

```javascript
// dump 后检测 ELF 完整性
function verifyElf(base, size) {
    // ELF header
    var magic = base.readCString(4);
    if (magic !== "\x7fELF") {
        console.log("[-] Invalid ELF magic");
        return false;
    }
    
    // 读取 ELF header 关键字段
    var e_type = base.add(0x10).readU16();     // ET_EXEC=2, ET_DYN=3
    var e_machine = base.add(0x12).readU16();  // ARM=40, AARCH64=183
    var e_phoff = base.add(0x20).readU32();    // Program header offset
    var e_shoff = base.add(0x28).readU32();    // Section header offset
    
    console.log("[+] ELF type: " + e_type + 
               " machine: " + e_machine +
               " phoff: " + e_phoff +
               " shoff: " + e_shoff);
    
    // 如果 shoff 为 0，表示 section header 已被抹除
    if (e_shoff === 0) {
        console.log("[!] Section headers stripped, need repair");
        console.log("[!] Use readelf -S to verify, or IDA Pro to analyze");
    }
    
    return true;
}
```

### 4.3 结合 RegisterNatives 验证 SO 完整性

```javascript
// 引用：awesomeFrida/hook_RegisterNatives.js（69行）
// 在 dump SO 后，通过 RegisterNatives 输出验证

function hookRegisterNatives() {
    var symbols = Module.enumerateSymbolsSync("libart.so");
    var addrRegisterNatives = null;
    
    for (var i = 0; i < symbols.length; i++) {
        var sym = symbols[i];
        if (sym.name.indexOf("art") >= 0 &&
            sym.name.indexOf("JNI") >= 0 &&
            sym.name.indexOf("RegisterNatives") >= 0 &&
            sym.name.indexOf("CheckJNI") < 0) {
            addrRegisterNatives = sym.address;
            break;
        }
    }
    
    if (addrRegisterNatives) {
        Interceptor.attach(addrRegisterNatives, {
            onEnter: function(args) {
                var methods_ptr = ptr(args[2]);
                var method_count = parseInt(args[3]);
                
                for (var i = 0; i < method_count; i++) {
                    var fnPtr_ptr = Memory.readPointer(
                        methods_ptr.add(i * Process.pointerSize * 3 + 
                                       Process.pointerSize * 2));
                    var find_module = Process.findModuleByAddress(fnPtr_ptr);
                    
                    if (find_module) {
                        // 这个 fnPtr 在我们的 SO 中，说明 dump 应该包含它
                        var offset = ptr(fnPtr_ptr).sub(find_module.base);
                        console.log("  JNI func at " + find_module.name + 
                                   " + 0x" + offset.toString(16));
                    }
                }
            }
        });
    }
}
```

## 5. 通用 SO Dump 脚本模板

```javascript
// 通用 SO Dump 脚本
// 支持：延迟 dump、多 SO 批量 dump、dlopen 自动 dump

'use strict';

// 配置
var config = {
    soNames: ["libshield.so", "libsec.so", "libjiagu.so"],  // 目标 SO
    outputDir: "/data/local/tmp/",
    delayMs: 2000,       // 等待时间
    autoDlopen: true,    // 是否自动 Hook dlopen
};

// dump 单个 SO
function dumpSingleSo(soName) {
    var module = Process.findModuleByName(soName);
    if (!module) {
        console.log("[-] Module not found: " + soName);
        return;
    }
    
    try {
        Memory.protect(module.base, module.size, 'rwx');
    } catch(e) {
        console.log("[-] Memory.protect failed: " + e);
        return;
    }
    
    var buffer = module.base.readByteArray(module.size);
    var filename = config.outputDir + module.name + 
                  "_" + module.base.toString(16) + 
                  "_0x" + module.size.toString(16) + ".so";
    
    var file = new File(filename, "wb");
    file.write(buffer);
    file.flush();
    file.close();
    console.log("[+] Dumped: " + filename + 
               " (" + module.size + " bytes)");
}

// 批量 dump
function dumpAllTargets() {
    config.soNames.forEach(function(name) {
        setTimeout(function() {
            dumpSingleSo(name);
        }, config.delayMs);
    });
}

// 主入口
function main() {
    console.log("[*] SO Dump script loaded");
    console.log("[*] Targets: " + config.soNames.join(", "));
    
    if (config.autoDlopen) {
        // Hook dlopen 实现自动 dump
        ["dlopen", "android_dlopen_ext"].forEach(function(fn) {
            var addr = Module.findExportByName(null, fn);
            if (addr) {
                Interceptor.attach(addr, {
                    onEnter: function(args) {
                        var libName = ptr(args[0]).readCString();
                        config.soNames.forEach(function(target) {
                            if (libName.indexOf(target) >= 0) {
                                console.log("[*] " + target + " loading via " + fn);
                                this.target = target;
                            }
                        });
                    },
                    onLeave: function(retval) {
                        if (this.target) {
                            setTimeout(function() {
                                dumpSingleSo(this.target);
                            }.bind(this), 1000);
                        }
                    }
                });
            }
        });
    }
    
    // 立即 dump（适用于已加载的 SO）
    dumpAllTargets();
}

setImmediate(main);
```

## 6. 常见问题与解决

### 6.1 SO 不可读

```javascript
// 问题：Memory.protect 失败
// 解决：尝试逐页修改保护属性

function forceReadable(base, size) {
    var mprotect = new NativeFunction(
        Module.findExportByName("libc.so", "mprotect"),
        "int", ["pointer", "int", "int"]
    );
    
    var pageSize = 4096;
    var start = ptr(base).and(~(pageSize - 1));
    var alignedSize = (size + pageSize - 1) & ~(pageSize - 1);
    
    var result = mprotect(start, alignedSize, 7);  // RWX
    if (result !== 0) {
        console.log("[-] mprotect failed, errno: " + 
                   Module.findExportByName("libc.so", "errno").readInt());
        return false;
    }
    return true;
}
```

### 6.2 SO 大小不对

```javascript
// 问题：dump 出的 SO 比实际小或包含多余数据
// 解决：通过 Program Header 获取正确大小

function readElfSegmentInfo(base) {
    // 读取 ELF header 中的 program header offset
    var e_phoff = base.add(0x1C).readU32();  // 32-bit ELF
    var e_phentsize = base.add(0x2A).readU16();
    var e_phnum = base.add(0x2C).readU16();
    
    // 遍历 program headers 计算实际范围
    var minVaddr = 0xFFFFFFFF;
    var maxVaddr = 0;
    
    for (var i = 0; i < e_phnum; i++) {
        var ph = base.add(e_phoff + i * e_phentsize);
        var p_type = ph.readU32();
        if (p_type === 1) {  // PT_LOAD
            var p_vaddr = ph.add(0x0C).readU32();
            var p_memsz = ph.add(0x14).readU32();
            if (p_vaddr < minVaddr) minVaddr = p_vaddr;
            if (p_vaddr + p_memsz > maxVaddr) maxVaddr = p_vaddr + p_memsz;
        }
    }
    
    return { memSize: maxVaddr - minVaddr };
}
```

## 总结

- **核心 API**：`Process.getModuleByName()` 获取模块信息，`Memory.protect()` 修改权限，`readByteArray()` 读取内存
- **关键步骤**：修改内存保护属性 `rwx` → 读取完整内存 → 写入文件
- **脱壳时机**：推荐在 dlopen 的 onLeave 中或延迟 1-5 秒后 dump
- **增强技巧**：配合 RegisterNatives Hook 验证 dump 的完整性
- **常见问题**：内存不可读（用 mprotect 绕过）、section header 被抹除（需要手动修复）
