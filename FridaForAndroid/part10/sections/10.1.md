# 10.1 DEX文件结构与加载流程

> 核心要点：DEX（Dalvik Executable）是 Android 平台的可执行文件格式，ART 虚拟机从 DEX 中加载类和方法。理解 DEX 结构是手工脱壳的基础——加固壳的解密 DEX 最终以标准 DEX 格式存在于内存中，通过解析结构特征即可定位和 dump。

## 1. DEX 文件整体布局

```
DEX 文件布局：
┌─────────────────────────┐
│       DEX Header        │  ← 文件头 (0x70 字节)
├─────────────────────────┤
│     String IDs          │  ← 字符串索引表
├─────────────────────────┤
│       Type IDs          │  ← 类型索引表
├─────────────────────────┤
│     Proto IDs           │  ← 方法原型索引表
├─────────────────────────┤
│     Field IDs           │  ← 字段索引表
├─────────────────────────┤
│     Method IDs          │  ← 方法索引表
├─────────────────────────┤
│     Class Defs          │  ← 类定义表（核心定位特征）
├─────────────────────────┤
│       Data Section      │  ← 数据段（代码、注解、调试信息）
├─────────────────────────┤
│     Link Data           │  ← 链接数据
└─────────────────────────┘
```

## 2. DEX Header（文件头）

DEX 文件头固定为 0x70 字节，包含校验和、签名和布局信息：

```c
// dalvik/libdex/DexFile.h
typedef struct DexHeader {
    u1  magic[8];           // DEX 魔数: "dex\n035\0" 或 "dex\n037\0" 或 "dex\n038\0"
    u4  checksum;           // 文件校验和（adler32）
    u1  signature[20];      // SHA-1 签名
    u4  fileSize;           // DEX 文件总大小（字节）
    u4  headerSize;         // 文件头大小（通常 0x70）
    u4  endianTag;          // 字节序标记（0x12345678 = 小端，0x78563412 = 大端）
    u4  linkSize;           // 链接段大小
    u4  linkOff;            // 链接段偏移
    u4  mapOff;             // map 偏移
    u4  stringIdsSize;      // 字符串 ID 数量
    u4  stringIdsOff;       // 字符串 ID 表偏移
    u4  typeIdsSize;        // 类型 ID 数量
    u4  typeIdsOff;         // 类型 ID 表偏移
    u4  protoIdsSize;       // 原型 ID 数量
    u4  protoIdsOff;        // 原型 ID 表偏移
    u4  fieldIdsSize;       // 字段 ID 数量
    u4  fieldIdsOff;        // 字段 ID 表偏移
    u4  methodIdsSize;      // 方法 ID 数量
    u4  methodIdsOff;       // 方法 ID 表偏移
    u4  classDefsSize;      // 类定义数量
    u4  classDefsOff;       // 类定义表偏移
    u4  dataSize;           // 数据段大小
    u4  dataOff;            // 数据段偏移
} DexHeader;
```

### 2.1 Magic 魔数详解

```python
# DEX 魔数格式
magic = b"dex\n035\0"  # Android 4.4 及以下
magic = b"dex\n037\0"  # Android 5.0 - 7.x
magic = b"dex\n038\0"  # Android 8.0 及以上
magic = b"dex\n039\0"  # Android 9.0+ (新版格式)

# 脱壳时通过搜索魔数定位 DEX
# memory.search("6465780A30333500")  # "dex\n035\0" 的十六进制
```

### 2.2 Checksum 校验和

```c
// checksum 使用 adler32 算法
// 计算范围：从 checksum 字段之后开始，到文件末尾
// 即：跳过 magic(8) + checksum(4) = 12 字节后开始

uint32_t compute_adler32(const uint8_t* data, size_t len) {
    const uint32_t MOD_ADLER = 65521;
    uint32_t a = 1, b = 0;
    for (size_t i = 0; i < len; i++) {
        a = (a + data[i]) % MOD_ADLER;
        b = (b + a) % MOD_ADLER;
    }
    return (b << 16) | a;
}
```

## 3. 关键结构体解析

### 3.1 StringIds（字符串索引）

```c
typedef struct DexStringId {
    u4 stringDataOff;  // 指向字符串数据的偏移
} DexStringId;
```

每个方法名、字段名、类名、描述符等都是以字符串形式存储在该段，`stringDataOff` 指向一个 **modified UTF-8** 编码的字符串。

```python
# 字符串存储格式
# [uleb128长度][UTF-8字符数据]
def read_uleb128(data, offset):
    result = 0
    shift = 0
    while True:
        byte = data[offset]
        result |= (byte & 0x7F) << shift
        shift += 7
        offset += 1
        if (byte & 0x80) == 0:
            break
    return result, offset
```

### 3.2 TypeIds（类型索引）

```c
typedef struct DexTypeId {
    u4 descriptorIdx;  // 指向 StringIds 的索引
} DexTypeId;
```

`descriptorIdx` 指向字符串索引，读取字符串内容即为类型描述符，如 `Ljava/lang/String;`、`I`、`[B` 等。

### 3.3 MethodIds（方法索引）

```c
typedef struct DexMethodId {
    u2  classIdx;    // 指向 TypeIds 的索引
    u2  protoIdx;    // 指向 ProtoIds 的索引
    u4  nameIdx;     // 指向 StringIds 的索引
} DexMethodId;
```

通过 `classIdx` 获取所属类，通过 `nameIdx` 获取方法名，通过 `protoIdx` 获取方法签名。

### 3.4 ClassDef（类定义）—— 脱壳核心定位特征

```c
typedef struct DexClassDef {
    u4  classIdx;           // 指向 TypeIds 的索引
    u4  accessFlags;        // 访问标志（public/private/static/final 等）
    u4  superclassIdx;      // 父类索引
    u4  interfacesOff;      // 接口列表偏移
    u4  sourceFileIdx;      // 源文件名索引
    u4  annotationsOff;     // 注解数据偏移
    u4  classDataOff;       // 类数据偏移 ← 指向类的方法/字段定义
    u4  staticValuesOff;    // 静态字段初始值偏移
} DexClassDef;
```

**脱壳中的重要作用：**

1. `classDefsSize` 指示 DEX 中定义的类数量，非零值是确认找到完整 DEX 的重要依据
2. `classDefsOff` 指向 ClassDef 数组，该数组通常位于 Data Section 之前
3. **通过解析 ClassDef 可以验证 DEX 的完整性**

### 3.5 ClassData（类数据）

```c
typedef struct DexClassData {
    DexClassDataHeader header;   // 字段/方法数量
    DexField*          staticFields;    // 静态字段
    DexField*          instanceFields;  // 实例字段
    DexMethod*         directMethods;   // 直接方法（构造函数、私有方法）
    DexMethod*         virtualMethods;  // 虚方法（可继承/重写的方法）
} DexClassData;

typedef struct DexClassDataHeader {
    u4 staticFieldsSize;      // 静态字段数量
    u4 instanceFieldsSize;    // 实例字段数量
    u4 directMethodsSize;     // 直接方法数量
    u4 virtualMethodsSize;    // 虚方法数量
} DexClassDataHeader;

typedef struct DexMethod {
    u4 methodIdx;    // 指向 MethodIds 的索引
    u4 accessFlags;  // 访问标志
    u4 codeOff;      // 指向 CodeItem 的偏移 ← 方法的实际代码
} DexMethod;
```

### 3.6 CodeItem（代码项）

```c
typedef struct DexCode {
    u2  registersSize;   // 寄存器数量
    u2  insSize;         // 参数数量
    u2  outsSize;        // 调用其他方法时的参数数量
    u2  triesSize;       // try-catch 块数量
    u4  debugInfoOff;    // 调试信息偏移
    u4  insnsSize;       // 指令数量（2 字节为单位）
    u2* insns;           // Dalvik 字节码指令流
} DexCode;
```

## 4. DEX 加载流程

### 4.1 类加载器链条

```
Java 层：
    PathClassLoader / DexClassLoader
        ↓
    BaseDexClassLoader.findClass()
        ↓
    DexPathList.findClass()

Native 层（ART）：
    ClassLinker::FindClass()
        ↓
    ClassLinker::LoadClass()  →  从 DexFile 中读取 ClassDef
        ↓
    DexFile::OpenDex()  →  解析 DEX Header
        ↓
    oat_file_manager  →  OAT 编译优化
```

### 4.2 加固壳的典型加载流程

```
App 启动 → Application.onCreate()
    ↓
加固壳初始化（壳 SO 被 dlopen）
    ↓
1. 壳从自身资源/assets中读取加密 DEX
2. 解密 DEX → 获得原始 DEX 字节流
3. 将解密后的 DEX 写入 /data/data/xxx/app_dex/ （部分壳）
4. 调用 OpenDexFile 加载到 ART 运行时
    ↓
完成：加固 DEX 以明文存在于内存中
    ↓
★★★ 脱壳时机：在步骤 4 之后，通过内存搜索魔数定位 ★★★
```

### 4.3 重要数据结构：DexFile

```c
// ART 运行时中 DexFile 的内部表示
class DexFile {
    const uint8_t* begin_;       // DEX 数据起始指针 ← 脱壳直接目标
    size_t         size_;        // DEX 数据大小
    const DexHeader* header_;    // 指向 DEX Header
    // ... 其他字段
};
```

在加固壳完成类加载后，`begin_` 指向的就是解密后的完整 DEX 数据。FRIDA-DEXDump 和手动脚本正是通过搜索魔数或解析 `DexFile` 结构来定位这个内存区域。

## 5. DEX 魔数搜索示例

```javascript
// 手动搜索内存中所有 DEX 魔数
function searchDexInMemory() {
    var dexMagics = ["dex\n035\0", "dex\n037\0", "dex\n038\0", "dex\n039\0"];
    var modules = Process.enumerateModules();
    
    dexMagics.forEach(function(magic) {
        modules.forEach(function(module) {
            var results = Memory.scanSync(
                module.base, 
                module.size, 
                magicsToPattern(magic)
            );
            results.forEach(function(m) {
                // 验证找到的是否是有效的 DEX Header
                var header = parseDexHeader(m.address);
                if (header.isValid()) {
                    console.log("[DEX Found] " + m.address + 
                              " size: " + header.fileSize + 
                              " classes: " + header.classDefsSize);
                }
            });
        });
    });
}

function magicsToPattern(magic) {
    var hex = "";
    for (var i = 0; i < magic.length; i++) {
        hex += ("0" + magic.charCodeAt(i).toString(16)).substr(-2);
    }
    return hex;
}
```

## 6. 验证 DEX 完整性的方法

```javascript
function verifyDex(address, size) {
    var magic = ptr(address).readCString(8);
    if (magic.indexOf("dex\n") !== 0) {
        return { valid: false, reason: "Invalid magic" };
    }
    
    var headerSize = ptr(address).add(0x70).readU32();
    if (headerSize !== 0x70) {
        return { valid: false, reason: "Invalid header size" };
    }
    
    var classDefsSize = ptr(address).add(0x58).readU32();  // offset 0x58
    var classDefsOff = ptr(address).add(0x5C).readU32();   // offset 0x5C
    
    if (classDefsSize === 0 || classDefsOff < 0x70) {
        return { valid: false, reason: "Invalid classDefs" };
    }
    
    var dataOff = ptr(address).add(0x68).readU32();
    var dataSize = ptr(address).add(0x6C).readU32();
    
    if (dataOff > size || dataOff + dataSize > size) {
        return { valid: false, reason: "Data section out of bounds" };
    }
    
    return { valid: true, classCount: classDefsSize };
}
```

## 总结

- **DEX Header 的 magic 字段**（`dex\n035\0` / `037\0` / `038\0` / `039\0`）是脱壳时定位 DEX 的首要特征
- **ClassDefs** 是验证 DEX 完整性的关键：`classDefsSize` 不为零且 `classDefsOff` 指向合理位置
- **脱壳的黄金时机**在加固壳完成 DEX 解密并调用 `OpenDexFile` 之后
- 通过搜索魔数 + 验证 header 字段，即可从内存中完整 dump 出加固后的原始 DEX
- DEX 结构理解是手工脱壳的基础，加固壳最终呈现的还是标准 DEX 格式
