# 6.1 JustTrustMe 通杀方案

> **JustTrustMe** 最初是基于 Xposed 的 SSL Pinning 绕过模块。本节内容将其所有 Hook 点移植到 Frida 环境中，同时提供自动定位混淆 OkHttp 类名的额外功能，覆盖 OkHttp / 原生 TLS / HttpURLConnection 等 20+ 个 Hook 点。

## 1. 整体架构

```
just_trust_me.js
├── 辅助函数（classExists, loadDexfile, newArrayList）
├── processOkHttp()          — OkHttp 相关 Hook
├── processXutils()          — XUtils 相关 Hook
├── processHttpClientAndroidLib()  — HttpClient 相关 Hook
├── processConscryptPlatform()     — Conscrypt 平台相关
├── processPinningTrustManager()   — Appcelerator 相关
└── Java.perform (主体)      — DefaultHttpClient / SSLContext / TrustManager
```

> **引用**：`awesomeFrida/just_trust_me.js`（409 行完整代码）

## 2. 辅助函数

```javascript
// 引用：awesomeFrida/just_trust_me.js 第 1-16 行
function classExists(className) {
    var exists = false;
    try {
        var clz = Java.use(className);
        exists = true;
    } catch (err) {}
    return exists;
}

function loadDexfile(dexfile) {
    Java.perform(function () {
        Java.openClassFile(dexfile).load();
    });
}

function newArrayList() {
    var ArrayListClz = Java.use('java.util.ArrayList');
    return ArrayListClz.$new();
}
```

- `classExists()` — 安全地检查类是否存在，避免 `Java.use` 抛出异常
- `loadDexfile()` — 加载外部 Dex 辅助文件（如 JustTrustMe 的 Helper 类）
- `newArrayList()` — 快速创建空的 ArrayList，用于替代证书校验返回值

## 3. OkHttp 相关 Hook（processOkHttp）

`just_trust_me.js` 第 38-108 行是核心 OkHttp 绕过：

### 3.1 CertificatePinner.check

```javascript
// 引用：just_trust_me.js 第 38-64 行
function processOkHttp() {
    // squareup OkHttp（系统内置）
    if (classExists("com.squareup.okhttp.CertificatePinner")) {
        var squareupPinner = Java.use('com.squareup.okhttp.CertificatePinner');
        var check = squareupPinner.check.overload('java.lang.String', 'java.util.List');
        check.implementation = function(v0, v1) {
            console.log("com.squareup.okhttp.CertificatePinner.check was hooked!");
            // 什么都不做，直接 return
        };
    }

    // okhttp3.CertificatePinner
    if (classExists("okhttp3.CertificatePinner")) {
        try {
            var okhttp3Pinner = Java.use('okhttp3.CertificatePinner');
            var check = okhttp3Pinner.check.overload('java.lang.String', 'java.util.List');
            check.implementation = function(v0, v1) {
                console.log("okhttp3.CertificatePinner.check was hooked!");
            };
        } catch (error) {
            console.error("CertificatePinner.check 可能被混淆了");
        }
    }
    // ...
}
```

### 3.3 sslSocketFactory 替换

```javascript
// 引用：just_trust_me.js 第 88-108 行
if (classExists("okhttp3.OkHttpClient$Builder")) {
    try {
        var Builder = Java.use('okhttp3.OkHttpClient$Builder');
        // 单参数 overload
        Builder.sslSocketFactory.overload('javax.net.ssl.SSLSocketFactory')
            .implementation = function(sSLSocketFactory) {
            var ret = this.sslSocketFactory.call(this,
                Java.use("gz.justtrustme.Helper").getEmptySSLFactory());
            return ret;
        };
        // 双参数 overload
        Builder.sslSocketFactory.overload(
            'javax.net.ssl.SSLSocketFactory', 'javax.net.ssl.X509TrustManager')
            .implementation = function(sSLSocketFactory, x509TrustManager) {
            var ret = this.sslSocketFactory.call(this,
                Java.use("gz.justtrustme.Helper").getEmptySSLFactory(),
                x509TrustManager);
            return ret;
        };
    } catch (error) {
        console.error("sslSocketFactory 方法可能被混淆了");
    }
}
```

## 6. Conscrypt Platform（新增 Hook 点）

```javascript
// 引用：just_trust_me.js 第 146-204 行
function processConscryptPlatform() {
    if (!classExists("com.android.org.conscrypt.Platform")) return;

    var Platform = Java.use('com.android.org.conscrypt.Platform');

    // 4 个 checkServerTrusted 的重载都要 Hook
    var sigs = [
        ['javax.net.ssl.X509TrustManager', '[Ljava.security.cert.X509Certificate;',
         'java.lang.String', 'com.android.org.conscrypt.OpenSSLEngineImpl'],
        ['javax.net.ssl.X509TrustManager', '[Ljava.security.cert.X509Certificate;',
         'java.lang.String', 'com.android.org.conscrypt.OpenSSLSocketImpl'],
        ['javax.net.ssl.X509TrustManager', '[Ljava.security.cert.X509Certificate;',
         'java.lang.String', 'com.android.org.conscrypt.AbstractConscryptSocket'],
        ['javax.net.ssl.X509TrustManager', '[Ljava.security.cert.X509Certificate;',
         'java.lang.String', 'com.android.org.conscrypt.ConscryptEngine']
    ];

    sigs.forEach(function(sig) {
        try {
            var method = Platform.checkServerTrusted.overload.apply(
                Platform.checkServerTrusted, sig);
            method.implementation = function() {
                console.log("Platform.checkServerTrusted was hooked!");
            };
        } catch(e) {}
    });
}
```

## 7. 自动定位混淆 OkHttp 类（Hook Finder）

`just_trust_me_okhttp_hook_finder.js` 可以在运行时自动定位混淆后的 OkHttp 类名：

```javascript
// 引用：awesomeFrida/just_trust_me_okhttp_hook_finder.js 完整 70 行
var matchClassNameRegExp = RegExp(/^okhttp3/);  // 可修改前缀

Java.perform(function() {
    var OkHttp3FakeFinder = Java.use("gz.justtrustme.OkHttp3FakeFinder");
    Java.enumerateLoadedClasses({
        onMatch: function(className) {
            if (className.match(matchClassNameRegExp)) {
                try {
                    let tempClassHandle = Java.use(className).class;
                    // 检查是否是混淆的 OkHttpClient.Builder
                    let builderCheck =
                        OkHttp3FakeFinder.okHttpClientBuilderCheck(tempClassHandle);
                    if (builderCheck) {
                        printOkhttp3FakeClass(builderCheck);
                        return;
                    }
                    // 检查是否是混淆的 CertificatePinner
                    let pinnerCheck =
                        OkHttp3FakeFinder.okHttpCertificatePinnerCheck(tempClassHandle);
                    if (pinnerCheck) {
                        printOkhttp3FakeClass(pinnerCheck);
                        return;
                    }
                    // 检查是否是混淆的 HostnameVerifier
                    let verifierCheck =
                        OkHttp3FakeFinder.okHttpOkHostnameVerifierVerify(tempClassHandle);
                    if (verifierCheck) {
                        printOkhttp3FakeClass(verifierCheck);
                        return;
                    }
                } catch(error) {}
            }
        },
        onComplete: function() {}
    });
});
```

`loadDexfile('/data/user/0/com.smile.gifmaker/radar.dex')` 加载了 JustTrustMe 的 Dex 辅助包，其中的 `OkHttp3FakeFinder` 类负责自动匹配混淆后的类名。

## 8. 使用说明

```bash
# 步骤 1：将 just_trust_me.js 和辅助 Dex 推送到设备
adb push just_trust_me.js /data/local/tmp/
# JustTrustMe 的 Dex 辅助包需要放在对应目录
# 引用：just_trust_me.js 第 22 行，默认路径 /data/user/0/com.smile.gifmaker/radar.dex

# 步骤 2：注入 Frida
frida -U -f com.target.app -l just_trust_me.js --no-pause

# 步骤 3：如果 OkHttp 被混淆，使用 Hook Finder
frida -U -f com.target.app -l just_trust_me_okhttp_hook_finder.js --no-pause
```

> **引用**：`awesomeFrida/just_trust_me.js`（409 行）完整实现了 JustTrustMe 的 Frida 版本，覆盖 20+ 个 Hook 点；`awesomeFrida/just_trust_me_okhttp_hook_finder.js`（70 行）提供了混淆 OkHttp 自动定位能力。
