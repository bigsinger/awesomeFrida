# 6.3 字节系 App 抓包方案

> 字节跳动系 App（抖音、今日头条、西瓜视频等）使用自研的 **chromium 网络栈**，核心 SO 为 `libsscronet.so`。该 SO 使用 BoringSSL 的自定义校验机制，常规 SSL Pinning 绕过方案无效。本节介绍三类绕过方案。

## 1. 字节系 App 的网络架构

```
App (抖音/头条/西瓜视频)
   ↓
libsscronet.so (Chromium 网络栈)
   ├── CronetEngine (Java 层)
   ├── BoringSSL (Native 层)
   └── SSL_CTX_set_custom_verify (自定义校验)
```

> **引用**：`awesomeFrida/byPassSSL.js` / `byteDance.js`

## 2. 方案一：Hook SSL_CTX_set_custom_verify（推荐）

这是最通用的方案，也是 `byteDance.js` 推荐的方法：

```javascript
// 引用：awesomeFrida/byteDance.js 第 88-111 行
function main() {
    Java.perform(function () {
        const soName = 'libsscronet.so';

        onLoad(soName, () => {
            // SSL_CTX_set_custom_verify 原型：
            // void SSL_CTX_set_custom_verify(SSL_CTX *ctx, int mode,
            //     enum ssl_verify_result_t (*callback)(SSL *ssl, uint8_t *out_alert));
            // BoringSSL 文档：ctx->custom_verify_callback = callback;

            let SSL_CTX_set_custom_verify = Module.getExportByName(
                soName, 'SSL_CTX_set_custom_verify');

            if (SSL_CTX_set_custom_verify != null) {
                Interceptor.attach(SSL_CTX_set_custom_verify, {
                    onEnter: function(args) {
                        // args[2] 是注册的自定义校验回调
                        Interceptor.attach(args[2], {
                            onLeave: function(retval) {
                                // ssl_verify_result_t:
                                //   ssl_verify_ok     = 0
                                //   ssl_verify_invalid = 1
                                //   ssl_verify_retry  = 2
                                // 大于 0 的返回值替换为 ssl_verify_ok
                                if (retval > 0x0) retval.replace(0x0);
                            }
                        });
                    }
                });
            }
        });
    });
}
```

### 执行流程

```
1. onLoad('libsscronet.so') → 监听 SO 加载
2. SO 加载完成 → 查找 SSL_CTX_set_custom_verify 导出符号
3. Hook 该函数 → 在 onEnter 中再次 Hook callback 参数
4. callback.onLeave → 将返回值从 ssl_verify_invalid(1) 替换为 ssl_verify_ok(0)
```

## 3. 方案二：内存搜索 + 指令 Patch

```javascript
// 引用：awesomeFrida/byteDance.js 第 48-71 行（方法1，已注释）
onLoad(soName, function () {
    let libsscronet = Process.getModuleByName(soName);

    // 搜索特征码（ARM64 指令）
    const verifyCertMatches = Memory.scanSync(
        libsscronet.base,
        libsscronet.size,
        "E0 E3 00 91 C1 14 80 12"  // 特征码
    );

    verifyCertMatches.forEach(function (result) {
        // 将 mov w0, #1 改为 mov w0, #0
        let verifyCert = result.address.add(0xC);
        if (Instruction.parse(verifyCert).toString() === "mov w0, #1") {
            Memory.protect(verifyCert, 4, 'rwx');
            Memory.writeByteArray(verifyCert, [0x00, 0x00, 0x80, 0x52]);  // mov w0, #0
        }

        // 同时 Patch 调用 handleVerifyResult 的结果
        let handleVerifyInstruction = Instruction.parse(result.address.add(0x1A4));
        if (Instruction.parse(result.address.add(0x1A0)).toString() === "mov x0, x19"
            && handleVerifyInstruction.mnemonic === "bl") {
            let handleVerifyResult = new NativePointer(
                handleVerifyInstruction.opStr.replace('#', ''));
            Interceptor.attach(handleVerifyResult, {
                onLeave: function(retval) {
                    if (retval > 0x0) retval.replace(0x0);
                }
            });
        }
    });
});
```

**缺点**：特征码依赖 SO 版本，不同版本的字节系 App 可能有不同的特征码。

## 4. 方案三：固定偏移 Patch

```javascript
// 引用：awesomeFrida/byteDance.js 第 74-84 行（方法2，已注释）
onLoad(soName, function () {
    let libsscronet = Module.getBaseAddress(soName);
    let verifyCert = libsscronet.add(0x3700F0);         // 验证结果处理函数
    let handleVerifyResult1 = libsscronet.add(0x370448); // 校验结果1
    let handleVerifyResult2 = libsscronet.add(0x370494); // 校验结果2

    console.log("修改前:", Instruction.parse(verifyCert),
        Instruction.parse(handleVerifyResult1),
        Instruction.parse(handleVerifyResult2));

    patch(verifyCert);          // 改为 mov w0, #0
    patch(handleVerifyResult1);  // 改为 mov w0, #0
    patch(handleVerifyResult2);  // 改为 mov w0, #0

    console.log("修改后:", Instruction.parse(verifyCert),
        Instruction.parse(handleVerifyResult1),
        Instruction.parse(handleVerifyResult2));
});
```

**缺点**：偏移量随版本变化，需要逆向分析每个版本。

## 5. 三种方案对比

| 方案 | 实现复杂度 | 兼容性 | 推荐度 |
|:---:|:---:|:---:|:---:|
| **方案一：Hook SSL_CTX_set_custom_verify** | ⭐ 低 | ✅ 跨版本通用 | ⭐⭐⭐ **推荐** |
| **方案二：内存扫描特征码** | ⭐⭐⭐ 高 | ⚠️ 依赖特征码 | ⭐⭐ |
| **方案三：固定偏移 Patch** | ⭐⭐ 中 | ❌ 每个版本不同 | ⭐ |

> `byteDance.js` 的代码注释也明确建议使用方案一（第 86 行注释）。

## 6. onLoad 工具函数

两种实现方式：

### 6.1 原生方式（推荐）

```javascript
// 引用：byteDance.js 第 26-42 行
function onLoad(name, callback) {
    const android_dlopen_ext = Module.findExportByName(null, "android_dlopen_ext");
    if (android_dlopen_ext != null) {
        Interceptor.attach(android_dlopen_ext, {
            onEnter: function(args) {
                if (args[0].readCString().indexOf(name) !== -1) {
                    this.hook = true;
                }
            },
            onLeave: function(retval) {
                if (this.hook) callback();
            }
        });
    }
}
```

### 6.2 Java System.loadLibrary 方式

```javascript
// 引用：byteDance.js 第 6-23 行（已注释的备选方案）
function onLoad(name, callback) {
    var Runtime = Java.use('java.lang.Runtime');
    var VMStack = Java.use('dalvik.system.VMStack');

    System.loadLibrary.overload('java.lang.String').implementation = function(libName) {
        if (libName.includes(name)) {
            callback();  // 注意：这里不会执行到，因为 SO 加载是异步的
        }
    };
}
```

这种方式已被注释掉，因为 `System.loadLibrary` 的 Java 层 Hook 无法捕获到原生加载完成后的事件。

## 7. 使用命令

```bash
# 直接使用，等待 App 加载 libsscronet.so 后自动绕过
frida -U -f com.ss.android.ugc.aweme -l byteDance.js --no-pause

# 超时等待 SO 加载（如果 Spawn 后 SO 未立即加载）

```
