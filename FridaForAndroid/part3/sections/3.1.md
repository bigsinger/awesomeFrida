# 3.1 Java.use 定位类与方法

## 概述

`Java.use(className)` 是 Frida Java 层 Hook 的核心入口。它获取一个 Java 类的包装对象，之后可通过该对象访问类的所有方法、构造函数和字段。

## 基本用法

```javascript
// 引用自 Chap03/1.js 第 5 行（Java.use 用法）
function main(){
    console.log("Script loaded successfully ")
    Java.perform(function(){
        console.log("Inside java perform function")
        var MainAcitivity = Java.use('com.roysue.demo02.MainActivity')
        console.log("Java.Use.Successfully!") //定位类成功！
        // ...
    })
}
setImmediate(main)
```

> **来源**：`Chap03/1.js` 第 5 行，展示了 `Java.use()` 的标准用法。

### 返回值

`Java.use()` 返回一个 **类包装对象**，包含：

- 类的所有方法（可 Hook）
- 构造函数（`$init`）
- 字段（可读写）
- 静态字段和方法

```javascript
// Java.use 返回的对象可以做什么？
var MainActivity = Java.use('com.roysue.demo02.MainActivity');

// Hook 方法
MainActivity.fun.implementation = function() { };

// 访问静态字段
MainActivity.STATIC_FIELD.value

// 调用静态方法
MainActivity.staticMethod();

// 获取类的父类
var ParentClass = MainActivity.$super;

// 获取类加载器
var classLoader = MainActivity.class.getClassLoader();
```

## 定位系统的类

```javascript
// 系统类同样可以使用 Java.use 定位
Java.perform(function() {
    // Android 系统类
    var Activity = Java.use('android.app.Activity');
    var TextView = Java.use('android.widget.TextView');
    var Context = Java.use('android.content.Context');
    var Intent = Java.use('android.content.Intent');
    var Bundle = Java.use('android.os.Bundle');
    
    // Java 标准库
    var HashMap = Java.use('java.util.HashMap');
    var String = Java.use('java.lang.String');
    var Integer = Java.use('java.lang.Integer');
    
    console.log("[+] System classes loaded successfully");
});
```

## beautiful 类名格式

Frida 中类名使用 **Java 内部格式**：

| 类型 | Java 代码中的类名 | Frida 中的类名 |
|------|-------------------|----------------|
| 普通类 | `MainActivity` | `com.roysue.demo02.MainActivity` |
| 内部类 | `InnerClass` | `com.roysue.demo02.MainActivity$InnerClass` |
| 数组 | `int[]` | `[I` |
| 匿名类 | `$1` | `com.roysue.demo02.MainActivity$1` |

```javascript
// 内部类的定位方式
var InnerClass = Java.use('com.roysue.demo02.MainActivity$InnerClass');
var AnonymousClass = Java.use('com.roysue.demo02.MainActivity$1');
```

## 枚举类的所有方法

使用 `class.methods` 可以获取类的所有方法列表（用于探索未知类）：

```javascript
// 引用自 awesomeFrida/all_classes.js 第 1-12 行（枚举类结构）
function enumerateClassMethods(className) {
    Java.perform(function() {
        try {
            var clazz = Java.use(className);
            
            // 枚举所有方法
            var methods = clazz.class.getDeclaredMethods();
            console.log("[+] Methods of " + className + ":");
            
            for (var i = 0; i < methods.length; i++) {
                var method = methods[i];
                console.log("  " + method.getName() + 
                           " (" + method.getParameterTypes() + ")");
            }
            
            // 枚举所有字段
            var fields = clazz.class.getDeclaredFields();
            console.log("[+] Fields:");
            for (var i = 0; i < fields.length; i++) {
                console.log("  " + fields[i].getName() + 
                           " : " + fields[i].getType());
            }
            
        } catch(e) {
            console.log("[-] Error: " + e);
        }
    });
}

// 使用
enumerateClassMethods('com.roysue.demo02.MainActivity');
```

> **来源**：`awesomeFrida/all_classes.js` 第 1-12 行，展示了枚举类的通用模式。

## 验证类是否存在

```javascript
function classExists(className) {
    try {
        Java.perform(function() {
            Java.use(className);
            console.log("[+] Class " + className + " exists");
        });
        return true;
    } catch(e) {
        console.log("[-] Class " + className + " not found");
        return false;
    }
}

// 批量检查
var classesToCheck = [
    'com.roysue.demo02.MainActivity',
    'com.roysue.demo02.util.Arith',
    'com.roysue.demo02.BuildConfig'
];

classesToCheck.forEach(function(cls) {
    classExists(cls);
});
```

## 动态加载类的处理

对于使用 `DexClassLoader` 动态加载的类，`Java.use()` 可能暂时找不到。这时需要延迟重试或 Hook 类加载器：

```javascript
// 引用自 awesomeFrida/loadClass.js 第 1-27 行（loadClass 监控）
function hookClassLoader() {
    Java.perform(function() {
        var DexClassLoader = Java.use("dalvik.system.DexClassLoader");
        DexClassLoader.loadClass.overload('java.lang.String')
            .implementation = function(name) {
                var result = this.loadClass(name);
                console.log("[DexClassLoader] Loading: " + name);
                
                // 找到目标类后自动 Hook
                if (name.indexOf("Application") != -1 && 
                    name != "android.app.Application") {
                    hookApplicationClass(name);
                }
                
                return result;
            };
    });
}

function hookApplicationClass(className) {
    Java.perform(function() {
        var app = Java.use(className);
        console.log("[+] Dynamically loaded: " + className);
        // 在这里添加对动态加载类的 Hook
    });
}

setImmediate(hookClassLoader);
```

> **来源**：`awesomeFrida/loadClass.js` 第 1-27 行，展示了 `DexClassLoader.loadClass` 的 Hook 方法。

## 常见错误

### Error: ClassNotFound

```javascript
// ❌ 类不存在（拼写错误或未加载）
// var Cls = Java.use('com.roysue.demo02.MainActivityy'); // Error!

// ✅ 确认类名正确
// 可以通过 jadx-gui 或 frida-discover 查看实际类名
```

### Error: must be called from Java.perform

```javascript
// ❌ Java.use 必须在 Java.perform 内调用
// var Cls = Java.use('...');  // Error!

// ✅ 正确
Java.perform(function() {
    var Cls = Java.use('com.roysue.demo02.MainActivity');
});
```

## 小结

- `Java.use(className)` 是 Java 层 Hook 的**入口函数**
- 必须放在 `Java.perform()` 内部调用
- 类名使用 **完整包名**：`com.roysue.demo02.MainActivity`
- 内部类用 `$` 分隔：`OuterClass$InnerClass`
- 通过 `class.getDeclaredMethods()` 枚举方法
- 动态加载的类可通过 Hook `DexClassLoader.loadClass` 捕获
