# 3.2 Hook 方法与修改返回值

## 概述

Hook 方法的核心是替换方法的 `implementation` 属性。通过这种方式，你可以：

- 拦截方法调用，查看参数
- 修改参数值
- 修改返回值
- 完全替换方法逻辑

## 基本实现语法

```javascript
ClassName.methodName.implementation = function(original_args...) {
    // 你的代码
    var result = this.methodName(original_args...);  // 调用原始方法
    return result;  // 返回结果（可修改）
};
```

## 标准 Hook 示例

```javascript
// 引用自 Chap03/1.js 第 5-11 行（implementation 标准用法）
function main(){
    console.log("Script loaded successfully ")
    Java.perform(function(){
        console.log("Inside java perform function")
        var MainAcitivity = Java.use('com.roysue.demo02.MainActivity')
        console.log("Java.Use.Successfully!") //定位类成功！
        MainAcitivity.fun.implementation = function(x,y){
            console.log("x => ",x,", y => ",y)
            var ret_value = this.fun(x, y);
            return ret_value
        }
    })
}
setImmediate(main)
```

> **来源**：`Chap03/1.js` 第 5-11 行，展示了 `implementation` 的标准用法。

### 代码解析

| 行 | 代码 | 作用 |
|----|------|------|
| 5 | `Java.use('...MainActivity')` | 定位目标类 |
| 7 | `MainAcitivity.fun.implementation = ...` | 替换方法实现 |
| 8 | `console.log(...)` | 打印调用参数 |
| 9 | `var ret_value = this.fun(x, y);` | 调用原始方法 |
| 10 | `return ret_value` | 返回原始结果 |

## 修改参数

```javascript
function main() {
    Java.perform(function() {
        var MainActivity = Java.use('com.roysue.demo02.MainActivity');
        
        MainActivity.fun.implementation = function(x, y) {
            console.log("Original args: x =>", x, ", y =>", y);
            
            // 修改参数值
            var new_x = 2;
            var new_y = 5;
            console.log("Modified args: x =>", new_x, ", y =>", new_y);
            
            // 用修改后的参数调用原始方法
            var ret_value = this.fun(new_x, new_y);
            return ret_value;
        };
    });
}
setImmediate(main);
```

> **来源**：`Chap03/1.js` 第 27-39 行（`change_args` 函数），展示了参数篡改。

### 实战场景：参数篡改

```javascript
// 场景：篡改登录参数
Java.perform(function() {
    var LoginActivity = Java.use('com.example.app.LoginActivity');
    
    LoginActivity.login.implementation = function(username, password) {
        console.log("[*] Original login attempt:");
        console.log("    username:", username);
        console.log("    password:", password);
        
        // 篡改密码为万能密码
        var hackedPassword = "admin' OR '1'='1";
        console.log("[*] Modified password:", hackedPassword);
        
        return this.login(username, hackedPassword);
    };
});
```

## 修改返回值

```javascript
function main() {
    Java.perform(function() {
        var MainActivity = Java.use('com.roysue.demo02.MainActivity');
        
        MainActivity.fun.implementation = function(x, y) {
            console.log("x =>", x, ", y =>", y);
            var ret_value = this.fun(x, y);
            
            // 修改返回值
            var modified_ret = "<MODIFIED>" + ret_value;
            console.log("Original return:", ret_value);
            console.log("Modified return:", modified_ret);
            
            return modified_ret;  // 返回修改后的值
        };
    });
}
setImmediate(main);
```

### 实战场景：返回值篡改

```javascript
// 场景：绕过 VIP 检查
function bypassVIPCheck() {
    Java.perform(function() {
        var UserManager = Java.use('com.example.app.UserManager');
        
        UserManager.isVip.implementation = function() {
            console.log("[*] isVip() called - returning true");
            return true;  // 总是返回 VIP 状态
        };
        
        UserManager.getCoinBalance.implementation = function() {
            var original = this.getCoinBalance();
            console.log("[*] getCoinBalance() called - original:", original);
            return 999999;  // 返回大量虚拟币
        };
    });
}
setImmediate(bypassVIPCheck);
```

## 保留原始方法引用的陷阱

```javascript
// ❌ 错误：在 implementation 中使用原始方法引用方式不当
Java.perform(function() {
    var Utils = Java.use('com.example.Utils');
    var originalMethod = Utils.getSecret;  // 保存引用
    
    Utils.getSecret.implementation = function() {
        // ❌ 错误！函数已经被替换，这里会无限递归
        // return originalMethod();  
        
        // ✅ 正确：使用 this 调用原始方法
        return this.getSecret();
    };
});
```

## 返回值的类型处理

| 原始返回值 | Hook 中可返回的类型 | 示例 |
|-----------|-------------------|------|
| int | number | `return 0;` |
| boolean | boolean | `return true;` |
| String | string | `return "hacked";` |
| 对象 | Java.use 的对象 | `return Java.use('java.lang.String').$new("x");` |
| void | 不返回 | `return;` |

```javascript
// 不同类型返回值的处理
Java.perform(function() {
    var Target = Java.use('com.roysue.demo02.MainActivity');
    
    // 返回 int
    Target.getInt.implementation = function() {
        return 12345;
    };
    
    // 返回 boolean
    Target.isValid.implementation = function() {
        return true;
    };
    
    // 返回 String
    Target.getName.implementation = function() {
        var String = Java.use('java.lang.String');
        return String.$new("FridaHooked");
    };
    
    // 返回对象
    Target.getData.implementation = function() {
        return null;  // 或返回一个新的 Java 对象
    };
});
```

## 高级：条件拦截

```javascript
function main() {
    Java.perform(function() {
        var MainActivity = Java.use('com.roysue.demo02.MainActivity');
        
        MainActivity.fun.implementation = function(x, y) {
            console.log(`fun(${x}, ${y})`);
            
            // 根据条件决定是否拦截
            if (x > 100 && y > 100) {
                console.log("[!] Large values detected, intercepting");
                return "INTERCEPTED";  // 直接返回，不调用原方法
            }
            
            // 正常调用
            var result = this.fun(x, y);
            
            // 根据返回值条件修改
            if (result === "error") {
                console.log("[!] Error result detected, modifying");
                result = "success";
            }
            
            return result;
        };
    });
}
setImmediate(main);
```

## 完整实战：Hook 加密方法

```javascript
function hookEncryption() {
    Java.perform(function() {
        var CryptoUtil = Java.use('com.example.app.util.CryptoUtil');
        
        // Hook 加密方法，查看明文和密文
        CryptoUtil.encrypt.implementation = function(plaintext, key) {
            console.log("[*] encrypt() called");
            console.log("    plaintext:", plaintext);
            console.log("    key:", key);
            
            var ciphertext = this.encrypt(plaintext, key);
            
            console.log("    ciphertext:", ciphertext);
            
            // 可以通过 send 发送到 Python 客户端
            send({
                event: "encryption",
                plaintext: plaintext,
                key: key,
                ciphertext: ciphertext
            });
            
            return ciphertext;
        };
        
        // Hook 解密方法
        CryptoUtil.decrypt.implementation = function(ciphertext, key) {
            console.log("[*] decrypt() called");
            console.log("    ciphertext:", ciphertext);
            
            var plaintext = this.decrypt(ciphertext, key);
            
            console.log("    plaintext:", plaintext);
            
            return plaintext;
        };
    });
}
setImmediate(hookEncryption);
```

## 小结

| 操作 | 语法 | 用途 |
|------|------|------|
| 查看参数 | `console.log(arg1, arg2)` | 调试分析 |
| 修改参数 | 在调用原方法前替换变量 | 参数注入 |
| 修改返回值 | 在 `return` 前替换值 | 绕过检查 |
| 完全替换 | 不调用 `this.method()` | 彻底控制 |
| 条件拦截 | 根据条件判断是否调用原方法 | 针对特定输入 |

> **核心原则**：`this.methodName(args)` 调用原始方法，`return` 语句控制返回值。Hook 的本质是在这两者之间插入你的逻辑。
