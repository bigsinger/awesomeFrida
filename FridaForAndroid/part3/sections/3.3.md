# 3.3 重载函数 Hook

## 什么是重载？

Java 中的 **方法重载（Overload）** 指同一个方法名有多个不同参数列表的实现。例如：

```java
// Java 重载示例
public class MainActivity {
    public String fun(int x, int y) { ... }
    public String fun(String x, String y) { ... }
    public String fun(int x, int y, int z) { ... }
    public String fun(double x, double y) { ... }
}
```

如果直接使用 `implementation` 而不指定参数类型，Frida 会不知道要 Hook 哪个版本。

## overload 语法

```javascript
ClassName.method.overload('paramType1', 'paramType2', ...).implementation = function(...) {
    // Hook 逻辑
};
```

## 参数类型映射

Java 类型必须转换为 Frida 能识别的字符串格式：

| Java 类型 | Frida 类型字符串 |
|-----------|-----------------|
| `int` | `'int'` |
| `long` | `'long'` |
| `double` | `'double'` |
| `float` | `'float'` |
| `boolean` | `'boolean'` |
| `char` | `'char'` |
| `byte` | `'byte'` |
| `short` | `'short'` |
| `String` | `'java.lang.String'` |
| `Object` | `'java.lang.Object'` |
| `int[]` | `'[I'` |
| `String[]` | `'[Ljava.lang.String;'` |
| `List<String>` | `'java.util.List'` |

## 基本重载 Hook

```javascript
// 引用自 Chap03/2.js 第 1-14 行（overload 用法）
function main(){
    console.log("Script loaded successfully ")
    Java.perform(function(){
        console.log("Inside java perform function")

        // hook 重载函数：fun(int, int)

            console.log("orignal args: x => ",x,", y => ",y)

        }
    })
}
setImmediate(main)
```

## 完整多重重载 Hook

```javascript
function main() {
    Java.perform(function() {
        var MainActivity = Java.use('com.roysue.demo02.MainActivity');
        
        // 1. fun(int, int)
        MainActivity.fun.overload('int', 'int').implementation = function(x, y) {
            console.log("[int, int] fun(", x, ",", y, ")");
            var ret = this.fun(x, y);
            console.log("  =>", ret);
            return ret;
        };
        
        // 2. fun(String, String)
        MainActivity.fun.overload('java.lang.String', 'java.lang.String')
            .implementation = function(a, b) {
                console.log("[String, String] fun(", a, ",", b, ")");
                var ret = this.fun(a, b);
                console.log("  =>", ret);
                return ret;
            };
        
        // 3. fun(int, int, int)
        MainActivity.fun.overload('int', 'int', 'int')
            .implementation = function(x, y, z) {
                console.log("[int, int, int] fun(", x, ",", y, ",", z, ")");
                var ret = this.fun(x, y, z);
                console.log("  =>", ret);
                return ret;
            };
        
        // 4. fun(double, double)
        MainActivity.fun.overload('double', 'double').implementation = function(x, y) {
            console.log("[double, double] fun(", x, ",", y, ")");
            var ret = this.fun(x, y);
            console.log("  =>", ret);
            return ret;
        };
    });
}
setImmediate(main);
```

## 自动枚举所有重载

如果不确定目标方法有哪些重载版本，可以自动枚举：

```javascript
function enumOverloads(className, methodName) {
    Java.perform(function() {
        var clazz = Java.use(className);
        
        // 获取所有方法
        var methods = clazz.class.getDeclaredMethods();
        for (var i = 0; i < methods.length; i++) {
            var method = methods[i];
            if (method.getName() === methodName) {
                var paramTypes = method.getParameterTypes();
                var params = [];
                for (var j = 0; j < paramTypes.length; j++) {
                    params.push(paramTypes[j].getName());
                }
                console.log(`[${i}] ${methodName}(${params.join(', ')})`);
            }
        }
    });
}

// 使用
enumOverloads('com.roysue.demo02.MainActivity', 'fun');
```

## 批量 Hook 所有重载

```javascript
function hookAllOverloads(className, methodName) {
    Java.perform(function() {
        var clazz = Java.use(className);
        var targetMethod = clazz[methodName];
        
        // 如果方法没有重载，直接 Hook
        if (targetMethod.overloads === undefined) {
            targetMethod.implementation = function() {
                console.log(`[${methodName}] called`);
                return this[methodName].apply(this, arguments);
            };
            return;
        }
        
        // 枚举所有重载
        var overloads = targetMethod.overloads;
        console.log(`[+] Found ${overloads.length} overload(s) of ${methodName}`);
        
        for (var i = 0; i < overloads.length; i++) {
            overloads[i].implementation = function() {
                var args = Array.prototype.slice.call(arguments);
                console.log(`[${methodName}#${i}] args:`, args.map(String).join(', '));
                
                // 调用原始方法
                var result = this[methodName].apply(this, arguments);
                
                console.log(`[${methodName}#${i}] =>`, result);
                return result;
            };
        }
    });
}

// 使用
hookAllOverloads('com.roysue.demo02.MainActivity', 'fun');
```

## 实战：Hook 常见的系统类重载

### Toast.makeText

```javascript
Java.perform(function() {
    var Toast = Java.use('android.widget.Toast');
    
    // Hook makeText(Context, CharSequence, int)
    Toast.makeText.overload('android.content.Context', 
                            'java.lang.CharSequence', 'int')
        .implementation = function(context, text, duration) {
            console.log("[Toast] makeText:", text);
            
            // 可以修改 Toast 内容
```

## 重载错误排查

### 错误 1：overload 类型不匹配

```javascript
// ❌ 错误：类型字符串写错
// fun.overload('Integer', 'Integer')  // 应该用 'int'

// ✅ 正确
fun.overload('int', 'int')
```

### 错误 2：忘记处理所有重载

```javascript
// ❌ 如果只 Hook 了一个重载，其他重载不受影响
// fun.overload('int', 'int').implementation = ...

// ✅ 批量处理全部重载
// 使用 hookAllOverloads 或逐个处理
```

### 错误 3：在 overload 中调用 this.method 不带参数

```javascript
// ❌ 错误
fun.overload('int', 'int').implementation = function(x, y) {
    return this.fun();  // 缺少参数，调用的是无参版本
};

// ✅ 正确
fun.overload('int', 'int').implementation = function(x, y) {
    return this.fun(x, y);  // 传回所有参数
};
```

## 小结

| 概念 | 说明 |
|------|------|
| `overload('type1', 'type2')` | 指定重载版本的参数类型 |
| `overloads[]` | 自动获取所有重载列表 |
| 类型映射 | Java 类型 → Frida 字符串格式 |
| 枚举重载 | 通过 `class.getDeclaredMethods()` 发现 |

> **核心**：有重载就用 `overload()` 明确指定参数类型，否则 Frida 无法确定要 Hook 哪个版本。
