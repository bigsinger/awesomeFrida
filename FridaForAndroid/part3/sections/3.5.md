# 3.5 修改成员变量与字段

## 概述

除了 Hook 方法，Frida 还可以直接 **读取和修改** Java 对象的成员变量（字段）。这在对安全验证、状态标记等场景非常有用。

## 字段访问语法

```javascript
// 读取字段
var value = instance.fieldName.value;

// 修改字段
instance.fieldName.value = new_value;
```

## 基本示例：读取字段

```javascript
// 引用自 Chap03/4.js 第 17-37 行（getTotalValue 函数，字段访问）
function getTotalValue(){
    Java.perform(function(){
        var MainAcitivity = Java.use('com.roysue.demo02.MainActivity')
        
        // 动态函数主动调用
        Java.choose('com.roysue.demo02.MainActivity',{
            onMatch: function(instance){
                console.log('total value = ',instance.total.value)
            },
            onComplete: function(){
                console.log('search Complete')
            }
        })
    })
}
setImmediate(getTotalValue)
```

> **来源**：`Chap03/4.js` 第 17-37 行（`getTotalValue` 函数），展示了通过 `Java.choose` 获取实例后读取字段的值。

## 修改字段值

```javascript
function modifyField() {
    Java.perform(function() {
        Java.choose('com.roysue.demo02.MainActivity', {
            onMatch: function(instance) {
                console.log("[*] Found instance, current total =", 
                          instance.total.value);
                
                // 修改字段值
                instance.total.value = 999999;
                
                console.log("[*] Modified total to:", 
                          instance.total.value);
                
                // 修改 String 类型字段
                if (instance.name) {
                    console.log("[*] Original name:", instance.name.value);
                    var String = Java.use('java.lang.String');
                    instance.name.value = String.$new("FridaModified");
                    console.log("[*] Modified name:", instance.name.value);
                }
            },
            onComplete: function() {
                console.log("[*] Field modification complete");
            }
        });
    });
}
setImmediate(modifyField);
```

## 在方法 Hook 中修改字段

```javascript
function main() {
    Java.perform(function() {
        var MainActivity = Java.use('com.roysue.demo02.MainActivity');
        
        MainActivity.fun.implementation = function(x, y) {
            console.log("fun(", x, ",", y, ")");
            
            // 在方法执行中修改字段
            // 可以通过 this 访问当前实例
            console.log("Current total before:", this.total.value);
            
            // 修改字段
            this.total.value = this.total.value + 1;
            
            var result = this.fun(x, y);
            
            console.log("Current total after:", this.total.value);
            
            return result;
        };
    });
}
setImmediate(main);
```

## 静态字段处理

静态字段的访问方式与实例字段略有不同：

```javascript
function hookStaticField() {
    Java.perform(function() {
        var MainActivity = Java.use('com.roysue.demo02.MainActivity');
        
        // 读取静态字段（通过类本身访问）
        console.log("[*] Static field value:", 
                   MainActivity.STATIC_COUNTER.value);
        
        // 修改静态字段
        MainActivity.STATIC_COUNTER.value = 0;
        console.log("[*] Static field reset to:", 
                   MainActivity.STATIC_COUNTER.value);
        
        // Hook 静态方法并在其中修改静态字段
        MainActivity.resetCounter.implementation = function() {
            console.log("[*] resetCounter() called");
            
            // 修改静态字段
            MainActivity.STATIC_COUNTER.value = 0;
            
            return this.resetCounter();
        };
    });
}
setImmediate(hookStaticField);
```

## 私有字段操作

```java
// Java 中的私有字段
public class MainActivity {
    private String secretKey = "default_key";
    private int[] secureData = {1, 2, 3, 4};
    private boolean isLicensed = false;
}
```

```javascript
function accessPrivateField() {
    Java.perform(function() {
        Java.choose('com.roysue.demo02.MainActivity', {
            onMatch: function(instance) {
                // 私有字段同样可以访问
                console.log("[*] Private secretKey:", 
                          instance.secretKey.value);
                
                // 修改私有字段
                var String = Java.use('java.lang.String');
                instance.secretKey.value = String.$new("hacked_key");
                
                // 修改私有 boolean 字段
                instance.isLicensed.value = true;
                console.log("[*] isLicensed set to:", 
                          instance.isLicensed.value);
                
                // 修改 int 数组（第0个元素）
                if (instance.secureData.value) {
                    var arr = instance.secureData.value;
                    console.log("[*] Original array:", arr);
                    // Java 数组不能直接修改，需要通过 set operations
                }
            },
            onComplete: function() {}
        });
    });
}
setImmediate(accessPrivateField);
```

## 字段类型对应关系

| Java 字段类型 | Frida 读取 | Frida 写入 |
|:---:|:---:|:---:|
| `int` | `instance.field.value` → Number | `instance.field.value = 42` |
| `long` | 同上 | `instance.field.value = 999999999` |
| `boolean` | `instance.field.value` → Boolean | `instance.field.value = true` |
| `float` | `instance.field.value` → Number | `instance.field.value = 3.14` |
| `String` | `instance.field.value` → String | `instance.field.value = Java.use('java.lang.String').$new("x")` |
| `Object` | `instance.field.value` → Object | `instance.field.value = obj` |
| `int[]` | `instance.field.value` → Array | 需通过 Java 方法修改 |
| `List` | `instance.field.value` → Java List | `instance.field.value.add(obj)` |

## 实战：绕过许可证检查

```javascript
function bypassLicenseCheck() {
    Java.perform(function() {
        Java.choose('com.example.app.LicenseManager', {
            onMatch: function(instance) {
                console.log("[*] LicenseManager instance found");
                
                // 修改所有许可证相关字段
                instance.isLicensed.value = true;
                instance.isPro.value = true;
                instance.isTrial.value = false;
                instance.expiryDate.value = "2099-12-31";
                instance.trialCount.value = 999;
                
                console.log("[*] License bypassed!");
            },
            onComplete: function() {}
        });
        
        // 也可以在方法 Hook 中动态修改
        var LicenseManager = Java.use('com.example.app.LicenseManager');
        
        LicenseManager.checkLicense.implementation = function() {
            console.log("[*] checkLicense() called - returning true");
            return true;
        };
    });
}
setImmediate(bypassLicenseCheck);
```

## 字段变化监控

```javascript
function monitorFieldChanges() {
    Java.perform(function() {
        var AtomicInteger = Java.use('java.util.concurrent.atomic.AtomicInteger');
        
        // 监控 AtomicInteger 的变化——通过 Hook 其 set 方法
        AtomicInteger.set.implementation = function(newValue) {
            console.log("[AtomicInteger] set(" + newValue + ")");
            
            // 可以在这里记录状态变化
            var currentValue = this.get();
            console.log("    was: " + currentValue + ", now: " + newValue);
            
            return this.set(newValue);
        };
        
        // 通过 Hook getter/setter 来监控字段访问
        var MainActivity = Java.use('com.roysue.demo02.MainActivity');
        
        // 假设该类有 setTotal/getTotal 方法
        if (MainActivity.setTotal) {
            MainActivity.setTotal.overload('int').implementation = function(value) {
                console.log("[Field Monitor] setTotal(" + value + ")");
                return this.setTotal(value);
            };
        }
        
        if (MainActivity.getTotal) {
            MainActivity.getTotal.implementation = function() {
                var value = this.getTotal();
                console.log("[Field Monitor] getTotal() => " + value);
                return value;
            };
        }
    });
}
setImmediate(monitorFieldChanges);
```

## 字段与 Java.choose 配合

`Java.choose` 是访问 **对象字段** 的关键入口：

```javascript
function inspectAllInstances() {
    Java.perform(function() {
        // 遍历所有实例
        Java.choose('com.roysue.demo02.MainActivity', {
            onMatch: function(instance) {
                console.log("=== Instance " + instance.hashCode() + " ===");
                
                // 枚举所有字段
                var fields = instance.getClass().getDeclaredFields();
                for (var i = 0; i < fields.length; i++) {
                    var field = fields[i];
                    field.setAccessible(true);  // 访问私有字段
                    
                    var name = field.getName();
                    var value = instance[name].value;
                    
                    console.log("  " + name + " = " + value);
                }
            },
            onComplete: function() {
                console.log("[*] Inspection complete");
            }
        });
    });
}
setImmediate(inspectAllInstances);
```

## 小结

| 操作 | 语法 | 说明 |
|------|------|------|
| 读字段 | `instance.field.value` | 获取当前值 |
| 写字段 | `instance.field.value = newVal` | 设置新值 |
| 静态字段 | `ClassName.STATIC_FIELD.value` | 通过类直接访问 |
| 私有字段 | 同上（Frida 无视访问权限） | 所有字段都可访问 |
| String 字段 | 需要 `Java.use('java.lang.String').$new()` | 通过 Java 字符串包装 |
| 获取实例 | `Java.choose()` | 枚举堆中对象 |

> **核心**：`.value` 是字段读写的统一入口。`Java.choose` + `instance.field.value` 组合可以操作任意 Java 对象的任意字段（包括私有字段）。
