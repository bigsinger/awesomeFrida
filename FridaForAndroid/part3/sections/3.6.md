# 3.6 枚举类与动态加载类

## 概述

在实际逆向中，你经常需要：

1. **发现目标 App 加载了哪些类** — 用于探索和定位
2. **监控动态加载的类** — 通过 `DexClassLoader` 在运行时加载的类
3. **Hook 系统类** — 如 HashMap、TextUtils 等

本章涵盖这三种场景的完整解决方案。

## 枚举所有已加载的类

### 基本枚举

```javascript
// 引用自 awesomeFrida/all_classes.js 第 1-12 行（完整代码）
function hook() {
  Java.perform(function () {
    Java.enumerateLoadedClasses({
      onMatch: function (className) {
        console.log(className);
      },
      onComplete: function () { }
    });
  });
}

setImmediate(hook());
```

> **来源**：`awesomeFrida/all_classes.js` 第 1-12 行（完整文件），展示了 `Java.enumerateLoadedClasses` 的标准用法。

### 带过滤的枚举

```javascript
function enumerateFiltered(prefix) {
    Java.perform(function() {
        console.log("[*] Enumerating classes with prefix: " + prefix);
        var count = 0;
        
        Java.enumerateLoadedClasses({
            onMatch: function(className) {
                // 只显示特定包名的类
                if (className.indexOf(prefix) === 0) {
                    console.log("  " + className);
                    count++;
                }
            },
            onComplete: function() {
                console.log("[*] Found " + count + " classes");
            }
        });
    });
}

// 使用示例
// enumerateFiltered('com.roysue');
// enumerateFiltered('okhttp3');
// enumerateFiltered('com.example');
```

### 分类统计

```javascript
function classStatistics() {
    Java.perform(function() {
        var stats = {};
        
        Java.enumerateLoadedClasses({
            onMatch: function(className) {
                // 提取包名首位
                var parts = className.split('.');
                var topLevel = parts.length > 0 ? parts[0] : "(default)";
                
                if (!stats[topLevel]) {
                    stats[topLevel] = 0;
                }
                stats[topLevel]++;
            },
            onComplete: function() {
                console.log("[*] Class statistics by top-level package:");
                var packages = Object.keys(stats).sort();
                for (var i = 0; i < packages.length; i++) {
                    var pkg = packages[i];
                    console.log("  " + pkg + ": " + stats[pkg] + " classes");
                }
                
                // 显示总类数
                var total = packages.reduce(function(sum, pkg) {
                    return sum + stats[pkg];
                }, 0);
                console.log("[*] Total: " + total + " classes");
            }
        });
    });
}
setImmediate(classStatistics);
```

## 动态加载类监控

### Hook DexClassLoader.loadClass

```javascript
// 引用自 awesomeFrida/loadClass.js 第 1-63 行（核心结构）
function hook() {
  Java.perform(function () {
    var dexclassLoader = Java.use("dalvik.system.DexClassLoader");
    
    // Hook loadClass 方法，监控哪些类被动态加载
    dexclassLoader.loadClass.overload('java.lang.String')
        .implementation = function (name) {
      var result = this.loadClass(name);
      console.log(name);
      
      // 当加载到 Application 类时，自动 Hook
      if (name.indexOf("Application") != -1 && 
          name != "android.app.Application") {
        var application = Java.use(name);
        console.log("application:" + application);
        
        // Hook Application.onCreate
        application.onCreate.implementation = function () {
          var ret = this.onCreate();
          hookTargetClass(); // 在 Application 初始化后 Hook 目标类
          return ret;
        }
      }
      
      return result;
    }
    
    // 也 Hook Application.attach 确保早期 Hook
    var application = Java.use('android.app.Application');
    console.log("application:" + application);
    
    application.attach.overload('android.content.Context')
        .implementation = function (context) {
      console.log("application attach called");
      var result = this.attach(context);
      
      hookTargetClass(); // 在 attach 时 Hook 目标类
      
      return result;
    }
    
    application.onCreate.implementation = function () {
      console.log("application onCreate called");
      hookTargetClass(); // 二选一均可
      var result = this.onCreate();
      return result;
    }
  });
}

function hookTargetClass() {
    // 在这里添加你要 Hook 的目标类
    // 例如：
    // var Target = Java.use('com.example.dynamic.DynamicClass');
    // Target.method.implementation = function() { ... };
}

// 打印堆栈
function printStack() {
  Java.perform(function () {
    console.log(
      Java.use("android.util.Log")
        .getStackTraceString(Java.use("java.lang.Exception").$new())
    );
  });
}

setImmediate(hook());
```

> **来源**：`awesomeFrida/loadClass.js` 第 1-63 行（完整文件），展示了 `DexClassLoader.loadClass` Hook 的完整实现。

## 系统类 Hook 实战

### HashMap.put 监控

```javascript
// 引用自 awesomeFrida/HashMap.js 第 1-22 行（完整代码）
Java.perform(function(){
	var textUtils = Java.use("android.text.TextUtils")
	textUtils.isEmpty.overload('java.lang.CharSequence')
        .implementation = function(a) {
		printStack();
		console.log("TextUtils.isEmpty: " + a);
		return this.isEmpty(a);
	};
	
	var hashMap = Java.use("java.util.HashMap");
	hashMap.put.implementation = function(key, value) {
		console.log("HashMap.put key: " + key + " value: " + value);
		if(key.equals("targetKey")){
			printStack();
		}
		return this.put(key, value);
	};
});

function printStack() {
  Java.perform(function () {
    console.log(
      Java.use("android.util.Log")
        .getStackTraceString(Java.use("java.lang.Exception").$new())
    );
  });
}
```

### 其他常用系统类 Hook

```javascript
// SharedPreferences 监控
function hookSharedPreferences() {
    Java.perform(function() {
        var SharedPreferences = Java.use('android.content.SharedPreferences');
        var Editor = Java.use('android.content.SharedPreferences$Editor');
        
        // Hook 读取
        SharedPreferences.getString.implementation = function(key, defValue) {
            var value = this.getString(key, defValue);
            console.log("[SharedPrefs] getString(" + key + ") => " + value);
            return value;
        };
        
        // Hook 写入
        Editor.putString.overload('java.lang.String', 'java.lang.String')
            .implementation = function(key, value) {
                console.log("[SharedPrefs] putString(" + key + ", " + value + ")");
                return this.putString(key, value);
            };
        
        Editor.commit.implementation = function() {
            console.log("[SharedPrefs] commit()");
            return this.commit();
        };
    });
}
setImmediate(hookSharedPreferences);
```

## 在枚举结果中找到目标后自动 Hook

```javascript
function autoHookByClassPrefix(prefix, hookFunction) {
    Java.perform(function() {
        Java.enumerateLoadedClasses({
            onMatch: function(className) {
                if (className.indexOf(prefix) === 0) {
                    console.log("[+] Found target class: " + className);
                    try {
                        var clazz = Java.use(className);
                        hookFunction(className, clazz);
                    } catch(e) {
                        console.log("[-] Error hooking " + className + ": " + e);
                    }
                }
            },
            onComplete: function() {
                console.log("[*] Auto-hook scan complete");
            }
        });
    });
}

// 使用
autoHookByClassPrefix('com.example.app.utils', function(name, clazz) {
    // 自动 Hook 该类中的所有方法
    console.log("[*] Hooking class: " + name);
    
    // 这里可以添加针对该类的方法 Hook
});
```

## 实战：脱壳前的类枚举

在脱壳前，枚举所有已加载类可以帮助确认脱壳是否成功：

```javascript
function dumpAllClasses() {
    var allClasses = [];
    
    Java.perform(function() {
        Java.enumerateLoadedClasses({
            onMatch: function(className) {
                allClasses.push(className);
            },
            onComplete: function() {
                // 输出到控制台
                console.log("[*] Total classes: " + allClasses.length);
                
                // 通过 send 发送到 Python 客户端
                send({
                    event: "class_dump",
                    total: allClasses.length,
                    classes: allClasses
                });
                
                // 输出到本地文件（在模拟器环境下）
                allClasses.forEach(function(cls) {
                    console.log(cls);
                });
            }
        });
    });
}

// 延迟执行，确保所有 DEX 都已加载
setTimeout(function() {
    dumpAllClasses();
}, 3000);
```

## 常见问题

### Q1: enumerateLoadedClasses 输出的类名不全

```javascript
// 原因是枚举的时机太早，DEX 尚未完全加载
// 解决方案：延迟执行
setTimeout(function() {
    Java.perform(function() {
        Java.enumerateLoadedClasses({ /* ... */ });
    });
}, 5000);  // 延迟 5 秒
```

### Q2: Java.use 找不到动态加载的类

```javascript
// 解决：在 DexClassLoader.loadClass 被调用后 Hook
var originalLoadClass = DexClassLoader.loadClass.overload('java.lang.String');
originalLoadClass.implementation = function(name) {
    var result = this.loadClass(name);
    
    // 当目标类被加载时，立即 Hook
    if (name.indexOf('com.example.dynamic') === 0) {
        setTimeout(function() {
            Java.perform(function() {
                var Target = Java.use(name);
                console.log("[+] Dynamically loaded and hooked: " + name);
                // 现在可以安全地 Hook 这个类了
            });
        }, 100);
    }
    
    return result;
};
```

### Q3: 枚举时 App 崩溃

```javascript
// 在 Java.perform 中添加 try-catch 保护
Java.perform(function() {
    Java.enumerateLoadedClasses({
        onMatch: function(className) {
            try {
                // 你的处理逻辑
                console.log(className);
            } catch(e) {
                // 跳过有问题的类
            }
        },
        onComplete: function() {}
    });
});
```

## 小结

| 技术 | API | 适用场景 |
|------|-----|---------|
| 枚举已加载类 | `Java.enumerateLoadedClasses()` | 发现目标类、统计类数量 |
| 监控动态加载 | `DexClassLoader.loadClass` Hook | 拦截动态加载的类 |
| 系统类 Hook | `Java.use('java.util.HashMap')` | 监控全局行为 |
| 自动 Hook | `onMatch` 回调 + `Java.use()` | 批量自动化 Hook |
| 过滤枚举 | `className.indexOf(prefix)` | 缩小范围、精准定位 |

> **核心**：枚举找到目标类 → `Java.use()` 定位 → Hook。动态加载的类需要通过 `DexClassLoader.loadClass` 先捕获加载事件。
