# 3.4 Hook 构造函数

## 概述

在 Frida 中，Java 类的构造函数通过特殊属性 `$init` 来访问。Hook 构造函数可以让你：

- 拦截对象的创建过程
- 查看和修改构造参数
- 阻止对象的创建（通过抛出异常）
- 在对象初始化时执行自定义逻辑

## 基本语法

```javascript
ClassName.$init.overload('paramTypes...').implementation = function(args...) {
    // 构造参数已传入
    // this 指向正在创建的对象（尚未完全初始化）
    console.log("Constructor called with:", args);
    
    // 可选的：调用原始构造函数（必须在最后）
    // 注意：$init 的原始调用是隐式的，不需要显式调用 this.$init()
};
```

## 基础示例

```javascript
// 引用自 Chap03/hello_world.js 第 1-5 行 + 构造函数调用概念
// 最简单的入口结构，为构造函数 Hook 做准备
setTimeout(function(){
    Java.perform(function(){
        console.log("hello world!");
    });
});
```

> 构造函数 Hook 的核心是 `$init` 属性，下面展示完整示例。

```javascript
function hookConstructor() {
    Java.perform(function() {
        var MainActivity = Java.use('com.roysue.demo02.MainActivity');
        
        // Hook 无参构造函数
        MainActivity.$init.overload().implementation = function() {
            console.log("[*] MainActivity() constructor called");
            
            // 调用原始构造函数
            this.$init();
            
            console.log("[*] MainActivity() constructor finished");
        };
        
        // Hook 带参构造函数
        MainActivity.$init.overload('int').implementation = function(x) {
            console.log("[*] MainActivity(int) constructor called with:", x);
            
            // 可以修改参数
            var modified_x = x * 2;
            console.log("    Modified argument:", modified_x);
            
            // 用修改后的参数调用原始构造函数
            this.$init(modified_x);
        };
    });
}
setImmediate(hookConstructor);
```

## 实例创建与监控

### 拦截所有 Activity 的创建

```javascript
// Hook Activity 的构造函数来监控 Activity 创建
function monitorActivityCreation() {
    Java.perform(function() {
        var Activity = Java.use('android.app.Activity');
        
        Activity.$init.overload().implementation = function() {
            console.log("[Activity] New activity created");
            this.$init();
        };
        
        // Hook onCreate 作为替代方案
        Activity.onCreate.overload('android.os.Bundle')
            .implementation = function(savedInstanceState) {
                console.log("[Activity] onCreate: " + this.getClass().getName());
                return this.onCreate(savedInstanceState);
            };
    });
}
setImmediate(monitorActivityCreation);
```

### 监控自定义类的实例化

```javascript
function monitorCustomClass() {
    Java.perform(function() {
        var TargetClass = Java.use('com.roysue.demo02.MainActivity');
        
        // 记录所有构造函数调用
        var constructors = TargetClass.class.getDeclaredConstructors();
        console.log("[+] Found " + constructors.length + " constructor(s)");
        
        // Hook 所有构造函数
        TargetClass.$init.overload('int', 'int').implementation = function(x, y) {
            console.log("[*] MainActivity(" + x + ", " + y + ")");
            
            // 记录创建时间
            var timestamp = new Date().getTime();
            
            // 调用原始构造函数
            this.$init(x, y);
            
            // 构造函数执行完成后
            console.log("[*] MainActivity instance created at:", timestamp);
            console.log("[*] Object hash:", this.hashCode());
            
            // 通过 send 发送给 Python 客户端
            send({
                event: "object_created",
                class_name: "com.roysue.demo02.MainActivity",
                args: {x: x, y: y},
                object_hash: this.hashCode()
            });
        };
    });
}
setImmediate(monitorCustomClass);
```

## 读取构造函数中的参数

```javascript
function interceptConstructorParams() {
    Java.perform(function() {
        // 场景：App 在构造函数中设置密钥等敏感信息
        var SecureConfig = Java.use('com.example.app.SecureConfig');
        
        SecureConfig.$init.overload('java.lang.String', 'java.lang.String')
            .implementation = function(apiKey, secretKey) {
                console.log("[!] SecureConfig constructor intercepted!");
                console.log("    apiKey:", apiKey);
                console.log("    secretKey:", secretKey);
                
                // 保存到全局变量供后续使用
                globalConfig = {
                    apiKey: apiKey,
                    secretKey: secretKey
                };
                
                // 仍然调用原始构造函数
                this.$init(apiKey, secretKey);
            };
    });
}
setImmediate(interceptConstructorParams);
```

## 阻止对象创建

```javascript
function blockObjectCreation() {
    Java.perform(function() {
        var MaliciousClass = Java.use('com.example.app.MaliciousClass');
        
        MaliciousClass.$init.overload('java.lang.String')
            .implementation = function(data) {
                console.log("[!] Blocked MaliciousClass creation with data:", data);
                
                // 不调用 this.$init() — 对象不会被完全构造
                // 注意：这会留下一个半初始化对象，可能导致后续崩溃
                // 更好的做法是抛出异常
                
                throw new Error("Creation blocked by Frida");
            };
    });
}
setImmediate(blockObjectCreation);
```

## 使用 Java.choose 获取现有实例的构造函数信息

```javascript
// 引用自 awesomeFrida/call_constructors.js 中监控对象的概念
// 还可以通过 Java.choose 获取已存在的对象
function inspectExistingInstances() {
    Java.perform(function() {
        Java.choose('com.roysue.demo02.MainActivity', {
            onMatch: function(instance) {
                console.log("[+] Found instance:", instance);
                console.log("    Class:", instance.getClass().getName());
                console.log("    Hash:", instance.hashCode());
                
                // 读取字段值
                try {
                    console.log("    total =", instance.total.value);
                } catch(e) {
                    console.log("    Could not read total field");
                }
            },
            onComplete: function() {
                console.log("[*] Instance enumeration complete");
            }
        });
    });
}
setImmediate(inspectExistingInstances);
```

> **来源**：`awesomeFrida/call_constructors.js` 展示了监控对象创建过程的通用模式。

## 实战：监控 OkHttpClient 创建

```javascript
// 监控 OkHttpClient 的创建，获取其配置
function monitorOkHttpCreation() {
    Java.perform(function() {
        var OkHttpClient = Java.use('okhttp3.OkHttpClient');
        
        // Hook OkHttpClient 的无参构造函数
        OkHttpClient.$init.overload().implementation = function() {
            console.log("[OkHttpClient] Created new instance");
            
            // 调用原始构造函数
            this.$init();
            
            // 之后可以通过 Java.choose 获取该实例
            // 并检查其配置（如 SSL Socket Factory 等）
        };
        
        // Hook Builder 构造函数
        var Builder = Java.use('okhttp3.OkHttpClient$Builder');
        
        Builder.$init.overload().implementation = function() {
            console.log("[OkHttpClient.Builder] Created");
            this.$init();
        };
        
        Builder.build.implementation = function() {
            console.log("[OkHttpClient.Builder] build() called");
            var client = this.build();
            console.log("[OkHttpClient.Builder] Client created:", client);
            return client;
        };
    });
}
setImmediate(monitorOkHttpCreation);
```

## 构造函数 Hook 的限制

1. **不能 Hook 抽象类/接口的构造函数**：它们没有可实例化的构造函数
2. **`this.$init()` 必须在最后调用**：在构造函数中调用原始初始化
3. **不能完全替换实现**：不像普通方法，构造函数必须最终调用 `this.$init()`
4. **匿名内部类的构造函数**：类名格式为 `OuterClass$1`

## 小结

| 概念 | 语法 | 说明 |
|------|------|------|
| 构造函数属性 | `$init` | Hook 构造函数的入口 |
| 无参构造 | `$init.overload()` | 不带参数的构造函数 |
| 带参构造 | `$init.overload('int', 'String')` | 指定参数类型 |
| 调用原始构造 | `this.$init(args)` | 必须在实现中调用 |
| 阻止创建 | 不调用 `this.$init()` | 可能导致异常 |
| 获取实例 | `Java.choose()` | 枚举已存在的对象 |

> **核心**：`ClassName.$init` = 构造函数，`this.$init(args)` = 调用原始构造函数。Hook 构造函数可以拦截对象的创建全过程。
